# 024 - 两阶段分析流程系统

## 概述

实现智能的两阶段分析流程系统，先进行初步快速分析，然后根据用户需求决定是否进入深度分析，提供灵活高效的分析体验。

## 业务需求

### 核心功能

- **初步分析**：快速扫描和基础分析，提供即时反馈
- **深度分析**：详细深入的全面分析，提供专业洞察
- **流程控制**：智能推荐是否需要进入深度分析
- **进度跟踪**：清晰显示分析进度和状态
- **结果对比**：对比初步和深度分析结果

### 分析阶段定义

1. **初步分析阶段（Phase 1）**
   - 快速数据扫描（30秒内完成）
   - 基础模式识别
   - 关键指标提取
   - 初步建议生成

2. **深度分析阶段（Phase 2）**
   - 全面的数据深度挖掘
   - 复杂模式分析
   - 多维度交叉验证
   - 专业级建议生成

```typescript
// src/analysis/TwoPhaseEngine.ts
export interface AnalysisRequest {
  data: any;
  context: AnalysisContext;
  options: AnalysisOptions;
  userPreferences: UserPreferences;
}

export interface AnalysisContext {
  urgency: 'low' | 'medium' | 'high';
  complexity: 'simple' | 'medium' | 'complex';
  domain: string;
  objectives: string[];
  constraints: string[];
}

export interface AnalysisOptions {
  enableDeepAnalysis: boolean;
  autoProgression: boolean;
  timeout: number;
  maxDepth: number;
  focusAreas: string[];
}

export class TwoPhaseAnalysisEngine {
  private phase1Processor: Phase1Processor;
  private phase2Processor: Phase2Processor;
  private decisionEngine: ProgressionDecisionEngine;

  async executeAnalysis(request: AnalysisRequest): Promise<AnalysisResult> {
    // Phase 1: 初步分析
    const phase1Result = await this.executePhase1(request);

    // 决定是否进入Phase 2
    const shouldProgress = await this.decideProgression(phase1Result, request);

    if (!shouldProgress) {
      return {
        phase: 'phase1',
        result: phase1Result,
        recommendation: this.generateRecommendation(phase1Result),
        nextSteps: this.suggestNextSteps(phase1Result),
      };
    }

    // Phase 2: 深度分析
    const phase2Result = await this.executePhase2(phase1Result, request);

    return {
      phase: 'phase2',
      result: phase2Result,
      phase1Result,
      comparison: this.compareResults(phase1Result, phase2Result),
      recommendation: this.generateFinalRecommendation(phase2Result),
    };
  }

  private async executePhase1(request: AnalysisRequest): Promise<Phase1Result> {
    const startTime = Date.now();

    try {
      // 快速数据预处理
      const preprocessedData = await this.phase1Processor.preprocess(
        request.data
      );

      // 基础分析
      const basicInsights =
        await this.phase1Processor.analyzeBasic(preprocessedData);

      // 模式识别
      const patterns =
        await this.phase1Processor.identifyPatterns(preprocessedData);

      // 初步建议
      const initialRecommendations =
        await this.phase1Processor.generateRecommendations(
          basicInsights,
          patterns,
          request.context
        );

      return {
        insights: basicInsights,
        patterns,
        recommendations: initialRecommendations,
        confidence: this.calculatePhase1Confidence(basicInsights, patterns),
        processingTime: Date.now() - startTime,
        dataQuality: this.assessDataQuality(preprocessedData),
      };
    } catch (error) {
      throw new AnalysisError('Phase 1 analysis failed', error);
    }
  }

  private async executePhase2(
    phase1Result: Phase1Result,
    request: AnalysisRequest
  ): Promise<Phase2Result> {
    const startTime = Date.now();

    try {
      // 深度数据挖掘
      const deepInsights = await this.phase2Processor.deepAnalyze(
        request.data,
        phase1Result.insights
      );

      // 交叉验证
      const validationResults = await this.phase2Processor.crossValidate(
        deepInsights,
        request.context
      );

      // 多维度分析
      const multidimensionalAnalysis =
        await this.phase2Processor.multidimensionalAnalyze(
          request.data,
          request.focusAreas
        );

      // 专业级建议
      const expertRecommendations =
        await this.phase2Processor.generateExpertRecommendations(
          deepInsights,
          validationResults,
          multidimensionalAnalysis,
          request.context
        );

      return {
        deepInsights,
        validationResults,
        multidimensionalAnalysis,
        expertRecommendations,
        confidence: this.calculatePhase2Confidence(
          deepInsights,
          validationResults
        ),
        processingTime: Date.now() - startTime,
        qualityScore: this.calculateQualityScore(
          deepInsights,
          validationResults
        ),
      };
    } catch (error) {
      throw new AnalysisError('Phase 2 analysis failed', error);
    }
  }
}
```

```typescript
// src/analysis/ProgressionDecisionEngine.ts
export interface ProgressionCriteria {
  dataComplexity: number;
  confidenceThreshold: number;
  userIntent: string;
  businessImpact: string;
  riskFactors: string[];
}

export class ProgressionDecisionEngine {
  async shouldProgressToPhase2(
    phase1Result: Phase1Result,
    request: AnalysisRequest
  ): Promise<ProgressionDecision> {
    const criteria = this.evaluateCriteria(phase1Result, request);
    const riskAssessment = this.assessRisk(criteria);
    const benefitAnalysis = this.analyzeBenefits(criteria);

    return {
      shouldProgress: this.makeDecision(
        criteria,
        riskAssessment,
        benefitAnalysis
      ),
      reasoning: this.generateReasoning(
        criteria,
        riskAssessment,
        benefitAnalysis
      ),
      estimatedTime: this.estimatePhase2Time(criteria),
      resourceRequirements: this.calculateResourceRequirements(criteria),
      alternatives: this.suggestAlternatives(criteria),
    };
  }

  private evaluateCriteria(
    phase1Result: Phase1Result,
    request: AnalysisRequest
  ): ProgressionCriteria {
    return {
      dataComplexity: this.assessDataComplexity(phase1Result),
      confidenceThreshold: this.calculateRequiredConfidence(request),
      userIntent: this.inferUserIntent(request),
      businessImpact: this.assessBusinessImpact(phase1Result, request),
      riskFactors: this.identifyRiskFactors(phase1Result),
    };
  }

  private makeDecision(
    criteria: ProgressionCriteria,
    risk: RiskAssessment,
    benefit: BenefitAnalysis
  ): boolean {
    // 决策算法
    const complexityScore = criteria.dataComplexity;
    const confidenceScore = phase1Result.confidence;
    const impactScore = this.calculateImpactScore(criteria.businessImpact);
    const riskScore = risk.overallRisk;
    const benefitScore = benefit.expectedBenefit;

    // 决策公式
    const decisionScore =
      complexityScore * 0.3 +
      (100 - confidenceScore) * 0.2 +
      impactScore * 0.25 +
      (100 - riskScore) * 0.15 +
      benefitScore * 0.1;

    return decisionScore > 60; // 阈值可配置
  }
}
```

```typescript
// src/analysis/Phase1Processor.ts
export class Phase1Processor {
  private readonly MAX_PROCESSING_TIME = 30000; // 30秒

  async preprocess(data: any): Promise<PreprocessedData> {
    const startTime = Date.now();

    // 数据清洗和标准化
    const cleanedData = await this.cleanData(data);

    // 基础特征提取
    const basicFeatures = await this.extractBasicFeatures(cleanedData);

    // 数据质量评估
    const qualityMetrics = await this.assessDataQuality(cleanedData);

    return {
      originalData: data,
      cleanedData,
      basicFeatures,
      qualityMetrics,
      preprocessingTime: Date.now() - startTime,
    };
  }

  async analyzeBasic(
    preprocessedData: PreprocessedData
  ): Promise<BasicInsights> {
    // 快速统计分析
    const statistics = this.calculateStatistics(preprocessedData.basicFeatures);

    // 基础趋势识别
    const trends = this.identifyBasicTrends(preprocessedData.basicFeatures);

    // 异常检测
    const anomalies = this.detectAnomalies(preprocessedData.basicFeatures);

    // 关键指标提取
    const keyMetrics = this.extractKeyMetrics(statistics, trends);

    return {
      statistics,
      trends,
      anomalies,
      keyMetrics,
      summary: this.generateSummary(statistics, trends, anomalies),
    };
  }

  async identifyPatterns(
    preprocessedData: PreprocessedData
  ): Promise<Pattern[]> {
    const patterns: Pattern[] = [];

    // 基础模式识别
    patterns.push(...(await this.identifyCyclicalPatterns(preprocessedData)));
    patterns.push(...(await this.identifySeasonalPatterns(preprocessedData)));
    patterns.push(
      ...(await this.identifyCorrelationPatterns(preprocessedData))
    );
    patterns.push(...(await this.identifyOutlierPatterns(preprocessedData)));

    // 按置信度排序
    return patterns.sort((a, b) => b.confidence - a.confidence).slice(0, 10);
  }

  async generateRecommendations(
    insights: BasicInsights,
    patterns: Pattern[],
    context: AnalysisContext
  ): Promise<Recommendation[]> {
    const recommendations: Recommendation[] = [];

    // 基于洞察的建议
    recommendations.push(...this.generateInsightBasedRecommendations(insights));

    // 基于模式的建议
    recommendations.push(...this.generatePatternBasedRecommendations(patterns));

    // 基于上下文的建议
    recommendations.push(...this.generateContextBasedRecommendations(context));

    // 排序和筛选
    return recommendations.sort((a, b) => b.priority - a.priority).slice(0, 5); // 返回前5个建议
  }
}

// src/analysis/Phase2Processor.ts
export class Phase2Processor {
  async deepAnalyze(
    data: any,
    phase1Insights: BasicInsights
  ): Promise<DeepInsights> {
    // 高级统计分析
    const advancedStatistics = await this.performAdvancedStatistics(data);

    // 机器学习模型分析
    const mlInsights = await this.applyMachineLearning(data);

    // 因果关系分析
    const causalAnalysis = await this.performCausalAnalysis(
      data,
      phase1Insights
    );

    // 预测建模
    const predictions = await this.generatePredictions(data, phase1Insights);

    return {
      advancedStatistics,
      mlInsights,
      causalAnalysis,
      predictions,
      synthesis: this.synthesizeInsights(
        advancedStatistics,
        mlInsights,
        causalAnalysis,
        predictions
      ),
    };
  }

  async crossValidate(
    insights: DeepInsights,
    context: AnalysisContext
  ): Promise<ValidationResults> {
    const validationMethods = [
      this.crossValidateStatistical(insights),
      this.crossValidateML(insights),
      this.crossValidateCausal(insights),
      this.validateAgainstDomainKnowledge(insights, context),
    ];

    const results = await Promise.allSettled(validationMethods);

    return {
      overallReliability: this.calculateOverallReliability(results),
      validationScores: this.extractValidationScores(results),
      inconsistencies: this.identifyInconsistencies(results),
      confidenceIntervals: this.calculateConfidenceIntervals(insights, results),
    };
  }

  async multidimensionalAnalyze(
    data: any,
    focusAreas: string[]
  ): Promise<MultidimensionalAnalysis> {
    const dimensions = await this.identifyDimensions(data);
    const analyses: DimensionAnalysis[] = [];

    for (const dimension of dimensions) {
      if (focusAreas.length === 0 || focusAreas.includes(dimension.name)) {
        const analysis = await this.analyzeDimension(data, dimension);
        analyses.push(analysis);
      }
    }

    // 维度间关系分析
    const relationships = await this.analyzeDimensionRelationships(analyses);

    return {
      dimensions: analyses,
      relationships,
      synergies: this.identifySynergies(analyses, relationships),
      conflicts: this.identifyConflicts(analyses, relationships),
    };
  }

  async generateExpertRecommendations(
    deepInsights: DeepInsights,
    validation: ValidationResults,
    multidimensional: MultidimensionalAnalysis,
    context: AnalysisContext
  ): Promise<ExpertRecommendation[]> {
    const recommendations: ExpertRecommendation[] = [];

    // 基于深度洞察的建议
    recommendations.push(
      ...this.generateDeepInsightRecommendations(deepInsights)
    );

    // 基于验证结果的修正建议
    recommendations.push(
      ...this.generateValidationBasedRecommendations(validation)
    );

    // 基于多维度的综合建议
    recommendations.push(
      ...this.generateMultidimensionalRecommendations(multidimensional)
    );

    // 基于领域知识的专业建议
    recommendations.push(...this.generateDomainExpertRecommendations(context));

    // 排序和分类
    return this.categorizeAndPrioritize(recommendations);
  }
}
```

```tsx
// src/components/TwoPhaseAnalysis/index.tsx
export const TwoPhaseAnalysis: React.FC<TwoPhaseAnalysisProps> = ({
  initialData,
  onComplete,
  onProgress,
}) => {
  const [currentPhase, setCurrentPhase] = useState<
    'idle' | 'phase1' | 'decision' | 'phase2' | 'completed'
  >('idle');
  const [phase1Result, setPhase1Result] = useState<Phase1Result | null>(null);
  const [phase2Result, setPhase2Result] = useState<Phase2Result | null>(null);
  const [progress, setProgress] = useState<AnalysisProgress>();
  const [isProcessing, setIsProcessing] = useState(false);

  const analysisEngine = useMemo(() => new TwoPhaseAnalysisEngine(), []);

  const startAnalysis = async (options: AnalysisOptions) => {
    setIsProcessing(true);
    setCurrentPhase('phase1');

    try {
      // 执行Phase 1分析
      const request: AnalysisRequest = {
        data: initialData,
        context: await buildAnalysisContext(),
        options,
      };

      const result = await analysisEngine.executeAnalysis(request);

      if (result.phase === 'phase1') {
        setPhase1Result(result.result);
        setCurrentPhase('decision');
      } else {
        setPhase1Result(result.phase1Result);
        setPhase2Result(result.result);
        setCurrentPhase('completed');
        onComplete?.(result);
      }
    } catch (error) {
      console.error('Analysis failed:', error);
      // 错误处理
    } finally {
      setIsProcessing(false);
    }
  };

  const proceedToPhase2 = async () => {
    if (!phase1Result) return;

    setIsProcessing(true);
    setCurrentPhase('phase2');

    try {
      const request: AnalysisRequest = {
        data: initialData,
        context: await buildAnalysisContext(),
        options: { enableDeepAnalysis: true, autoProgression: false },
      };

      const result = await analysisEngine.executeAnalysis(request);
      setPhase2Result(result.result);
      setCurrentPhase('completed');
      onComplete?.(result);
    } catch (error) {
      console.error('Phase 2 analysis failed:', error);
    } finally {
      setIsProcessing(false);
    }
  };

  return (
    <div className="two-phase-analysis">
      <PhaseIndicator currentPhase={currentPhase} />

      {currentPhase === 'idle' && (
        <AnalysisSetup onStart={startAnalysis} initialData={initialData} />
      )}

      {currentPhase === 'phase1' && (
        <Phase1View progress={progress} isProcessing={isProcessing} />
      )}

      {currentPhase === 'decision' && phase1Result && (
        <DecisionView
          phase1Result={phase1Result}
          onProceed={proceedToPhase2}
          onSkip={() => {
            setCurrentPhase('completed');
            onComplete?.({ phase: 'phase1', result: phase1Result });
          }}
        />
      )}

      {currentPhase === 'phase2' && (
        <Phase2View progress={progress} isProcessing={isProcessing} />
      )}

      {currentPhase === 'completed' && (
        <ResultsView phase1Result={phase1Result} phase2Result={phase2Result} />
      )}
    </div>
  );
};
```

```typescript
// src/stores/analysisStore.ts
export const useAnalysisStore = defineStore('analysis', {
  state: () => ({
    currentPhase: 'idle' as AnalysisPhase,
    phase1Result: null as Phase1Result | null,
    phase2Result: null as Phase2Result | null,
    progress: null as AnalysisProgress | null,
    isProcessing: false,
    error: null as AnalysisError | null,
    history: [] as AnalysisSession[],
    settings: {
      autoProgression: true,
      confidenceThreshold: 70,
      maxPhase2Time: 300000, // 5 minutes
      enableNotifications: true,
    },
  }),

  getters: {
    canProceedToPhase2: state => {
      return state.currentPhase === 'decision' && state.phase1Result !== null;
    },
    overallProgress: state => {
      if (state.currentPhase === 'idle') return 0;
      if (state.currentPhase === 'phase1') return state.progress?.progress || 0;
      if (state.currentPhase === 'decision') return 25;
      if (state.currentPhase === 'phase2')
        return 25 + (state.progress?.progress || 0) * 0.75;
      if (state.currentPhase === 'completed') return 100;
      return 0;
    },
    currentResult: state => {
      return state.phase2Result || state.phase1Result;
    },
  },

  actions: {
    async startPhase1(data: any, options: AnalysisOptions) {
      this.currentPhase = 'phase1';
      this.isProcessing = true;
      this.error = null;

      try {
        const engine = new TwoPhaseAnalysisEngine();
        const request = buildAnalysisRequest(data, options);
        const result = await engine.executeAnalysis(request);

        if (result.phase === 'phase1') {
          this.phase1Result = result.result;
          this.currentPhase = 'decision';
        } else {
          this.phase1Result = result.phase1Result;
          this.phase2Result = result.result;
          this.currentPhase = 'completed';
        }

        this.saveToHistory(result);
      } catch (error) {
        this.error = error as AnalysisError;
        throw error;
      } finally {
        this.isProcessing = false;
      }
    },

    async startPhase2() {
      if (!this.phase1Result) {
        throw new Error('Phase 1 result is required');
      }

      this.currentPhase = 'phase2';
      this.isProcessing = true;
      this.error = null;

      try {
        const engine = new TwoPhaseAnalysisEngine();
        const request = buildPhase2Request(this.phase1Result);
        const result = await engine.executeAnalysis(request);

        this.phase2Result = result.result;
        this.currentPhase = 'completed';
        this.saveToHistory(result);
      } catch (error) {
        this.error = error as AnalysisError;
        throw error;
      } finally {
        this.isProcessing = false;
      }
    },

    updateProgress(progress: AnalysisProgress) {
      this.progress = progress;
    },

    reset() {
      this.currentPhase = 'idle';
      this.phase1Result = null;
      this.phase2Result = null;
      this.progress = null;
      this.error = null;
    },

    saveToHistory(result: AnalysisResult) {
      const session: AnalysisSession = {
        id: generateId(),
        timestamp: new Date(),
        result,
        settings: { ...this.settings },
      };

      this.history.unshift(session);
      // 限制历史记录数量
      if (this.history.length > 50) {
        this.history = this.history.slice(0, 50);
      }
    },
  },
});
```

```typescript
// src/optimization/AsyncProcessor.ts
export class AsyncProcessor {
  private workerPool: Worker[];
  private taskQueue: AnalysisTask[];

  constructor(poolSize: number = 4) {
    this.workerPool = this.createWorkerPool(poolSize);
    this.taskQueue = [];
  }

  async processAsync(task: AnalysisTask): Promise<any> {
    return new Promise((resolve, reject) => {
      this.taskQueue.push({
        ...task,
        resolve,
        reject,
      });

      this.processQueue();
    });
  }

  private async processQueue(): Promise<void> {
    if (this.taskQueue.length === 0) return;

    const availableWorker = this.getAvailableWorker();
    if (!availableWorker) return;

    const task = this.taskQueue.shift()!;
    this.assignTaskToWorker(availableWorker, task);
  }

  private createWorkerPool(size: number): Worker[] {
    return Array.from({ length: size }, () => {
      const worker = new Worker('/workers/analysis-worker.js');
      worker.onmessage = this.handleWorkerMessage.bind(this);
      return worker;
    });
  }
}
```

```typescript
// src/optimization/MemoryManager.ts
export class MemoryManager {
  private cache = new Map<string, any>();
  private maxCacheSize = 100 * 1024 * 1024; // 100MB
  private currentCacheSize = 0;

  set(key: string, value: any): void {
    const size = this.calculateSize(value);

    if (size > this.maxCacheSize) {
      console.warn('Object too large for cache');
      return;
    }

    // 清理空间如果需要
    while (this.currentCacheSize + size > this.maxCacheSize) {
      this.evictLeastRecentlyUsed();
    }

    this.cache.set(key, {
      value,
      size,
      lastAccessed: Date.now(),
    });

    this.currentCacheSize += size;
  }

  get(key: string): any {
    const item = this.cache.get(key);
    if (item) {
      item.lastAccessed = Date.now();
      return item.value;
    }
    return null;
  }

  private evictLeastRecentlyUsed(): void {
    let oldestKey: string | null = null;
    let oldestTime = Date.now();

    for (const [key, item] of this.cache.entries()) {
      if (item.lastAccessed < oldestTime) {
        oldestTime = item.lastAccessed;
        oldestKey = key;
      }
    }

    if (oldestKey) {
      const item = this.cache.get(oldestKey)!;
      this.currentCacheSize -= item.size;
      this.cache.delete(oldestKey);
    }
  }
}
```
