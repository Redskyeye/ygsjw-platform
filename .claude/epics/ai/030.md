# 030 - 工单系统实现

## 概述
实现完整的工单系统，为用户提供便捷的工单提交、跟踪和处理功能，支持多种工单类型、优先级管理、自动分配和状态跟踪。

## 业务需求

### 核心功能
- **工单提交**：用户可以创建不同类型的工单
- **工单跟踪**：实时查看工单处理状态和进度
- **优先级管理**：支持紧急程度标记和优先级排序
- **自动分配**：智能分配工单给合适的客服人员
- **沟通记录**：完整的工单沟通历史记录
- **满意度评价**：工单完成后用户满意度反馈

### 工单类型定义
1. **技术支持工单**：技术问题、bug报告、功能故障
2. **功能请求工单**：新功能建议、改进意见
3. **账户问题工单**：登录问题、账户设置、权限问题
4. **计费咨询工单**：费用问题、发票、订阅管理
5. **投诉建议工单**：服务投诉、改进建议

```typescript
interface TicketListProps {
  tickets: Ticket[];
  selectedTicket: Ticket | null;
  onSelectTicket: (ticket: Ticket) => void;
  userRole: 'customer' | 'agent' | 'admin';
}

export const TicketList: React.FC<TicketListProps> = ({
  tickets,
  selectedTicket,
  onSelectTicket,
  userRole
}) => {
  const [filter, setFilter] = useState({
    status: 'all',
    type: 'all',
    priority: 'all'
  });

  const [searchTerm, setSearchTerm] = useState('');

  const filteredTickets = tickets.filter(ticket => {
    // 状态筛选
    if (filter.status !== 'all' && ticket.status !== filter.status) {
      return false;
    }

    // 类型筛选
    if (filter.type !== 'all' && ticket.type !== filter.type) {
      return false;
    }

    // 优先级筛选
    if (filter.priority !== 'all' && ticket.priority !== filter.priority) {
      return false;
    }

    // 搜索筛选
    if (searchTerm && !ticket.title.toLowerCase().includes(searchTerm.toLowerCase())) {
      return false;
    }

    return true;
  });

  const sortedTickets = [...filteredTickets].sort((a, b) => {
    // 按优先级和更新时间排序
    const priorityOrder = { urgent: 4, high: 3, medium: 2, low: 1 };
    const priorityDiff = priorityOrder[b.priority] - priorityOrder[a.priority];

    if (priorityDiff !== 0) {
      return priorityDiff;
    }

    return new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime();
  });

  return (
    <div className="ticket-list">
      <div className="ticket-filters">
        <div className="search-box">
          <input
            type="text"
            placeholder="搜索工单..."
            value={searchTerm}
            onChange={(e) => setSearchTerm(e.target.value)}
          />
        </div>

        <div className="filter-controls">
          <StatusFilter
            value={filter.status}
            onChange={(status) => setFilter(prev => ({ ...prev, status }))}
          />

          <TypeFilter
            value={filter.type}
            onChange={(type) => setFilter(prev => ({ ...prev, type }))}
          />

          <PriorityFilter
            value={filter.priority}
            onChange={(priority) => setFilter(prev => ({ ...prev, priority }))}
          />
        </div>
      </div>

      <div className="ticket-items">
        {sortedTickets.map(ticket => (
          <TicketItem
            key={ticket.id}
            ticket={ticket}
            isSelected={selectedTicket?.id === ticket.id}
            onSelect={() => onSelectTicket(ticket)}
            userRole={userRole}
          />
        ))}

        {sortedTickets.length === 0 && (
          <div className="empty-state">
            <p>暂无工单</p>
          </div>
        )}
      </div>
    </div>
  );
};
```

```typescript
// src/services/TicketService.ts
export class TicketService {
  constructor(
    private ticketModel: Model<TicketDocument>,
    private userService: UserService,
    private notificationService: NotificationService,
    private assignmentEngine: TicketAssignmentEngine
  ) {}

  async createTicket(ticketData: CreateTicketDto): Promise<TicketDocument> {
    // 生成工单编号
    const ticketId = this.generateTicketId();

    // 创建工单文档
    const ticket = new this.ticketModel({
      ...ticketData,
      ticketId,
      status: TicketStatus.OPEN,
      createdAt: new Date(),
      updatedAt: new Date(),
      metadata: {
        source: ticketData.source || 'web',
        userAgent: ticketData.userAgent,
        ip: ticketData.ip
      }
    });

    // 保存工单
    await ticket.save();

    // 自动分配工单
    if (this.shouldAutoAssign(ticketData.type, ticketData.priority)) {
      await this.autoAssignTicket(ticket);
    }

    // 发送通知
    await this.sendTicketCreatedNotifications(ticket);

    return ticket;
  }

  async updateTicket(
    ticketId: string,
    updates: UpdateTicketDto,
    updatedBy: string
  ): Promise<TicketDocument> {
    const ticket = await this.ticketModel.findOne({ ticketId });
    if (!ticket) {
      throw new Error('工单不存在');
    }

    // 更新字段
    const updatedTicket = await this.ticketModel.findOneAndUpdate(
      { ticketId },
      {
        ...updates,
        updatedAt: new Date(),
        // 记录状态变更时间
        ...(updates.status === TicketStatus.RESOLVED && {
          'metadata.resolvedAt': new Date()
        }),
        // 记录首次响应时间
        ...(updates.status === TicketStatus.IN_PROGRESS && !ticket.metadata.firstResponseAt && {
          'metadata.firstResponseAt': new Date()
        })
      },
      { new: true }
    );

    // 发送状态更新通知
    await this.sendStatusUpdateNotifications(updatedTicket, updates);

    return updatedTicket;
  }

  async addMessage(
    ticketId: string,
    messageData: CreateMessageDto
  ): Promise<TicketDocument> {
    const ticket = await this.ticketModel.findOne({ ticketId });
    if (!ticket) {
      throw new Error('工单不存在');
    }

    const message: TicketMessage = {
      id: generateId(),
      userId: messageData.userId,
      content: messageData.content,
      type: messageData.type || 'text',
      timestamp: new Date(),
      isInternal: messageData.isInternal || false,
      attachments: messageData.attachments || []
    };

    // 添加消息到工单
    const updatedTicket = await this.ticketModel.findOneAndUpdate(
      { ticketId },
      {
        $push: { messages: message },
        updatedAt: new Date()
      },
      { new: true }
    );

    // 发送新消息通知
    await this.sendNewMessageNotifications(updatedTicket, message);

    return updatedTicket;
  }

  async getTickets(query: GetTicketsDto): Promise<PaginatedResult<TicketDocument>> {
    const {
      page = 1,
      limit = 20,
      status,
      type,
      priority,
      userId,
      assignedTo,
      searchTerm,
      sortBy = 'updatedAt',
      sortOrder = 'desc'
    } = query;

    // 构建查询条件
    const filter: any = {};

    if (status) filter.status = status;
    if (type) filter.type = type;
    if (priority) filter.priority = priority;
    if (userId) filter.userId = userId;
    if (assignedTo) filter.assignedTo = assignedTo;

    if (searchTerm) {
      filter.$or = [
        { title: { $regex: searchTerm, $options: 'i' } },
        { description: { $regex: searchTerm, $options: 'i' } }
      ];
    }

    // 构建排序
    const sort: any = {};
    sort[sortBy] = sortOrder === 'desc' ? -1 : 1;

    // 执行查询
    const skip = (page - 1) * limit;
    const [tickets, total] = await Promise.all([
      this.ticketModel
        .find(filter)
        .sort(sort)
        .skip(skip)
        .limit(limit)
        .populate('userId', 'name email avatar')
        .populate('assignedTo', 'name email avatar'),
      this.ticketModel.countDocuments(filter)
    ]);

    return {
      data: tickets,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit)
      }
    };
  }

  private generateTicketId(): string {
    const date = new Date();
    const dateStr = date.toISOString().slice(0, 10).replace(/-/g, '');
    const random = Math.random().toString(36).substr(2, 6).toUpperCase();
    return `TK${dateStr}${random}`;
  }

  private async autoAssignTicket(ticket: TicketDocument): Promise<void> {
    const assignee = await this.assignmentEngine.assignTicket(ticket);
    if (assignee) {
      await this.ticketModel.updateOne(
        { _id: ticket._id },
        { assignedTo: assignee.id, updatedAt: new Date() }
      );
    }
  }

  private shouldAutoAssign(type: TicketType, priority: TicketPriority): boolean {
    // 高优先级和技术支持工单自动分配
    return priority === TicketPriority.URGENT ||
           priority === TicketPriority.HIGH ||
           type === TicketType.TECHNICAL;
  }
}
```

### 智能分配引擎
```typescript
// src/services/TicketAssignmentEngine.ts
export class TicketAssignmentEngine {
  constructor(
    private agentService: AgentService,
    private mlService: MLService
  ) {}

  async assignTicket(ticket: TicketDocument): Promise<Agent | null> {
    // 获取可用客服人员
    const availableAgents = await this.agentService.getAvailableAgents({
      requiredSkills: this.getRequiredSkills(ticket.type),
      maxWorkload: 5
    });

    if (availableAgents.length === 0) {
      return null;
    }

    // 使用ML模型预测最佳分配
    const assignments = await this.predictBestAssignment(ticket, availableAgents);

    // 选择最佳匹配
    const bestAssignment = assignments[0];
    return bestAssignment.agent;
  }

  private getRequiredSkills(ticketType: TicketType): string[] {
    const skillMap = {
      [TicketType.TECHNICAL]: ['technical_support', 'troubleshooting', 'debugging'],
      [TicketType.FEATURE_REQUEST]: ['product_knowledge', 'communication', 'analysis'],
      [TicketType.ACCOUNT]: ['account_management', 'security', 'user_support'],
      [TicketType.BILLING]: ['billing_support', 'financial_knowledge', 'negotiation'],
      [TicketType.COMPLAINT]: ['conflict_resolution', 'communication', 'empathy']
    };

    return skillMap[ticketType] || [];
  }

  private async predictBestAssignment(
    ticket: TicketDocument,
    agents: Agent[]
  ): Promise<{ agent: Agent; score: number }[]> {
    const assignments = await Promise.all(
      agents.map(async agent => {
        const score = await this.calculateAssignmentScore(ticket, agent);
        return { agent, score };
      })
    );

    return assignments.sort((a, b) => b.score - a.score);
  }

  private async calculateAssignmentScore(ticket: TicketDocument, agent: Agent): Promise<number> {
    let score = 0;

    // 技能匹配度 (40%)
    const skillScore = this.calculateSkillMatch(ticket.type, agent.skills);
    score += skillScore * 0.4;

    // 工作负载 (25%)
    const workloadScore = this.calculateWorkloadScore(agent.currentWorkload);
    score += workloadScore * 0.25;

    // 历史表现 (20%)
    const performanceScore = await this.getPerformanceScore(agent.id, ticket.type);
    score += performanceScore * 0.2;

    // 可用性 (15%)
    const availabilityScore = this.calculateAvailabilityScore(agent);
    score += availabilityScore * 0.15;

    return score;
  }

  private calculateSkillMatch(ticketType: TicketType, agentSkills: string[]): number {
    const requiredSkills = this.getRequiredSkills(ticketType);
    const matchedSkills = requiredSkills.filter(skill => agentSkills.includes(skill));

    return matchedSkills.length / requiredSkills.length;
  }

  private calculateWorkloadScore(workload: number): number {
    // 工作负载越低分数越高
    return Math.max(0, 1 - (workload / 10));
  }

  private async getPerformanceScore(agentId: string, ticketType: TicketType): Promise<number> {
    // 获取客服人员在该类型工单的历史表现
    const stats = await this.agentService.getPerformanceStats(agentId, ticketType);

    // 基于解决率、满意度、响应时间等计算分数
    const resolutionRate = stats.resolutionRate || 0;
    const satisfactionRate = stats.averageSatisfaction || 0;
    const responseTimeScore = Math.max(0, 1 - (stats.averageResponseTime || 0) / 3600); // 1小时为满分

    return (resolutionRate * 0.4 + satisfactionRate * 0.4 + responseTimeScore * 0.2);
  }

  private calculateAvailabilityScore(agent: Agent): number {
    if (!agent.isOnline) return 0;

    const now = new Date();
    const workHours = agent.workHours;
    const currentTime = now.getHours() * 60 + now.getMinutes();

    if (currentTime >= workHours.start && currentTime <= workHours.end) {
      return 1;
    } else {
      return 0.5; // 非工作时间部分分数
    }
  }
}
```

```typescript
// src/services/NotificationService.ts
export class NotificationService {
  constructor(
    private emailService: EmailService,
    private smsService: SMSService,
    private pushService: PushNotificationService,
    private websocketService: WebSocketService
  ) {}

  async sendTicketCreatedNotifications(ticket: TicketDocument): Promise<void> {
    // 通知用户
    await this.notifyUserTicketCreated(ticket);

    // 通知相关客服人员
    await this.notifyAgentsTicketCreated(ticket);
  }

  async sendStatusUpdateNotifications(ticket: TicketDocument, updates: any): Promise<void> {
    // 通知用户状态变更
    await this.notifyUserStatusUpdate(ticket, updates);

    // 如果分配了客服，通知客服
    if (updates.assignedTo) {
      await this.notifyAgentAssignment(ticket, updates.assignedTo);
    }
  }

  async sendNewMessageNotifications(ticket: TicketDocument, message: TicketMessage): Promise<void> {
    // 确定通知对象
    const recipients = this.getMessageRecipients(ticket, message);

    // 发送实时通知
    await this.sendRealtimeNotification(recipients, {
      type: 'new_message',
      ticketId: ticket.ticketId,
      message: message,
      timestamp: new Date()
    });

    // 发送邮件通知（如果不是内部消息）
    if (!message.isInternal) {
      await this.sendEmailNotification(recipients, {
        subject: `工单 ${ticket.ticketId} 有新消息`,
        template: 'ticket-new-message',
        data: { ticket, message }
      });
    }
  }

  private async notifyUserTicketCreated(ticket: TicketDocument): Promise<void> {
    const user = await getUserById(ticket.userId);

    await this.emailService.sendEmail({
      to: user.email,
      subject: `工单 ${ticket.ticketId} 已创建`,
      template: 'ticket-created',
      data: { ticket, user }
    });

    await this.pushService.sendPushNotification(user.id, {
      title: '工单已创建',
      body: `您的工单 ${ticket.ticketId} 已成功提交，我们会尽快处理`,
      data: { ticketId: ticket.ticketId }
    });
  }

  private async notifyAgentsTicketCreated(ticket: TicketDocument): Promise<void> {
    const agents = await this.getRelevantAgents(ticket);

    for (const agent of agents) {
      await this.websocketService.sendToUser(agent.id, {
        type: 'ticket_created',
        data: ticket
      });

      if (agent.preferences.emailNotifications) {
        await this.emailService.sendEmail({
          to: agent.email,
          subject: `新工单 ${ticket.ticketId} 分配通知`,
          template: 'agent-new-ticket',
          data: { ticket, agent }
        });
      }
    }
  }

  private getMessageRecipients(ticket: TicketDocument, message: TicketMessage): User[] {
    const recipients = [];

    // 如果是客服消息，通知用户
    if (message.isInternal) {
      recipients.push(getUserById(ticket.userId));
    } else {
      // 如果是用户消息，通知分配的客服和管理员
      if (ticket.assignedTo) {
        recipients.push(getUserById(ticket.assignedTo));
      }

      // 通知其他在线客服
      const onlineAgents = this.getOnlineAgents();
      recipients.push(...onlineAgents);
    }

    return recipients.filter(Boolean);
  }
}
```

```typescript
  });
});
```
