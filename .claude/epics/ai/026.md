# 026 - 报告生成集成系统

## 概述
实现智能报告生成集成系统，通过与N8N工作流引擎的深度集成，提供自动化、定制化的报告生成和分发功能。

## 业务需求

### 核心功能
- **模板管理**：支持多种报告模板的创建和管理
- **数据集成**：自动从多个数据源收集信息
- **报告生成**：智能生成各类分析报告
- **工作流集成**：与N8N无缝集成实现自动化
- **分发管理**：支持多渠道报告分发

### 报告类型
1. **分析报告**：4D分析结果报告
2. **诊断报告**：问题诊断详细报告
3. **进度报告**：项目进展状态报告
4. **性能报告**：系统性能指标报告
5. **使用报告**：用户行为统计报告
6. **汇总报告**：多维度数据汇总报告
7. **自定义报告**：用户定制化报告

```typescript
// src/reports/ReportEngine.ts
export interface ReportTemplate {
  id: string;
  name: string;
  description: string;
  type: ReportType;
  category: ReportCategory;
  layout: TemplateLayout;
  sections: TemplateSection[];
  styling: ReportStyling;
  dataSource: DataSourceConfig;
  outputFormat: OutputFormat[];
  schedule?: ReportSchedule;
  recipients: RecipientConfig;
}

export interface ReportData {
  id: string;
  templateId: string;
  dataSource: DataSourceMap;
  metadata: ReportMetadata;
  generatedAt: Date;
  generatedBy: string;
  parameters: ReportParameters;
  filters: DataFilter[];
}

export interface GeneratedReport {
  id: string;
  templateId: string;
  data: ReportData;
  content: ReportContent;
  format: OutputFormat;
  files: ReportFile[];
  metadata: ReportMetadata;
  status: ReportStatus;
  createdAt: Date;
  expiresAt?: Date;
}

export class ReportEngine {
  private templateEngine: TemplateEngine;
  private dataCollector: DataCollector;
  private contentGenerator: ContentGenerator;
  private n8nIntegration: N8NIntegration;
  private distributionManager: DistributionManager;

  async generateReport(
    templateId: string,
    parameters: ReportParameters,
    options: ReportOptions = {}
  ): Promise<GeneratedReport> {
    // 1. 加载模板
    const template = await this.loadTemplate(templateId);
    if (!template) {
      throw new Error(`Template ${templateId} not found`);
    }

    // 2. 收集数据
    const data = await this.collectReportData(template, parameters, options);

    // 3. 生成内容
    const content = await this.generateReportContent(template, data);

    // 4. 应用样式
    const styledContent = await this.applyStyling(content, template.styling);

    // 5. 格式化输出
    const formattedReports = await this.formatOutputs(
      styledContent,
      template.outputFormat,
      options
    );

    // 6. 生成报告对象
    const report: GeneratedReport = {
      id: generateId(),
      templateId,
      data,
      content: styledContent,
      format: options.format || template.outputFormat[0],
      files: formattedReports,
      metadata: {
        generatedAt: new Date(),
        generatedBy: options.userId || 'system',
        templateName: template.name,
        parameters,
        processingTime: 0
      },
      status: 'completed',
      createdAt: new Date(),
      expiresAt: this.calculateExpiryDate(template)
    };

    // 7. 触发后续工作流
    await this.triggerPostGenerationWorkflow(report, template);

    return report;
  }

  private async loadTemplate(templateId: string): Promise<ReportTemplate | null> {
    return await this.templateEngine.getTemplate(templateId);
  }

  private async collectReportData(
    template: ReportTemplate,
    parameters: ReportParameters,
    options: ReportOptions
  ): Promise<ReportData> {
    const dataSourceMap = new Map<string, any>();

    for (const dataSource of template.dataSource.sources) {
      try {
        const data = await this.dataCollector.collectData(
          dataSource,
          parameters,
          options
        );
        dataSourceMap.set(dataSource.id, data);
      } catch (error) {
        console.error(`Failed to collect data from source ${dataSource.id}:`, error);
        dataSourceMap.set(dataSource.id, this.generateErrorData(error));
      }
    }

    return {
      id: generateId(),
      templateId: template.id,
      dataSource: Object.fromEntries(dataSourceMap),
      metadata: {
        collectedAt: new Date(),
        sourceCount: template.dataSource.sources.length,
        successCount: Array.from(dataSourceMap.values()).filter(
          data => !data.error
        ).length,
        totalRecords: this.calculateTotalRecords(dataSourceMap)
      },
      generatedAt: new Date(),
      generatedBy: options.userId || 'system',
      parameters,
      filters: this.buildDataFilters(template, parameters)
    };
  }

  private async generateReportContent(
    template: ReportTemplate,
    data: ReportData
  ): Promise<ReportContent> {
    const sections: ReportSection[] = [];

    for (const templateSection of template.sections) {
      const section = await this.generateSectionContent(
        templateSection,
        data,
        template
      );
      sections.push(section);
    }

    return {
      title: this.processTemplate(template.name, data),
      summary: await this.generateSummary(data, template),
      sections,
      footer: await this.generateFooter(data, template),
      metadata: {
        wordCount: this.calculateWordCount(sections),
        pageCount: this.estimatePageCount(sections),
        chartsCount: this.countCharts(sections)
      }
    };
  }

  private async generateSectionContent(
    templateSection: TemplateSection,
    data: ReportData,
    template: ReportTemplate
  ): Promise<ReportSection> {
    switch (templateSection.type) {
      case 'data-table':
        return await this.generateDataTableSection(templateSection, data);
      case 'chart':
        return await this.generateChartSection(templateSection, data);
      case 'text':
        return await this.generateTextSection(templateSection, data);
      case 'summary':
        return await this.generateSummarySection(templateSection, data);
      case 'recommendations':
        return await this.generateRecommendationsSection(templateSection, data);
      default:
        return await this.generateCustomSection(templateSection, data);
    }
  }
}
```

```typescript
// src/n8n/N8NIntegration.ts
export interface N8NWorkflow {
  id: string;
  name: string;
  description: string;
  triggers: N8NTrigger[];
  nodes: N8NNode[];
  connections: N8NConnection[];
  settings: N8NSettings;
  version: string;
  isActive: boolean;
}

export interface ReportWorkflowConfig {
  workflowId: string;
  triggerType: 'manual' | 'scheduled' | 'event-based';
  schedule?: ScheduleConfig;
  inputData: WorkflowInputData;
  outputMapping: OutputMapping;
  errorHandling: ErrorHandlingConfig;
  notifications: NotificationConfig;
}

export class N8NIntegration {
  private apiClient: N8NAPIClient;
  private workflowRegistry: WorkflowRegistry;
  private executionManager: ExecutionManager;

  constructor(config: N8NConfig) {
    this.apiClient = new N8NAPIClient(config);
    this.workflowRegistry = new WorkflowRegistry();
    this.executionManager = new ExecutionManager(this.apiClient);
  }

  async createReportWorkflow(
    template: ReportTemplate,
    config: ReportWorkflowConfig
  ): Promise<N8NWorkflow> {
    // 1. 构建工作流定义
    const workflowDefinition = this.buildWorkflowDefinition(template, config);

    // 2. 注册工作流
    const workflow = await this.apiClient.createWorkflow(workflowDefinition);

    // 3. 配置触发器
    await this.configureTriggers(workflow.id, config);

    // 4. 设置输入输出映射
    await this.setupInputOutputMapping(workflow.id, config);

    // 5. 配置错误处理
    await this.setupErrorHandling(workflow.id, config.errorHandling);

    // 6. 激活工作流
    if (config.triggerType !== 'manual') {
      await this.apiClient.activateWorkflow(workflow.id);
    }

    // 7. 注册到本地注册表
    this.workflowRegistry.register(template.id, workflow);

    return workflow;
  }

  private buildWorkflowDefinition(
    template: ReportTemplate,
    config: ReportWorkflowConfig
  ): N8NWorkflowDefinition {
    const nodes: N8NNode[] = [
      // 1. 触发器节点
      this.buildTriggerNode(config.triggerType, config.schedule),

      // 2. 数据收集节点
      ...this.buildDataCollectionNodes(template.dataSource),

      // 3. 报告生成节点
      this.buildReportGenerationNode(template),

      // 4. 格式化节点
      ...this.buildFormattingNodes(template.outputFormat),

      // 5. 分发节点
      ...this.buildDistributionNodes(template.recipients),

      // 6. 通知节点
      this.buildNotificationNode(config.notifications)
    ];

    // 构建节点连接
    const connections = this.buildNodeConnections(nodes);

    return {
      name: `Report: ${template.name}`,
      nodes,
      connections,
      settings: {
        timezone: 'Asia/Shanghai',
        retryOnError: true,
        maxRetries: 3
      },
      isActive: false
    };
  }

  private buildTriggerNode(
    triggerType: string,
    schedule?: ScheduleConfig
  ): N8NNode {
    switch (triggerType) {
      case 'scheduled':
        return {
          id: 'trigger',
          name: 'Schedule Trigger',
          type: 'n8n-nodes-base.cron',
          typeVersion: 1,
          position: [250, 300],
          parameters: {
            rule: {
              interval: schedule?.interval || '0 9 * * *', // 每天上午9点
              timezone: 'Asia/Shanghai'
            }
          }
        };

      case 'event-based':
        return {
          id: 'trigger',
          name: 'Webhook Trigger',
          type: 'n8n-nodes-base.webhook',
          typeVersion: 1,
          position: [250, 300],
          parameters: {
            httpMethod: 'POST',
            path: 'report-trigger',
            responseMode: 'responseNode',
            options: {}
          }
        };

      default: // manual
        return {
          id: 'trigger',
          name: 'Manual Trigger',
          type: 'n8n-nodes-base.manualTrigger',
          typeVersion: 1,
          position: [250, 300],
          parameters: {}
        };
    }
  }

  private buildDataCollectionNodes(dataSource: DataSourceConfig): N8NNode[] {
    const nodes: N8NNode[] = [];
    let positionY = 200;

    for (const source of dataSource.sources) {
      const node = this.buildDataCollectionNode(source, positionY);
      nodes.push(node);
      positionY += 150;
    }

    return nodes;
  }

  private buildDataCollectionNode(source: DataSource, positionY: number): N8NNode {
    switch (source.type) {
      case 'database':
        return {
          id: `data-${source.id}`,
          name: `DB: ${source.name}`,
          type: 'n8n-nodes-base.postgres',
          typeVersion: 2,
          position: [500, positionY],
          parameters: {
            connection: source.connectionId,
            query: source.query,
            options: {}
          }
        };

      case 'api':
        return {
          id: `data-${source.id}`,
          name: `API: ${source.name}`,
          type: 'n8n-nodes-base.httpRequest',
          typeVersion: 3,
          position: [500, positionY],
          parameters: {
            url: source.url,
            method: source.method || 'GET',
            headers: source.headers,
            options: {}
          }
        };

      case 'file':
        return {
          id: `data-${source.id}`,
          name: `File: ${source.name}`,
          type: 'n8n-nodes-base.readBinaryFile',
          typeVersion: 1,
          position: [500, positionY],
          parameters: {
            filePath: source.filePath,
            options: {}
          }
        };

      default:
        throw new Error(`Unsupported data source type: ${source.type}`);
    }
  }

  async executeWorkflow(
    workflowId: string,
    inputData: any = {}
  ): Promise<WorkflowExecution> {
    try {
      // 1. 准备执行数据
      const executionData = this.prepareExecutionData(inputData);

      // 2. 触发工作流执行
      const execution = await this.executionManager.execute(
        workflowId,
        executionData
      );

      // 3. 监控执行状态
      const result = await this.monitorExecution(execution.id);

      return result;
    } catch (error) {
      console.error(`Workflow execution failed: ${workflowId}`, error);
      throw error;
    }
  }

  async monitorExecution(executionId: string): Promise<WorkflowExecution> {
    const maxWaitTime = 600000; // 10分钟
    const checkInterval = 5000; // 5秒
    let elapsed = 0;

    while (elapsed < maxWaitTime) {
      const execution = await this.apiClient.getExecution(executionId);

      if (execution.finished) {
        return execution;
      }

      await new Promise(resolve => setTimeout(resolve, checkInterval));
      elapsed += checkInterval;
    }

    throw new Error(`Workflow execution timed out: ${executionId}`);
  }
}
```

```typescript
// src/reports/ContentGenerator.ts
export interface ContentGenerationContext {
  template: ReportTemplate;
  data: ReportData;
  parameters: ReportParameters;
  locale: string;
  format: OutputFormat;
}

export class ContentGenerator {
  private templateProcessor: TemplateProcessor;
  private chartGenerator: ChartGenerator;
  private tableGenerator: TableGenerator;
  private textGenerator: TextGenerator;
  private summaryGenerator: SummaryGenerator;

  async generateSectionContent(
    section: TemplateSection,
    context: ContentGenerationContext
  ): Promise<ReportSection> {
    const generator = this.getGenerator(section.type);
    return await generator.generate(section, context);
  }

  private getGenerator(type: SectionType): SectionGenerator {
    switch (type) {
      case 'data-table':
        return this.tableGenerator;
      case 'chart':
        return this.chartGenerator;
      case 'text':
        return this.textGenerator;
      case 'summary':
        return this.summaryGenerator;
      default:
        throw new Error(`Unsupported section type: ${type}`);
    }
  }

  async generateDataTable(
    section: TemplateSection,
    context: ContentGenerationContext
  ): Promise<TableSection> {
    const dataSource = context.data.dataSource[section.dataSource];
    const tableConfig = section.config as TableConfig;

    // 处理数据
    const processedData = await this.processTableData(dataSource, tableConfig);

    // 生成表格
    const table = this.tableGenerator.generateTable(processedData, tableConfig);

    return {
      type: 'data-table',
      title: section.title,
      content: table,
      metadata: {
        rowCount: processedData.length,
        columnCount: tableConfig.columns.length,
        hasPagination: tableConfig.pagination?.enabled || false
      }
    };
  }

  async generateChart(
    section: TemplateSection,
    context: ContentGenerationContext
  ): Promise<ChartSection> {
    const dataSource = context.data.dataSource[section.dataSource];
    const chartConfig = section.config as ChartConfig;

    // 准备图表数据
    const chartData = await this.prepareChartData(dataSource, chartConfig);

    // 生成图表
    const chart = await this.chartGenerator.generateChart(chartData, chartConfig);

    return {
      type: 'chart',
      title: section.title,
      content: chart,
      metadata: {
        chartType: chartConfig.type,
        dataPoints: chartData.length,
        hasLegend: chartConfig.legend?.enabled || false
      }
    };
  }

  async generateSummary(
    section: TemplateSection,
    context: ContentGenerationContext
  ): Promise<SummarySection> {
    const summaryData = await this.prepareSummaryData(context.data);
    const summaryConfig = section.config as SummaryConfig;

    const summary = await this.summaryGenerator.generateSummary(
      summaryData,
      summaryConfig,
      context
    );

    return {
      type: 'summary',
      title: section.title,
      content: summary,
      metadata: {
        keyPoints: summary.keyPoints?.length || 0,
        recommendations: summary.recommendations?.length || 0
      }
    };
  }

  private async processTableData(
    data: any,
    config: TableConfig
  ): Promise<TableData> {
    // 数据过滤
    let filteredData = this.applyFilters(data, config.filters);

    // 数据排序
    if (config.sort) {
      filteredData = this.applySorting(filteredData, config.sort);
    }

    // 数据分页
    const paginatedData = this.applyPagination(filteredData, config.pagination);

    // 数据转换
    const transformedData = this.applyDataTransform(paginatedData, config.transformations);

    return {
      headers: config.columns.map(col => ({
        key: col.key,
        label: col.label,
        type: col.type,
        width: col.width,
        align: col.align
      })),
      rows: transformedData,
      totalRows: filteredData.length
    };
  }

  private async prepareChartData(
    data: any,
    config: ChartConfig
  ): Promise<ChartData> {
    // 根据图表类型准备数据
    switch (config.type) {
      case 'bar':
      case 'line':
      case 'area':
        return this.prepareXYChartData(data, config);
      case 'pie':
      case 'donut':
        return this.preparePieChartData(data, config);
      case 'radar':
        return this.prepareRadarChartData(data, config);
      default:
        throw new Error(`Unsupported chart type: ${config.type}`);
    }
  }
}
```

```tsx
// src/components/ReportGeneration/ReportGenerator.tsx
export const ReportGenerator: React.FC<ReportGeneratorProps> = ({
  templateId,
  onGenerate,
  onCancel
}) => {
  const [template, setTemplate] = useState<ReportTemplate | null>(null);
  const [parameters, setParameters] = useState<ReportParameters>({});
  const [options, setOptions] = useState<ReportOptions>({});
  const [isGenerating, setIsGenerating] = useState(false);
  const [progress, setProgress] = useState<GenerationProgress>();

  useEffect(() => {
    if (templateId) {
      loadTemplate(templateId);
    }
  }, [templateId]);

  const loadTemplate = async (id: string) => {
    try {
      const templateData = await reportService.getTemplate(id);
      setTemplate(templateData);
      initializeParameters(templateData);
    } catch (error) {
      console.error('Failed to load template:', error);
    }
  };

  const initializeParameters = (template: ReportTemplate) => {
    const initialParams: ReportParameters = {};

    // 根据模板初始化参数
    template.dataSource.sources.forEach(source => {
      if (source.parameters) {
        initialParams[source.id] = source.parameters.default || {};
      }
    });

    setParameters(initialParams);
  };

  const handleGenerate = async () => {
    if (!template) return;

    setIsGenerating(true);
    setProgress({ stage: 'initialization', progress: 0 });

    try {
      const report = await reportService.generateReport(
        template.id,
        parameters,
        {
          ...options,
          onProgress: setProgress
        }
      );

      onGenerate?.(report);
    } catch (error) {
      console.error('Report generation failed:', error);
    } finally {
      setIsGenerating(false);
      setProgress(undefined);
    }
  };

  if (!template) {
    return <LoadingView message="加载模板中..." />;
  }

  return (
    <div className="report-generator">
      <div className="generator-header">
        <h2>生成报告</h2>
        <div className="template-info">
          <h3>{template.name}</h3>
          <p>{template.description}</p>
        </div>
      </div>

      <div className="generator-content">
        <div className="parameters-section">
          <h3>参数配置</h3>
          <ParametersForm
            template={template}
            parameters={parameters}
            onChange={setParameters}
          />
        </div>

        <div className="options-section">
          <h3>生成选项</h3>
          <OptionsForm
            template={template}
            options={options}
            onChange={setOptions}
          />
        </div>

        {isGenerating && progress && (
          <div className="progress-section">
            <h3>生成进度</h3>
            <GenerationProgress progress={progress} />
          </div>
        )}
      </div>

      <div className="generator-actions">
        <button
          className="btn btn-secondary"
          onClick={onCancel}
          disabled={isGenerating}
        >
          取消
        </button>
        <button
          className="btn btn-primary"
          onClick={handleGenerate}
          disabled={isGenerating || !isFormValid(parameters)}
        >
          {isGenerating ? '生成中...' : '生成报告'}
        </button>
      </div>
    </div>
  );
};
```

```tsx
  workflowId,

}) => {
  const [workflow, setWorkflow] = useState<N8NWorkflow | null>(null);
  const [nodes, setNodes] = useState<WorkflowNode[]>([]);
  const [connections, setConnections] = useState<WorkflowConnection[]>([]);
  const [selectedNode, setSelectedNode] = useState<WorkflowNode | null>(null);
  const [isSaving, setIsSaving] = useState(false);

  useEffect(() => {
    if (workflowId) {
      loadWorkflow(workflowId);
    } else {
      initializeNewWorkflow();
    }
  }, [workflowId]);

  const loadWorkflow = async (id: string) => {
    try {
      const workflowData = await n8nService.getWorkflow(id);
      setWorkflow(workflowData);
      setNodes(convertToNodes(workflowData.nodes));
      setConnections(convertToConnections(workflowData.connections));
    } catch (error) {
      console.error('Failed to load workflow:', error);
    }
  };

  const handleNodeAdd = (nodeType: NodeType) => {
    const newNode: WorkflowNode = {
      id: generateId(),
      type: nodeType,
      name: `${nodeType.name} ${nodes.length + 1}`,
      position: { x: 400 + Math.random() * 200, y: 100 + Math.random() * 200 },
      parameters: getDefaultParameters(nodeType),
      isActive: true
    };

    setNodes([...nodes, newNode]);
  };

  const handleNodeUpdate = (nodeId: string, updates: Partial<WorkflowNode>) => {
    setNodes(nodes.map(node =>
      node.id === nodeId ? { ...node, ...updates } : node
    ));
  };

  const handleConnectionAdd = (sourceId: string, targetId: string) => {
    const newConnection: WorkflowConnection = {
      id: generateId(),
      sourceNodeId: sourceId,
      targetNodeId: targetId,
      sourceIndex: 0,
      targetIndex: 0
    };

    setConnections([...connections, newConnection]);
  };

  const handleSave = async () => {
    setIsSaving(true);
    try {
      const workflowData = convertToWorkflow(nodes, connections);
      const savedWorkflow = await n8nService.saveWorkflow(workflowData);
      setWorkflow(savedWorkflow);
      onSave?.(savedWorkflow);
    } catch (error) {
      console.error('Failed to save workflow:', error);
    } finally {
      setIsSaving(false);
    }
  };

  return (
    <div className="workflow-editor">
      <div className="editor-header">
        <h2>工作流编辑器</h2>
        <div className="header-actions">
          <button className="btn btn-secondary" onClick={onCancel}>
            取消
          </button>
          <button
            className="btn btn-primary"
            onClick={handleSave}
            disabled={isSaving}
          >
            {isSaving ? '保存中...' : '保存工作流'}
          </button>
        </div>
      </div>

      <div className="editor-content">
        <div className="node-palette">
          <h3>节点库</h3>
          <NodePalette onNodeAdd={handleNodeAdd} />
        </div>

        <div className="workflow-canvas">
          <WorkflowCanvas
            nodes={nodes}
            connections={connections}
            selectedNode={selectedNode}
            onNodeSelect={setSelectedNode}
            onNodeUpdate={handleNodeUpdate}
            onConnectionAdd={handleConnectionAdd}
          />
        </div>

        <div className="node-properties">
          <h3>节点属性</h3>
          {selectedNode ? (
            <NodePropertiesPanel
              node={selectedNode}
              onUpdate={(updates) => handleNodeUpdate(selectedNode.id, updates)}
            />
          ) : (
            <p>选择一个节点以编辑属性</p>
          )}
        </div>
      </div>
    </div>
  );
};
```
