### 前端状态管理

```typescript
// parseResumeSlice.ts (Redux Toolkit)
const parseResumeSlice = createSlice({
  name: 'parseResume',
  initialState: {
    tasks: {},
    activeTaskId: null,
    isProcessing: false,
  },
  reducers: {
    startParsing: (state, action) => {
      const { fileId, options } = action.payload;
      state.tasks[fileId] = {
        id: fileId,
        status: 'processing',
        progress: 0,
        options,
      };
      state.activeTaskId = fileId;
      state.isProcessing = true;
    },

    updateProgress: (state, action) => {
      const { taskId, progress, status } = action.payload;
      if (state.tasks[taskId]) {
        state.tasks[taskId].progress = progress;
        state.tasks[taskId].status = status;
      }
    },

    parsingComplete: (state, action) => {
      const { taskId, result } = action.payload;
      if (state.tasks[taskId]) {
        state.tasks[taskId].status = 'completed';
        state.tasks[taskId].result = result;
        state.tasks[taskId].progress = 100;
      }
      state.isProcessing = false;
    },
  },
});
```

```typescript
// ParseCache.ts
export class ParseCache {
  private cache = new Map<string, CachedParseResult>();
  private cacheExpiry = 24 * 60 * 60 * 1000; // 24小时

  async getCachedResult(fileHash: string): Promise<ParsedResume | null> {
    const cached = this.cache.get(fileHash);
    if (cached && Date.now() - cached.timestamp < this.cacheExpiry) {
      return cached.result;
    }
    return null;
  }

  setCachedResult(fileHash: string, result: ParsedResume): void {
    this.cache.set(fileHash, {
      result,
      timestamp: Date.now(),
    });
  }
}
```

### 批处理优化

```typescript
// BatchProcessor.ts
export class BatchProcessor {
  private queue: ParseTask[] = [];
  private processing = false;
  private maxConcurrent = 3;

  async addToQueue(task: ParseTask): Promise<string> {
    this.queue.push(task);
    if (!this.processing) {
      this.processQueue();
    }
    return task.id;
  }

  private async processQueue(): Promise<void> {
    this.processing = true;

    while (this.queue.length > 0) {
      const batch = this.queue.splice(0, this.maxConcurrent);
      await Promise.all(batch.map(task => this.processTask(task)));
    }

    this.processing = false;
  }
}
```

### 重试机制

```typescript
// RetryManager.ts
export class RetryManager {
  async executeWithRetry<T>(
    operation: () => Promise<T>,
    maxRetries: number = 3,
    delay: number = 1000
  ): Promise<T> {
    for (let attempt = 1; attempt <= maxRetries; attempt++) {
      try {
        return await operation();
      } catch (error) {
        if (attempt === maxRetries || !this.isRetryableError(error)) {
          throw error;
        }

        await this.delay(delay * Math.pow(2, attempt - 1));
      }
    }

    throw new Error('Max retries exceeded');
  }

  private isRetryableError(error: any): boolean {
    return error.retryable === true;
  }

  private delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

## 安全考虑

### 数据安全

```typescript
// SecurityService.ts
export class SecurityService {
  sanitizeParsedData(data: ParsedResume): ParsedResume {
    // 移除敏感信息
    return {
      ...data,
      personalInfo: {
        ...data.personalInfo,
        phone: this.maskPhoneNumber(data.personalInfo.phone),
        email: this.maskEmail(data.personalInfo.email),
      },
    };
  }

  private maskPhoneNumber(phone: string): string {
    // 手机号脱敏处理
    return phone.replace(/(\d{3})\d{4}(\d{4})/, '$1****$2');
  }

  private maskEmail(email: string): string {
    // 邮箱脱敏处理
    const [username, domain] = email.split('@');
    return `${username.slice(0, 2)}****@${domain}`;
  }
}
```
