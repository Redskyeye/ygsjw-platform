```typescript
// tests/e2e/global-setup.ts
import { chromium, FullConfig } from '@playwright/test';
import path from 'path';

async function globalSetup(config: FullConfig) {
  console.log('🚀 开始E2E测试环境设置...');

  // 获取浏览器实例
  const browser = await chromium.launch();
  const context = await browser.newContext();
  const page = await context.newPage();

  try {
    // 设置测试数据
    await setupTestData(page);

    // 验证应用可访问性
    await verifyApplicationHealth(page);

    console.log('✅ E2E测试环境设置完成');
  } catch (error) {
    console.error('❌ E2E测试环境设置失败:', error);
    throw error;
  } finally {
    await browser.close();
  }
}

async function setupTestData(page: any) {
  // 创建测试用户
  await page.goto('/api/test/setup');
  await page.waitForResponse(response =>
    response.url().includes('/api/test/setup') && response.status() === 200
  );
}

async function verifyApplicationHealth(page: any) {
  // 检查应用健康状态
  const response = await page.goto('/api/health');
  if (!response || response.status() !== 200) {
    throw new Error('应用健康检查失败');
  }
}

export default globalSetup;
```

#### 测试辅助工具
```typescript
// tests/e2e/helpers/test-helpers.ts
import { Page, Locator, expect } from '@playwright/test';

export class TestHelpers {
  constructor(private page: Page) {}

  // 等待页面加载完成
  async waitForPageLoad(): Promise<void> {
    await this.page.waitForLoadState('networkidle');
    await this.page.waitForLoadState('domcontentloaded');
  }

  // 等待元素可见并可交互
  async waitForElement(selector: string): Promise<Locator> {
    const element = this.page.locator(selector);
    await element.waitFor({ state: 'visible' });
    await element.waitFor({ state: 'attached' });
    return element;
  }

  // 安全点击元素
  async safeClick(selector: string): Promise<void> {
    const element = await this.waitForElement(selector);
    await element.scrollIntoViewIfNeeded();
    await element.click();
  }

  // 安全输入文本
  async safeType(selector: string, text: string): Promise<void> {
    const element = await this.waitForElement(selector);
    await element.clear();
    await element.fill(text);
  }

  // 等待API响应
  async waitForApiResponse(urlPattern: string): Promise<any> {
    return this.page.waitForResponse(response =>
      response.url().includes(urlPattern)
    );
  }

  // 检查元素是否存在
  async elementExists(selector: string): Promise<boolean> {
    const element = this.page.locator(selector);
    const count = await element.count();
    return count > 0;
  }

  // 获取元素文本
  async getElementText(selector: string): Promise<string> {
    const element = await this.waitForElement(selector);
    return await element.textContent() || '';
  }

  // 截图
  async takeScreenshot(name: string): Promise<void> {
    await this.page.screenshot({
      path: `screenshots/${name}-${Date.now()}.png`,
      fullPage: true
    });
  }

  // 等待Toast消息
  async waitForToast(message: string): Promise<void> {
    await this.page.waitForSelector(`[data-testid="toast"]:has-text("${message}")`);
  }

  // 检查可访问性
  async checkAccessibility(): Promise<void> {
    // 使用playwright-accessibility-checker或其他工具
    await this.page.waitForTimeout(1000); // 等待可访问性树更新

    // 基础可访问性检查
    const landmarks = await this.page.locator('[role="main"], [role="navigation"], [role="banner"]').count();
    expect(landmarks).toBeGreaterThan(0);
  }

  // 模拟网络条件
  async setNetworkCondition(condition: 'slow3g' | 'fast3g' | 'offline' | 'online'): Promise<void> {
    const contexts = {
      slow3g: { downloadThroughput: 500 * 1024 / 8, uploadThroughput: 500 * 1024 / 8, latency: 400 },
      fast3g: { downloadThroughput: 1.6 * 1024 * 1024 / 8, uploadThroughput: 750 * 1024 / 8, latency: 150 },
      offline: { downloadThroughput: 0, uploadThroughput: 0, offline: true },
      online: { downloadThroughput: -1, uploadThroughput: -1, offline: false }
    };

    await this.page.context().route('**/*', route => {
      if (condition === 'offline') {
        route.abort();
      } else {
        route.continue();
      }
    });
  }

  // 清理本地存储
  async clearStorage(): Promise<void> {
    await this.page.evaluate(() => {
      localStorage.clear();
      sessionStorage.clear();
    });
  }
}
```

```yaml
# .github/workflows/e2e-tests.yml
name: E2E Tests

on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main ]

jobs:
  e2e-tests:
    runs-on: ubuntu-latest

    strategy:
      matrix:
        shard: [1/4, 2/4, 3/4, 4/4]

    steps:
    - uses: actions/checkout@v3

    - name: Setup Node.js
      uses: actions/setup-node@v3
      with:
        node-version: '18'
        cache: 'npm'

    - name: Install dependencies
      run: npm ci

    - name: Install Playwright Browsers
      run: npx playwright install --with-deps

    - name: Build application
      run: npm run build

    - name: Run E2E tests
      run: npx playwright test --shard=${{ matrix.shard }}
      env:
        E2E_BASE_URL: http://localhost:3000

    - uses: actions/upload-artifact@v3
      if: always()
      with:
        name: playwright-report-${{ matrix.shard }}
        path: playwright-report/
        retention-days: 30

    - uses: actions/upload-artifact@v3
      if: always()
      with:
        name: test-results-${{ matrix.shard }}
        path: test-results/
        retention-days: 30
```
