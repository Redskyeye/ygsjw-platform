```typescript
// hooks/useLazyLoad.ts
import { useState, useEffect } from 'react';

export const useLazyLoad = (importFunc: () => Promise<any>) => {
  const [component, setComponent] = useState<React.ComponentType | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    let isMounted = true;

    const loadComponent = async () => {
      if (!component) {
        setLoading(true);
        try {
          const module = await importFunc();
          if (isMounted) {
            setComponent(() => module.default);
          }
        } catch (err) {
          if (isMounted) {
            setError(err as Error);
          }
        } finally {
          if (isMounted) {
            setLoading(false);
          }
        }
      }
    };

    loadComponent();

    return () => {
      isMounted = false;
    };
  }, [importFunc, component]);

  return { component, loading, error };
};
```

```typescript
// lib/cache.ts
import { NextRequest, NextResponse } from 'next/server';

// 静态资源缓存策略
export const staticAssetCache = {
  // 图片和字体文件长期缓存
  'images': 'public, max-age=31536000, immutable',
  'fonts': 'public, max-age=31536000, immutable',
  // JS/CSS文件缓存
  'scripts': 'public, max-age=31536000, immutable',
  'styles': 'public, max-age=31536000, immutable',
  // HTML文件短期缓存
  'html': 'public, max-age=3600, must-revalidate'
};

// API响应缓存策略
export const apiCacheStrategy = {
  // 用户数据短期缓存
  'user-data': 'public, max-age=300, stale-while-revalidate=60',
  // AI模型响应中期缓存
  'ai-response': 'private, max-age=1800, must-revalidate',
  // 配置数据长期缓存
  'config': 'public, max-age=86400, stale-while-revalidate=3600'
};

// 中间件缓存控制
export function cacheMiddleware(request: NextRequest) {
  const response = NextResponse.next();

  // 根据路径设置缓存头
  if (request.nextUrl.pathname.startsWith('/api/')) {
    const cacheKey = getApiCacheKey(request.nextUrl.pathname);
    response.headers.set('Cache-Control', apiCacheStrategy[cacheKey] || 'no-cache');
  } else if (request.nextUrl.pathname.includes('.')) {
    // 静态资源
    const ext = request.nextUrl.pathname.split('.').pop();
    const cacheType = getStaticCacheType(ext);
    response.headers.set('Cache-Control', staticAssetCache[cacheType]);
  }

  return response;
}

function getApiCacheKey(pathname: string): keyof typeof apiCacheStrategy {
  if (pathname.includes('/user/')) return 'user-data';
  if (pathname.includes('/ai/')) return 'ai-response';
  if (pathname.includes('/config/')) return 'config';
  return 'user-data';
}

function getStaticCacheType(ext: string): keyof typeof staticAssetCache {
  switch (ext) {
    case 'jpg':
    case 'jpeg':
    case 'png':
    case 'webp':
    case 'svg':
      return 'images';
    case 'woff':
    case 'woff2':
    case 'ttf':
      return 'fonts';
    case 'js':
      return 'scripts';
    case 'css':
      return 'styles';
    case 'html':
      return 'html';
    default:
      return 'scripts';
  }
}
```

#### 数据缓存Hook
```typescript
// hooks/useCache.ts
import { useState, useEffect, useCallback } from 'react';

interface CacheEntry<T> {
  data: T;
  timestamp: number;
  ttl: number;
}

class LocalCache {
  private storage = new Map<string, CacheEntry<any>>();

  set<T>(key: string, data: T, ttl: number = 300000): void {
    this.storage.set(key, {
      data,
      timestamp: Date.now(),
      ttl
    });
  }

  get<T>(key: string): T | null {
    const entry = this.storage.get(key);
    if (!entry) return null;

    if (Date.now() - entry.timestamp > entry.ttl) {
      this.storage.delete(key);
      return null;
    }

    return entry.data;
  }

  clear(): void {
    this.storage.clear();
  }

  delete(key: string): void {
    this.storage.delete(key);
  }
}

export const cache = new LocalCache();

export function useCache<T>(
  key: string,
  fetcher: () => Promise<T>,
  ttl: number = 300000
) {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);

  const fetchData = useCallback(async () => {
    setLoading(true);
    setError(null);

    try {
      // 先检查缓存
      const cachedData = cache.get<T>(key);
      if (cachedData) {
        setData(cachedData);
        setLoading(false);
        return;
      }

      // 缓存未命中，获取数据
      const freshData = await fetcher();
      cache.set(key, freshData, ttl);
      setData(freshData);
    } catch (err) {
      setError(err as Error);
    } finally {
      setLoading(false);
    }
  }, [key, fetcher, ttl]);

  useEffect(() => {
    fetchData();
  }, [fetchData]);

  const refetch = useCallback(() => {
    cache.delete(key);
    fetchData();
  }, [key, fetchData]);

  return { data, loading, error, refetch };
}
```

```typescript
// lib/imageUtils.ts
export function getOptimizedImageUrl(
  src: string,
  width: number,
  quality: number = 75
): string {
  // 如果是Next.js内置图片优化
  if (src.startsWith('/')) {
    return `/_next/image?url=${encodeURIComponent(src)}&w=${width}&q=${quality}`;
  }

  // 外部图片URL优化
  const url = new URL(src);
  url.searchParams.set('w', width.toString());
  url.searchParams.set('q', quality.toString());

  // 添加WebP格式支持
  if (supportsWebP()) {
    url.searchParams.set('fm', 'webp');
  }

  return url.toString();
}

function supportsWebP(): boolean {
  if (typeof window === 'undefined') return true;

  const canvas = document.createElement('canvas');
  canvas.width = 1;
  canvas.height = 1;
  return canvas.toDataURL('image/webp').indexOf('data:image/webp') === 0;
}
```

```typescript
// scripts/analyze-bundle.js
const { execSync } = require('child_process');
const fs = require('fs');
const path = require('path');

async function analyzeBundle() {
  console.log('🔍 开始分析Bundle大小...');

  try {
    // 构建项目
    console.log('📦 构建项目...');
    execSync('npm run build', { stdio: 'inherit' });

    // 使用webpack-bundle-analyzer
    console.log('📊 生成Bundle分析报告...');
    execSync('npx webpack-bundle-analyzer .next/static/chunks/*.js', {
      stdio: 'inherit',
      cwd: process.cwd()
    });

    // 生成大小报告
    const staticDir = '.next/static';
    const chunks = fs.readdirSync(path.join(staticDir, 'chunks'));

    let totalSize = 0;
    const chunkSizes = [];

    for (const chunk of chunks) {
      if (chunk.endsWith('.js')) {
        const filePath = path.join(staticDir, 'chunks', chunk);
        const stats = fs.statSync(filePath);
        const sizeKB = Math.round(stats.size / 1024);
        totalSize += sizeKB;
        chunkSizes.push({ name: chunk, size: sizeKB });
      }
    }

    console.log('\n📈 Bundle分析结果:');
    console.log(`总大小: ${totalSize} KB`);
    console.log('\nChunk详情:');
    chunkSizes
      .sort((a, b) => b.size - a.size)
      .forEach(chunk => {
        console.log(`  ${chunk.name}: ${chunk.size} KB`);
      });

    // 检查是否超过建议大小
    if (totalSize > 500) {
      console.warn('⚠️ Bundle总大小超过建议值(500KB)，建议进一步优化');
    }

  } catch (error) {
    console.error('❌ Bundle分析失败:', error.message);
    process.exit(1);
  }
}

if (require.main === module) {
  analyzeBundle();
}

module.exports = { analyzeBundle };
```

```typescript
import { getCLS, getFID, getFCP, getLCP, getTTFB } from 'web-vitals';

interface PerformanceMetrics {
  cls: number;
  fid: number;
  fcp: number;
  lcp: number;
  ttfb: number;
}

export function collectPerformanceMetrics(callback: (metrics: PerformanceMetrics) => void) {
  const metrics: Partial<PerformanceMetrics> = {};

  getCLS((metric) => {
    metrics.cls = metric.value;
    if (Object.keys(metrics).length === 5) {
      callback(metrics as PerformanceMetrics);
    }
  });

  getFID((metric) => {
    metrics.fid = metric.value;
    if (Object.keys(metrics).length === 5) {
      callback(metrics as PerformanceMetrics);
    }
  });

  getFCP((metric) => {
    metrics.fcp = metric.value;
    if (Object.keys(metrics).length === 5) {
      callback(metrics as PerformanceMetrics);
    }
  });

  getLCP((metric) => {
    metrics.lcp = metric.value;
    if (Object.keys(metrics).length === 5) {
      callback(metrics as PerformanceMetrics);
    }
  });

  getTTFB((metric) => {
    metrics.ttfb = metric.value;
    if (Object.keys(metrics).length === 5) {
      callback(metrics as PerformanceMetrics);
    }
  });
}

// 性能指标上报
export function reportPerformanceMetrics(metrics: PerformanceMetrics) {
  // 上报到分析服务
  if (typeof window !== 'undefined' && window.gtag) {
    window.gtag('event', 'performance_metrics', {
      custom_map: {
        dimension1: 'cls',
        dimension2: 'fid',
        dimension3: 'fcp',
        dimension4: 'lcp',
        dimension5: 'ttfb'
      },
      cls: metrics.cls,
      fid: metrics.fid,
      fcp: metrics.fcp,
      lcp: metrics.lcp,
      ttfb: metrics.ttfb
    });
  }

  // 本地存储用于调试
  localStorage.setItem('performance_metrics', JSON.stringify({
    ...metrics,
    timestamp: Date.now()
  }));
}
```
