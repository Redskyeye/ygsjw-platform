```ts
// src/lib/n8n/client.ts
import { N8NConfig } from './config'

export class N8NClient {
  private config: N8NConfig
  private baseUrl: string

  constructor(config: N8NConfig) {
    this.config = config
    this.baseUrl = config.instanceUrl.replace(/\/$/, '')
  }

  private async request<T>(
    endpoint: string,
    options: RequestInit = {}
  ): Promise<T> {
    const url = `${this.baseUrl}${endpoint}`
    const headers = {
      'Content-Type': 'application/json',
      'X-N8N-API-KEY': this.config.apiKey,
      ...options.headers,
    }

    let lastError: Error

    for (let attempt = 1; attempt <= this.config.retryAttempts; attempt++) {
      try {
        const response = await fetch(url, {
          ...options,
          headers,
          signal: AbortSignal.timeout(this.config.timeout),
        })

        if (!response.ok) {
          const error = await response.json().catch(() => ({}))
          throw new Error(
            `N8N API Error: ${response.status} ${response.statusText} - ${error.message || 'Unknown error'}`
          )
        }

        return await response.json()
      } catch (error) {
        lastError = error as Error
        console.warn(`N8N API request attempt ${attempt} failed:`, error)

        if (attempt < this.config.retryAttempts) {
          await new Promise(resolve =>
            setTimeout(resolve, this.config.retryDelay * attempt)
          )
        }
      }
    }

    throw lastError!
  }

  // 工作流管理
  async getWorkflows() {
    return this.request('/api/v1/workflows')
  }

  async getWorkflow(id: string) {
    return this.request(`/api/v1/workflows/${id}`)
  }

  async createWorkflow(workflow: any) {
    return this.request('/api/v1/workflows', {
      method: 'POST',
      body: JSON.stringify(workflow),
    })
  }

  async updateWorkflow(id: string, workflow: any) {
    return this.request(`/api/v1/workflows/${id}`, {
      method: 'PUT',
      body: JSON.stringify(workflow),
    })
  }

  async deleteWorkflow(id: string) {
    return this.request(`/api/v1/workflows/${id}`, {
      method: 'DELETE',
    })
  }

  async activateWorkflow(id: string) {
    return this.request(`/api/v1/workflows/${id}/activate`, {
      method: 'POST',
    })
  }

  async deactivateWorkflow(id: string) {
    return this.request(`/api/v1/workflows/${id}/deactivate`, {
      method: 'POST',
    })
  }

  // 执行管理
  async executeWorkflow(id: string, data?: any) {
    return this.request(`/api/v1/workflows/${id}/execute`, {
      method: 'POST',
      body: JSON.stringify({ data }),
    })
  }

  async getExecutions(workflowId?: string) {
    const params = workflowId ? `?workflowId=${workflowId}` : ''
    return this.request(`/api/v1/executions${params}`)
  }

  async getExecution(id: string) {
    return this.request(`/api/v1/executions/${id}`)
  }

  async stopExecution(id: string) {
    return this.request(`/api/v1/executions/${id}/stop`, {
      method: 'POST',
    })
  }

  // Webhook管理
  async getWebhooks() {
    return this.request('/api/v1/webhooks')
  }

  async getWebhook(id: string) {
    return this.request(`/api/v1/webhooks/${id}`)
  }

  async createWebhook(webhook: any) {
    return this.request('/api/v1/webhooks', {
      method: 'POST',
      body: JSON.stringify(webhook),
    })
  }

  async deleteWebhook(id: string) {
    return this.request(`/api/v1/webhooks/${id}`, {
      method: 'DELETE',
    })
  }

  // 凭证管理
  async getCredentials() {
    return this.request('/api/v1/credentials')
  }

  async createCredential(credential: any) {
    return this.request('/api/v1/credentials', {
      method: 'POST',
      body: JSON.stringify(credential),
    })
  }

  async testCredential(id: string, data: any) {
    return this.request(`/api/v1/credentials/${id}/test`, {
      method: 'POST',
      body: JSON.stringify({ data }),
    })
  }
}
```

```ts
import { ProjectWorkflowTemplate, TaskWorkflowTemplate, AIWorkflowTemplate } from './types'
import {
  projectCreationTemplate,
  taskAssignmentTemplate,
  projectReportTemplate,
  taskReminderTemplate,
  aiChatTemplate,
  aiWorkflowTemplate,
} from './definitions'

export class WorkflowTemplateManager {
  private templates: Map<string, any> = new Map()

  constructor() {
    this.registerDefaultTemplates()
  }

  private registerDefaultTemplates() {
    // 项目相关模板
    this.register('project-creation', projectCreationTemplate)
    this.register('task-assignment', taskAssignmentTemplate)
    this.register('project-report', projectReportTemplate)
    this.register('task-reminder', taskReminderTemplate)

    // AI相关模板
    this.register('ai-chat', aiChatTemplate)
    this.register('ai-workflow', aiWorkflowTemplate)
  }

  register(id: string, template: any) {
    this.templates.set(id, template)
  }

  get(id: string) {
    return this.templates.get(id)
  }

  getAll() {
    return Array.from(this.templates.entries()).map(([id, template]) => ({
      id,
      ...template,
    }))
  }

  // 根据项目需求定制模板
  customizeTemplate(templateId: string, customizations: any) {
    const template = this.get(templateId)
    if (!template) {
      throw new Error(`Template ${templateId} not found`)
    }

    return {
      ...template,
      name: customizations.name || template.name,
      description: customizations.description || template.description,
      nodes: this.customizeNodes(template.nodes, customizations),
      connections: this.customizeConnections(template.connections, customizations),
    }
  }

  private customizeNodes(nodes: any[], customizations: any) {
    return nodes.map(node => {
      const customNode = customizations.nodes?.[node.name]
      if (customNode) {
        return {
          ...node,
          parameters: { ...node.parameters, ...customNode.parameters },
        }
      }
      return node
    })
  }

  private customizeConnections(connections: any, customizations: any) {
    // 根据需要自定义连接
    return connections
  }
}
```

```ts
// src/lib/services/n8n.service.ts
import { n8nClient } from '@/lib/n8n'
import { WorkflowTemplateManager } from '@/lib/n8n/templates'
import { prisma } from '@/lib/prisma'
import type { Workflow, WorkflowExecution } from '@prisma/client'

export interface CreateWorkflowData {
  name: string
  description?: string
  templateId?: string
  projectId?: string
  config?: any
}

export interface ExecuteWorkflowData {
  workflowId: string
  data?: any
  context?: any
}

export class N8NService {
  private templateManager = new WorkflowTemplateManager()

  // 工作流生命周期管理
  async createWorkflow(data: CreateWorkflowData) {
    let workflowData: any

    if (data.templateId) {
      // 使用模板创建工作流
      const template = this.templateManager.get(data.templateId)
      if (!template) {
        throw new Error(`Template ${data.templateId} not found`)
      }

      workflowData = this.templateManager.customizeTemplate(
        data.templateId,
        data.config || {}
      )
    } else {
      // 创建空白工作流
      workflowData = {
        name: data.name,
        description: data.description,
        nodes: [],
        connections: {},
        active: false,
        settings: {},
      }
    }

    // 在N8N中创建工作流
    const n8nWorkflow = await n8nClient.createWorkflow(workflowData)

    // 在数据库中保存工作流记录
    const workflow = await prisma.workflow.create({
      data: {
        name: data.name,
        description: data.description,
        n8nWorkflowId: n8nWorkflow.id,
        templateId: data.templateId,
        projectId: data.projectId,
        config: data.config,
        status: 'DRAFT',
        createdBy: 'system', // 需要从认证上下文获取
      },
    })

    return { workflow, n8nWorkflow }
  }

  async activateWorkflow(id: string) {
    const workflow = await prisma.workflow.findUnique({
      where: { id },
    })

    if (!workflow) {
      throw new Error('Workflow not found')
    }

    // 在N8N中激活工作流
    await n8nClient.activateWorkflow(workflow.n8nWorkflowId)

    // 更新数据库状态
    return prisma.workflow.update({
      where: { id },
      data: {
        status: 'ACTIVE',
        activatedAt: new Date(),
      },
    })
  }

  async deactivateWorkflow(id: string) {
    const workflow = await prisma.workflow.findUnique({
      where: { id },
    })

    if (!workflow) {
      throw new Error('Workflow not found')
    }

    // 在N8N中停用工作流
    await n8nClient.deactivateWorkflow(workflow.n8nWorkflowId)

    // 更新数据库状态
    return prisma.workflow.update({
      where: { id },
      data: {
        status: 'INACTIVE',
      },
    })
  }

  async deleteWorkflow(id: string) {
    const workflow = await prisma.workflow.findUnique({
      where: { id },
    })

    if (!workflow) {
      throw new Error('Workflow not found')
    }

    // 在N8N中删除工作流
    await n8nClient.deleteWorkflow(workflow.n8nWorkflowId)

    // 从数据库中删除记录
    return prisma.workflow.delete({
      where: { id },
    })
  }

  // 工作流执行管理
  async executeWorkflow(data: ExecuteWorkflowData) {
    const workflow = await prisma.workflow.findUnique({
      where: { id: data.workflowId },
    })

    if (!workflow) {
      throw new Error('Workflow not found')
    }

    if (workflow.status !== 'ACTIVE') {
      throw new Error('Workflow is not active')
    }

    // 在N8N中执行工作流
    const execution = await n8nClient.executeWorkflow(
      workflow.n8nWorkflowId,
      data.data
    )

    // 保存执行记录
    const workflowExecution = await prisma.workflowExecution.create({
      data: {
        workflowId: workflow.id,
        n8nExecutionId: execution.id,
        status: execution.finished ? 'COMPLETED' : 'RUNNING',
        startedAt: new Date(),
        data: data.data,
        result: execution.data,
      },
    })

    return { execution: workflowExecution, n8nExecution: execution }
  }

  async getWorkflowExecutions(workflowId: string) {
    return prisma.workflowExecution.findMany({
      where: { workflowId },
      include: {
        workflow: true,
      },
      orderBy: {
        startedAt: 'desc',
      },
    })
  }

  async updateExecutionStatus(executionId: string) {
    const execution = await prisma.workflowExecution.findUnique({
      where: { id: executionId },
    })

    if (!execution) {
      throw new Error('Execution not found')
    }

    // 从N8N获取最新状态
    const n8nExecution = await n8nClient.getExecution(execution.n8nExecutionId)

    const status = n8nExecution.finished ?
      (n8nExecution.stoppedAt ? 'COMPLETED' : 'FAILED') :
      'RUNNING'

    return prisma.workflowExecution.update({
      where: { id: executionId },
      data: {
        status,
        completedAt: n8nExecution.stoppedAt ? new Date(n8nExecution.stoppedAt) : null,
        result: n8nExecution.data,
        error: n8nExecution.error || null,
      },
    })
  }

  // 模板管理
  getTemplates() {
    return this.templateManager.getAll()
  }

  getTemplate(id: string) {
    return this.templateManager.get(id)
  }

  // 项目工作流管理
  async getProjectWorkflows(projectId: string) {
    return prisma.workflow.findMany({
      where: { projectId },
      include: {
        _count: {
          select: {
            executions: true,
          },
        },
      },
      orderBy: {
        createdAt: 'desc',
      },
    })
  }

  async createProjectWorkflows(projectId: string) {
    const project = await prisma.project.findUnique({
      where: { id: projectId },
      include: { owner: true },
    })

    if (!project) {
      throw new Error('Project not found')
    }

    // 为项目创建标准工作流
    const standardWorkflows = [
      {
        name: '项目生命周期管理',
        templateId: 'project-creation',
        config: {
          webhookUrl: `${process.env.API_URL}/api/v1/webhooks/project-created`,
          projectEmail: project.owner.email,
        },
      },
      {
        name: '任务自动化',
        templateId: 'task-assignment',
        config: {
          webhookUrl: `${process.env.API_URL}/api/v1/webhooks/task-assigned`,
        },
      },
      {
        name: 'AI助手集成',
        templateId: 'ai-chat',
        config: {
          openaiApiKey: process.env.OPENAI_API_KEY,
        },
      },
    ]

    const workflows = []
    for (const workflowData of standardWorkflows) {
      try {
        const workflow = await this.createWorkflow({
          ...workflowData,
          projectId,
        })
        workflows.push(workflow)
      } catch (error) {
        console.error(`Failed to create workflow ${workflowData.name}:`, error)
      }
    }

    return workflows
  }

  // Webhook处理
  async handleWebhook(event: string, data: any) {
    // 根据事件类型触发相应的工作流
    const eventWorkflows = await prisma.workflow.findMany({
      where: {
        status: 'ACTIVE',
        config: {
          path: ['event'],
          equals: event,
        },
      },
    })

    const executions = []
    for (const workflow of eventWorkflows) {
      try {
        const execution = await this.executeWorkflow({
          workflowId: workflow.id,
          data,
        })
        executions.push(execution)
      } catch (error) {
        console.error(`Failed to execute workflow ${workflow.id}:`, error)
      }
    }

    return executions
  }

  // 监控和统计
  async getWorkflowStats() {
    const [total, active, inactive, totalExecutions] = await Promise.all([
      prisma.workflow.count(),
      prisma.workflow.count({ where: { status: 'ACTIVE' } }),
      prisma.workflow.count({ where: { status: 'INACTIVE' } }),
      prisma.workflowExecution.count(),
    ])

    return {
      total,
      active,
      inactive,
      totalExecutions,
    }
  }

  async getExecutionsByStatus(status: string) {
    return prisma.workflowExecution.count({
      where: { status },
    })
  }
}
```

```tsx
// src/components/n8n/workflow-manager.tsx
"use client"

import { useState, useEffect } from 'react'
import { Button } from '@/components/ui/button'
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
import { Badge } from '@/components/ui/badge'
import { Play, Pause, Trash2, Settings, Activity } from 'lucide-react'
import { N8NService } from '@/lib/services/n8n.service'

interface Workflow {
  id: string
  name: string
  description?: string
  status: 'ACTIVE' | 'INACTIVE' | 'DRAFT'
  n8nWorkflowId: string
  createdAt: string
  activatedAt?: string
  _count: {
    executions: number
  }
}

interface WorkflowManagerProps {
  projectId?: string
}

export function WorkflowManager({ projectId }: WorkflowManagerProps) {
  const [workflows, setWorkflows] = useState<Workflow[]>([])
  const [loading, setLoading] = useState(true)
  const [actionLoading, setActionLoading] = useState<string | null>(null)

  const n8nService = new N8NService()

  useEffect(() => {
    loadWorkflows()
  }, [projectId])

  const loadWorkflows = async () => {
    try {
      setLoading(true)
      const data = projectId
        ? await n8nService.getProjectWorkflows(projectId)
        : await n8nService.getWorkflows()
      setWorkflows(data)
    } catch (error) {
      console.error('Failed to load workflows:', error)
    } finally {
      setLoading(false)
    }
  }

  const handleActivate = async (workflowId: string) => {
    try {
      setActionLoading(workflowId)
      await n8nService.activateWorkflow(workflowId)
      await loadWorkflows()
    } catch (error) {
      console.error('Failed to activate workflow:', error)
    } finally {
      setActionLoading(null)
    }
  }

  const handleDeactivate = async (workflowId: string) => {
    try {
      setActionLoading(workflowId)
      await n8nService.deactivateWorkflow(workflowId)
      await loadWorkflows()
    } catch (error) {
      console.error('Failed to deactivate workflow:', error)
    } finally {
      setActionLoading(null)
    }
  }

  const handleDelete = async (workflowId: string) => {
    if (!confirm('确定要删除这个工作流吗？')) return

    try {
      setActionLoading(workflowId)
      await n8nService.deleteWorkflow(workflowId)
      await loadWorkflows()
    } catch (error) {
      console.error('Failed to delete workflow:', error)
    } finally {
      setActionLoading(null)
    }
  }

  const getStatusColor = (status: string) => {
    switch (status) {
      case 'ACTIVE':
        return 'bg-green-100 text-green-800'
      case 'INACTIVE':
        return 'bg-gray-100 text-gray-800'
      case 'DRAFT':
        return 'bg-yellow-100 text-yellow-800'
      default:
        return 'bg-gray-100 text-gray-800'
    }
  }

  const getStatusText = (status: string) => {
    switch (status) {
      case 'ACTIVE':
        return '活跃'
      case 'INACTIVE':
        return '停用'
      case 'DRAFT':
        return '草稿'
      default:
        return status
    }
  }

  if (loading) {
    return <div>加载中...</div>
  }

  return (
    <div className="space-y-4">
      <div className="flex justify-between items-center">
        <h2 className="text-lg font-semibold">工作流管理</h2>
        <Button>创建工作流</Button>
      </div>

      {workflows.length === 0 ? (
        <Card>
          <CardContent className="p-6 text-center">
            <p className="text-muted-foreground">暂无工作流</p>
          </CardContent>
        </Card>
      ) : (
        <div className="grid gap-4">
          {workflows.map((workflow) => (
            <Card key={workflow.id}>
              <CardHeader>
                <div className="flex justify-between items-start">
                  <div>
                    <CardTitle className="text-base">{workflow.name}</CardTitle>
                    {workflow.description && (
                      <CardDescription className="mt-1">
                        {workflow.description}
                      </CardDescription>
                    )}
                  </div>
                  <Badge className={getStatusColor(workflow.status)}>
                    {getStatusText(workflow.status)}
                  </Badge>
                </div>
              </CardHeader>
              <CardContent>
                <div className="flex justify-between items-center">
                  <div className="flex items-center space-x-4 text-sm text-muted-foreground">
                    <div className="flex items-center">
                      <Activity className="h-4 w-4 mr-1" />
                      {workflow._count.executions} 次执行
                    </div>
                    {workflow.activatedAt && (
                      <div>
                        激活于 {new Date(workflow.activatedAt).toLocaleDateString()}
                      </div>
                    )}
                  </div>
                  <div className="flex space-x-2">
                    {workflow.status === 'ACTIVE' ? (
                      <Button
                        variant="outline"
                        size="sm"
                        onClick={() => handleDeactivate(workflow.id)}
                        disabled={actionLoading === workflow.id}
                      >
                        {actionLoading === workflow.id ? (
                          <>处理中...</>
                        ) : (
                          <>
                            <Pause className="h-4 w-4 mr-1" />
                            停用
                          </>
                        )}
                      </Button>
                    ) : (
                      <Button
                        variant="outline"
                        size="sm"
                        onClick={() => handleActivate(workflow.id)}
                        disabled={actionLoading === workflow.id}
                      >
                        {actionLoading === workflow.id ? (
                          <>处理中...</>
                        ) : (
                          <>
                            <Play className="h-4 w-4 mr-1" />
                            激活
                          </>
                        )}
                      </Button>
                    )}
                    <Button variant="outline" size="sm">
                      <Settings className="h-4 w-4" />
                    </Button>
                    <Button
                      variant="outline"
                      size="sm"
                      onClick={() => handleDelete(workflow.id)}
                      disabled={actionLoading === workflow.id}
                    >
                      {actionLoading === workflow.id ? (
                        <>处理中...</>
                      ) : (
                        <Trash2 className="h-4 w-4" />
                      )}
                    </Button>
                  </div>
                </div>
              </CardContent>
            </Card>
          ))}
        </div>
      )}
    </div>
  )
}
```
