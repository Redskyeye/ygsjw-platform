// SubmissionWorkflowManager.ts
export class SubmissionWorkflowManager {
  private currentStep: SubmissionStep = 'parse';
  private workflowData: WorkflowData = {};
  private stepHistory: SubmissionStep[] = [];

  async startWorkflow(file: File): Promise<WorkflowResult> {
    this.workflowData = {
      originalFile: file,
      startTime: new Date(),
      steps: []
    };

    return await this.executeStep('parse');
  }

  async nextStep(stepData?: any): Promise<WorkflowResult> {
    const currentStepIndex = this.getStepIndex(this.currentStep);
    const nextStep = this.getStepByIndex(currentStepIndex + 1);

    if (!nextStep) {
      throw new Error('No next step available');
    }

    return await this.executeStep(nextStep, stepData);
  }

  async previousStep(): Promise<WorkflowResult> {
    if (this.stepHistory.length === 0) {
      throw new Error('No previous step available');
    }

    const previousStep = this.stepHistory.pop()!;
    this.currentStep = previousStep;

    return {
      step: previousStep,
      data: this.workflowData,
      canProceed: true,
      canGoBack: this.stepHistory.length > 0
    };
  }

  private async executeStep(step: SubmissionStep, data?: any): Promise<WorkflowResult> {
    try {
      let result: any;

      switch (step) {
        case 'parse':
          result = await this.executeParseStep();
          break;
        case 'confirm':
          result = await this.executeConfirmStep(data);
          break;
        case 'optimize':
          result = await this.executeOptimizeStep(data);
          break;
        case 'preview':
          result = await this.executePreviewStep();
          break;
        case 'submit':
          result = await this.executeSubmitStep();
          break;
        default:
          throw new Error(`Unknown step: ${step}`);
      }

      this.workflowData[step] = result;
      this.stepHistory.push(this.currentStep);
      this.currentStep = step;

      return {
        step,
        data: this.workflowData,
        canProceed: await this.canProceedToNext(step),
        canGoBack: this.stepHistory.length > 0
      };

    } catch (error) {
      throw new Error(`Failed to execute step ${step}: ${error.message}`);
    }
  }

  private async executeParseStep(): Promise<ParseResult> {
    const file = this.workflowData.originalFile;
    return await this.parseService.parseResume(file);
  }

  private async executeConfirmStep(confirmedData: ConfirmedParseData): Promise<ConfirmedParseData> {
    // 验证确认的数据
    if (!this.validateConfirmedData(confirmedData)) {
      throw new Error('Invalid confirmed data');
    }
    return confirmedData;
  }

  private async executeOptimizeStep(config: OptimizationConfig): Promise<OptimizationResult> {
    const confirmedData = this.workflowData.confirm;
    return await this.optimizationService.optimize(confirmedData, config);
  }

  private async executePreviewStep(): Promise<PreviewData> {
    const optimizedData = this.workflowData.optimize;
    return await this.generatePreview(optimizedData);
  }

  private async executeSubmitStep(): Promise<SubmissionResult> {
    const finalData = this.workflowData.preview;
    return await this.submitOptimizedResume(finalData);
  }
}

// FinalSubmissionHandler.ts
export class FinalSubmissionHandler {
  async submitOptimizedResume(
    submissionData: FinalSubmissionData
  ): Promise<SubmissionResult> {
    // 1. 生成最终文档
    const finalDocument = await this.generateFinalDocument(submissionData);

    // 2. 创建提交记录
    const submissionRecord = await this.createSubmissionRecord(submissionData);

    // 3. 保存到存储
    const savedDocument = await this.saveDocument(finalDocument, submissionRecord);

    // 4. 发送通知
    await this.sendSubmissionNotification(submissionRecord);

    // 5. 更新统计信息
    await this.updateStatistics(submissionData);

    return {
      success: true,
      submissionId: submissionRecord.id,
      documentUrl: savedDocument.url,
      downloadFormats: savedDocument.formats,
      timestamp: submissionRecord.timestamp
    };
  }

  private async generateFinalDocument(
    data: FinalSubmissionData
  ): Promise<FinalDocument> {
    const formats = ['pdf', 'docx', 'html'];
    const documents: Record<string, Buffer> = {};

    for (const format of formats) {
      documents[format] = await this.documentGenerator.generate(
        data.optimizedData,
        format,
        {
          template: 'professional',
          includeMetadata: true,
          watermark: false
        }
      );
    }

    return {
      documents,
      metadata: {
        originalFilename: data.originalData.filename,
        optimizationConfig: data.config,
        optimizationScore: data.optimizedData.optimization.score,
        generatedAt: new Date()
      }
    };
  }

  private async createSubmissionRecord(
    data: FinalSubmissionData
  ): Promise<SubmissionRecord> {
    const record: SubmissionRecord = {
      id: generateId(),
      userId: data.userId || 'anonymous',
      originalData: data.originalData,
      optimizedData: data.optimizedData,
      config: data.config,
      timestamp: new Date(),
      ipAddress: await this.getClientIP(),
      userAgent: await this.getUserAgent(),
      processingTime: this.calculateProcessingTime(data)
    };

    await this.database.saveSubmission(record);
    return record;
  }

  private async saveDocument(
    document: FinalDocument,
    record: SubmissionRecord
  ): Promise<SavedDocument> {
    const savedFormats: string[] = [];

    for (const [format, content] of Object.entries(document.documents)) {
      const filename = `${record.id}_resume.${format}`;
      const path = await this.storageService.save(filename, content);
      savedFormats.push(format);
    }

    // 保存元数据
    await this.storageService.save(
      `${record.id}_metadata.json`,
      JSON.stringify(document.metadata)
    );

    return {
      url: await this.generateDownloadUrl(record.id),
      formats: savedFormats,
      size: this.calculateTotalSize(document.documents)
    };
  }

  private async sendSubmissionNotification(record: SubmissionRecord): Promise<void> {
    if (record.userId && record.userId !== 'anonymous') {
      await this.notificationService.send({
        type: 'submission_complete',
        userId: record.userId,
        data: {
          submissionId: record.id,
          timestamp: record.timestamp,
          downloadUrl: await this.generateDownloadUrl(record.id)
        }
      });
    }
  }
}

// SubmissionErrorHandler.ts
export class SubmissionErrorHandler {
  async handleSubmissionError(
    error: Error,
    context: SubmissionContext
  ): Promise<ErrorHandlingResult> {
    // 1. 记录错误信息
    await this.logError(error, context);

    // 2. 分析错误类型
    const errorType = this.classifyError(error);

    // 3. 确定处理策略
    const strategy = this.determineHandlingStrategy(errorType, context);

    // 4. 执行错误处理
    const result = await this.executeErrorHandling(strategy, error, context);

    return result;
  }

  private classifyError(error: Error): ErrorType {
    if (error.name === 'ValidationError') {
      return 'validation_error';
    } else if (error.name === 'NetworkError') {
      return 'network_error';
    } else if (error.name === 'FileSystemError') {
      return 'file_system_error';
    } else if (error.name === 'ServiceUnavailableError') {
      return 'service_unavailable';
    } else {
      return 'unknown_error';
    }
  }

  private determineHandlingStrategy(
    errorType: ErrorType,
    context: SubmissionContext
  ): ErrorHandlingStrategy {
    switch (errorType) {
      case 'validation_error':
        return 'user_correction_required';
      case 'network_error':
        return 'retry_with_backoff';
      case 'file_system_error':
        return 'rollback_and_retry';
      case 'service_unavailable':
        return 'defer_processing';
      default:
        return 'log_and_continue';
    }
  }

  async rollbackSubmission(submissionId: string): Promise<RollbackResult> {
    try {
      // 1. 获取提交记录
      const submission = await this.getSubmissionRecord(submissionId);
      if (!submission) {
        throw new Error('Submission record not found');
      }

      // 2. 删除生成的文件
      await this.deleteGeneratedFiles(submissionId);

      // 3. 回滚数据库状态
      await this.rollbackDatabaseState(submission);

      // 4. 清理缓存
      await this.clearSubmissionCache(submissionId);

      // 5. 记录回滚操作
      await this.logRollbackOperation(submissionId);

      return {
        success: true,
        message: 'Submission successfully rolled back',
        rollbackTimestamp: new Date()
      };

    } catch (error) {
      return {
        success: false,
        message: `Rollback failed: ${error.message}`,
        rollbackTimestamp: new Date()
      };
    }
  }
}

// ExportService.ts
export class ExportService {
  async exportResume(
    submissionId: string,
    formats: ExportFormat[],
    options: ExportOptions = {}
  ): Promise<ExportResult> {
    const submission = await this.getSubmission(submissionId);
    if (!submission) {
      throw new Error('Submission not found');
    }

    const results: Record<string, ExportFile> = {};

    for (const format of formats) {
      try {
        const file = await this.exportInFormat(
          submission.optimizedData,
          format,
          options
        );
        results[format] = file;
      } catch (error) {
        results[format] = {
          success: false,
          error: error.message
        };
      }
    }

    // 创建压缩包
    if (options.createZip && formats.length > 1) {
      const zipFile = await this.createZipArchive(results, submissionId);
      results.zip = zipFile;
    }

    return {
      submissionId,
      exports: results,
      timestamp: new Date()
    };
  }

  private async exportInFormat(
    data: OptimizedResume,
    format: ExportFormat,
    options: ExportOptions
  ): Promise<ExportFile> {
    switch (format) {
      case 'pdf':
        return await this.exportToPDF(data, options);
      case 'docx':
        return await this.exportToDOCX(data, options);
      case 'html':
        return await this.exportToHTML(data, options);
      case 'txt':
        return await this.exportToTXT(data, options);
      case 'json':
        return await this.exportToJSON(data, options);
      default:
        throw new Error(`Unsupported export format: ${format}`);
    }
  }

  private async exportToPDF(
    data: OptimizedResume,
    options: ExportOptions
  ): Promise<ExportFile> {
    const pdfBuffer = await this.pdfGenerator.generate(data, {
      template: options.template || 'modern',
      pageSize: 'A4',
      margins: 'normal',
      includeWatermark: options.includeWatermark || false
    });

    return {
      success: true,
      filename: `resume_${Date.now()}.pdf`,
      content: pdfBuffer,
      mimeType: 'application/pdf',
      size: pdfBuffer.length
    };
  }

  private async createZipArchive(
    files: Record<string, ExportFile>,
    submissionId: string
  ): Promise<ExportFile> {
    const zip = new JSZip();

    Object.entries(files).forEach(([format, file]) => {
      if (file.success) {
        zip.file(file.filename, file.content);
      }
    });

    const zipBuffer = await zip.generateAsync({ type: 'nodebuffer' });

    return {
      success: true,
      filename: `resume_export_${submissionId}.zip`,
      content: zipBuffer,
      mimeType: 'application/zip',
      size: zipBuffer.length
    };
  }
}
