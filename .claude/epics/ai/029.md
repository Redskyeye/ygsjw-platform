# 029 - 客服集成系统

## 概述

实现完整的在线客服集成系统，支持实时聊天、智能客服机器人、客服工单管理、客户满意度评价等功能，为用户提供全方位的客户服务支持。

## 业务需求

### 核心功能

- **实时聊天**：在线客服与用户的实时对话
- **智能客服**：AI驱动的自动回复和问题解答
- **工单系统**：客服工单的创建、分配和跟踪
- **满意度评价**：服务质量评价和反馈收集
- **知识库集成**：客服知识库的智能检索和推荐

### 客服功能模块

1. **聊天管理**：多客服同时接待，聊天分配和转接
2. **机器人客服**：7x24小时智能客服自动回复
3. **工单系统**：问题工单的创建、分配、处理、关闭
4. **客户管理**：客户信息管理和历史记录查看
5. **质检管理**：客服对话质量检查和评分
6. **统计分析**：客服工作量和满意度统计
7. **多渠道支持**：网站、APP、微信等多渠道接入

// src/customer-service/CustomerServiceEngine.ts
export interface ChatSession {
  id: string;
  customerId: string;
  customerInfo: CustomerInfo;
  assignedAgent?: AgentInfo;
  status: SessionStatus;
  channel: ChannelType;
  messages: ChatMessage[];
  metadata: SessionMetadata;
  createdAt: Date;
  updatedAt: Date;
  closedAt?: Date;
}

export interface ChatMessage {
  id: string;
  sessionId: string;
  sender: MessageSender;
  content: MessageContent;
  type: MessageType;
  timestamp: Date;
  status: MessageStatus;
  metadata: MessageMetadata;
}

export interface AgentInfo {
  id: string;
  name: string;
  avatar: string;
  department: string;
  skills: string[];
  status: AgentStatus;
  currentChats: number;
  maxChats: number;
  rating: AgentRating;
}

export interface Ticket {
  id: string;
  title: string;
  description: string;
  category: TicketCategory;
  priority: TicketPriority;
  status: TicketStatus;
  customerId: string;
  assignedAgent?: string;
  messages: TicketMessage[];
  attachments: Attachment[];
  createdAt: Date;
  updatedAt: Date;
  dueDate?: Date;
  resolvedAt?: Date;
}

export class CustomerServiceEngine {
  private chatManager: ChatManager;
  private botService: BotService;
  private ticketManager: TicketManager;
  private agentManager: AgentManager;
  private knowledgeService: KnowledgeService;
  private notificationService: NotificationService;

  async initiateChat(
    customerInfo: CustomerInfo,
    channel: ChannelType,
    initialMessage?: string
  ): Promise<ChatSession> {
    // 1. 创建聊天会话
    const session: ChatSession = {
      id: generateId(),
      customerId: customerInfo.id,
      customerInfo,
      status: 'waiting',
      channel,
      messages: [],
      metadata: {
        source: channel,
        userAgent: navigator.userAgent,
        referrer: document.referrer,
      },
      createdAt: new Date(),
      updatedAt: new Date(),
    };

    // 2. 保存会话
    const savedSession = await this.chatManager.saveSession(session);

    // 3. 如果有初始消息，添加到会话
    if (initialMessage) {
      await this.addMessage(savedSession.id, {
        sender: { type: 'customer', id: customerInfo.id },
        content: { type: 'text', text: initialMessage },
        type: 'user_message',
      });
    }

    // 4. 尝试分配客服
    await this.attemptAgentAssignment(savedSession);

    // 5. 如果没有可用客服，启动机器人客服
    if (!savedSession.assignedAgent) {
      await this.activateBotService(savedSession);
    }

    return savedSession;
  }

  async addMessage(
    sessionId: string,
    messageData: Omit<
      ChatMessage,
      'id' | 'sessionId' | 'timestamp' | 'status' | 'metadata'
    >
  ): Promise<ChatMessage> {
    // 1. 获取会话信息
    const session = await this.chatManager.getSession(sessionId);
    if (!session) {
      throw new Error(`Session ${sessionId} not found`);
    }

    // 2. 创建消息对象
    const message: ChatMessage = {
      id: generateId(),
      sessionId,
      sender: messageData.sender,
      content: messageData.content,
      type: messageData.type,
      timestamp: new Date(),
      status: 'sent',
      metadata: {
        channel: session.channel,
        deliveryAttempts: 0,
      },
    };

    // 3. 保存消息
    const savedMessage = await this.chatManager.saveMessage(message);

    // 4. 更新会话时间
    await this.chatManager.updateSession(sessionId, {
      updatedAt: new Date(),
    });

    // 5. 处理消息
    await this.processMessage(session, savedMessage);

    // 6. 发送通知
    await this.sendNotification(session, savedMessage);

    return savedMessage;
  }

  private async processMessage(
    session: ChatSession,
    message: ChatMessage
  ): Promise<void> {
    // 如果是客户消息，需要特殊处理
    if (message.sender.type === 'customer') {
      // 1. 更新客服状态
      if (session.assignedAgent) {
        await this.agentManager.updateCustomerMessage(
          session.assignedAgent.id,
          session.id
        );
      }

      // 2. 检查是否需要智能回复
      if (this.shouldUseBotReply(session, message)) {
        await this.generateBotReply(session, message);
      }

      // 3. 检查是否需要创建工单
      if (this.shouldCreateTicket(session, message)) {
        await this.createTicketFromChat(session, message);
      }
    }

    // 如果是客服消息，检查是否需要自动回复
    if (message.sender.type === 'agent') {
      await this.processAgentMessage(session, message);
    }

    // 如果是机器人消息，跟踪效果
    if (message.sender.type === 'bot') {
      await this.trackBotEffectiveness(session, message);
    }
  }

  private async attemptAgentAssignment(session: ChatSession): Promise<void> {
    // 1. 查找可用客服
    const availableAgents = await this.agentManager.findAvailableAgents({
      department: this.getDepartmentFromChannel(session.channel),
      skills: this.getRequiredSkills(session),
      language: session.customerInfo.language || 'zh-CN',
    });

    if (availableAgents.length === 0) {
      return; // 没有可用客服
    }

    // 2. 选择最合适的客服
    const selectedAgent = this.selectBestAgent(availableAgents, session);

    // 3. 分配客服
    await this.assignAgentToSession(session.id, selectedAgent.id);
  }

  private async generateBotReply(
    session: ChatSession,
    userMessage: ChatMessage
  ): Promise<void> {
    try {
      // 1. 分析用户意图
      const intent = await this.botService.analyzeIntent(userMessage.content);

      // 2. 搜索知识库
      const knowledgeResults = await this.knowledgeService.search(
        userMessage.content.text || '',
        {
          category: 'faq',
          limit: 5,
        }
      );

      // 3. 生成回复
      const botReply = await this.botService.generateReply({
        intent,
        context: this.getChatContext(session),
        knowledgeResults,
        customerInfo: session.customerInfo,
      });

      // 4. 添加机器人回复
      await this.addMessage(session.id, {
        sender: { type: 'bot', id: 'bot-1' },
        content: { type: 'text', text: botReply.text },
        type: 'bot_reply',
      });

      // 5. 如果机器人无法解决，建议转人工
      if (botReply.confidence < 0.7 && !session.assignedAgent) {
        await this.suggestHumanTransfer(session);
      }
    } catch (error) {
      console.error('Bot reply generation failed:', error);
      await this.addMessage(session.id, {
        sender: { type: 'bot', id: 'bot-1' },
        content: {
          type: 'text',
          text: '抱歉，我暂时无法理解您的问题。让我为您转接人工客服。',
        },
        type: 'bot_error',
      });
    }
  }

  async createTicketFromChat(
    session: ChatSession,
    message: ChatMessage
  ): Promise<Ticket> {
    // 1. 分析是否需要创建工单
    const shouldCreate = await this.analyzeTicketNecessity(session, message);
    if (!shouldCreate.create) {
      throw new Error('Ticket creation not necessary');
    }

    // 2. 创建工单
    const ticket: Omit<Ticket, 'id' | 'createdAt' | 'updatedAt'> = {
      title: shouldCreate.title || '客户咨询',
      description: this.generateTicketDescription(session, message),
      category: shouldCreate.category || 'general',
      priority: shouldCreate.priority || 'medium',
      status: 'open',
      customerId: session.customerId,
      messages: [
        {
          id: generateId(),
          ticketId: '', // 将在保存后设置
          sender: message.sender,
          content: message.content,
          timestamp: message.timestamp,
          isInternal: false,
        },
      ],
      attachments: [],
      assignedAgent: session.assignedAgent?.id,
    };

    // 3. 保存工单
    const savedTicket = await this.ticketManager.createTicket(ticket);

    // 4. 更新会话关联
    await this.chatManager.updateSession(session.id, {
      linkedTicketId: savedTicket.id,
    });

    // 5. 通知相关客服
    if (savedTicket.assignedAgent) {
      await this.notificationService.notifyAgent(savedTicket.assignedAgent, {
        type: 'ticket_assigned',
        ticketId: savedTicket.id,
        priority: savedTicket.priority,
      });
    }

    return savedTicket;
  }

  private selectBestAgent(
    agents: AgentInfo[],
    session: ChatSession
  ): AgentInfo {
    // 实现客服选择算法
    const scoredAgents = agents.map(agent => ({
      agent,
      score: this.calculateAgentScore(agent, session),
    }));

    scoredAgents.sort((a, b) => b.score - a.score);
    return scoredAgents[0].agent;
  }

  private calculateAgentScore(agent: AgentInfo, session: ChatSession): number {
    let score = 0;

    // 基于工作负载
    const loadScore = (1 - agent.currentChats / agent.maxChats) * 40;
    score += loadScore;

    // 基于技能匹配
    const skillScore = this.calculateSkillMatch(agent.skills, session);
    score += skillScore * 30;

    // 基于历史评分
    const ratingScore = agent.rating.overall * 20;
    score += ratingScore;

    // 基于响应速度
    const speedScore =
      agent.rating.averageResponseTime > 0
        ? Math.max(0, 10 - agent.rating.averageResponseTime / 60) * 10
        : 10;
    score += speedScore;

    return score;
  }
}

// src/customer-service/BotService.ts
export interface BotIntent {
  name: string;
  confidence: number;
  entities: BotEntity[];
  parameters: Record<string, any>;
}

export interface BotEntity {
  name: string;
  value: string;
  type: string;
  confidence: number;
}

export interface BotReply {
  text: string;
  confidence: number;
  suggestions?: string[];
  actions?: BotAction[];
  knowledge?: KnowledgeReference[];
  escalateToHuman: boolean;
}

export interface BotAction {
  type: 'link' | 'button' | 'form' | 'file';
  label: string;
  value: string;
  metadata?: Record<string, any>;
}

export class BotService {
  private nluService: NLUService;
  private dialogManager: DialogManager;
  private knowledgeIntegrator: KnowledgeIntegrator;
  private escalationService: EscalationService;

  async analyzeIntent(content: MessageContent): Promise<BotIntent> {
    const text = this.extractTextFromContent(content);

    // 1. 文本预处理
    const preprocessedText = await this.preprocessText(text);

    // 2. 意图识别
    const intents = await this.nluService.detectIntent(preprocessedText);

    // 3. 实体提取
    const entities = await this.nluService.extractEntities(preprocessedText);

    // 4. 参数解析
    const parameters = await this.parseParameters(intents, entities);

    return {
      name: intents[0]?.name || 'unknown',
      confidence: intents[0]?.confidence || 0,
      entities,
      parameters,
    };
  }

  async generateReply(context: BotReplyContext): Promise<BotReply> {
    try {
      // 1. 检查是否在对话流中
      if (context.dialogState && context.dialogState.active) {
        return await this.continueDialog(context);
      }

      // 2. 基于意图生成回复
      switch (context.intent.name) {
        case 'greeting':
          return this.generateGreetingReply(context);
        case 'product_inquiry':
          return await this.generateProductReply(context);
        case 'technical_support':
          return await this.generateSupportReply(context);
        case 'complaint':
          return await this.generateComplaintReply(context);
        case 'account_issue':
          return await this.generateAccountReply(context);
        case 'faq_question':
          return await this.generateFAQReply(context);
        default:
          return await this.generateFallbackReply(context);
      }
    } catch (error) {
      console.error('Bot reply generation failed:', error);
      return this.generateErrorReply();
    }
  }

  private async generateProductReply(
    context: BotReplyContext
  ): Promise<BotReply> {
    const productName = context.intent.entities.find(
      e => e.type === 'product'
    )?.value;

    if (!productName) {
      return {
        text: '请问您想了解哪个产品的信息呢？我可以为您介绍我们的主要产品。',
        confidence: 0.8,
        suggestions: ['产品A', '产品B', '产品C'],
        escalateToHuman: false,
      };
    }

    // 搜索产品信息
    const productInfo = await this.searchProductInfo(productName);

    if (productInfo) {
      const reply: BotReply = {
        text: productInfo.description,
        confidence: 0.9,
        suggestions: ['了解更多', '价格咨询', '技术支持'],
        actions: [
          {
            type: 'link',
            label: '查看产品详情',
            value: productInfo.url,
          },
        ],
        escalateToHuman: false,
      };

      // 如果有相关知识，添加引用
      if (productInfo.knowledgeBase) {
        reply.knowledge = productInfo.knowledgeBase;
      }

      return reply;
    }

    return {
      text: `抱歉，我没有找到关于"${productName}"的信息。您可以尝试搜索其他产品，或者让我为您转接人工客服。`,
      confidence: 0.6,
      suggestions: ['搜索其他产品', '转人工客服'],
      escalateToHuman: true,
    };
  }

  private async generateSupportReply(
    context: BotReplyContext
  ): Promise<BotReply> {
    const issueType = context.intent.entities.find(
      e => e.type === 'issue_type'
    )?.value;

    if (!issueType) {
      return {
        text: '请描述您遇到的具体问题，这样我可以更好地为您提供帮助。',
        confidence: 0.7,
        suggestions: ['登录问题', '功能异常', '性能问题', '账号问题'],
        escalateToHuman: false,
      };
    }

    // 搜索解决方案
    const solutions = await this.knowledgeIntegrator.searchSolutions(
      issueType,
      {
        category: 'troubleshooting',
        limit: 3,
      }
    );

    if (solutions.length > 0) {
      const bestSolution = solutions[0];
      return {
        text: bestSolution.summary,
        confidence: 0.85,
        suggestions: ['问题解决了', '需要更多帮助', '其他问题'],
        actions: [
          {
            type: 'link',
            label: '查看详细解决方案',
            value: bestSolution.url,
          },
        ],
        knowledge: [bestSolution],
        escalateToHuman: false,
      };
    }

    // 如果没有找到解决方案，建议创建工单
    return {
      text: '我理解您遇到的问题。为了更好地帮助您，我建议创建一个技术支持工单，我们的专业技术人员会尽快为您处理。',
      confidence: 0.8,
      suggestions: ['创建工单', '描述更多细节', '转人工客服'],
      actions: [
        {
          type: 'button',
          label: '创建技术支持工单',
          value: 'create_support_ticket',
        },
      ],
      escalateToHuman: true,
    };
  }

  private async continueDialog(context: BotReplyContext): Promise<BotReply> {
    const dialogState = context.dialogState!;

    // 根据当前对话状态生成回复
    switch (dialogState.currentStep) {
      case 'collecting_info':
        return await this.collectMoreInfo(context);
      case 'verifying_issue':
        return await this.verifyIssue(context);
      case 'providing_solution':
        return await this.provideSolution(context);
      case 'confirming_resolution':
        return await this.confirmResolution(context);
      default:
        return await this.startNewDialog(context);
    }
  }

  private async collectMoreInfo(context: BotReplyContext): Promise<BotReply> {
    const requiredInfo = this.getRequiredInfoForIntent(context.intent.name);
    const missingInfo = requiredInfo.filter(info => !context.parameters[info]);

    if (missingInfo.length === 0) {
      // 信息收集完成，进入下一步
      await this.updateDialogState(context, 'verifying_issue');
      return await this.verifyIssue(context);
    }

    const nextInfo = missingInfo[0];
    const questions = {
      product: '请问您遇到问题的产品是哪个？',
      issue_type: '请问您遇到的是哪类问题？',
      error_message: '请问您看到什么错误信息？',
      steps: '请问您是如何遇到这个问题的？',
    };

    return {
      text: questions[nextInfo] || '请提供更多相关信息',
      confidence: 0.9,
      escalateToHuman: false,
    };
  }

  async handleFeedback(
    sessionId: string,
    feedback: BotFeedback
  ): Promise<void> {
    // 1. 保存反馈
    await this.saveBotFeedback(sessionId, feedback);

    // 2. 分析反馈
    await this.analyzeFeedback(feedback);

    // 3. 如果是负面反馈，考虑转人工
    if (feedback.rating <= 2) {
      await this.escalationService.evaluateEscalation(sessionId, {
        reason: 'negative_feedback',
        feedback,
        threshold: 0.7,
      });
    }

    // 4. 更新机器人性能
    await this.updateBotPerformance(feedback);
  }

  private extractTextFromContent(content: MessageContent): string {
    switch (content.type) {
      case 'text':
        return content.text || '';
      case 'image':
        return content.alt || ''; // 对于图片，使用alt文本
      default:
        return '';
    }
  }

  private async preprocessText(text: string): Promise<string> {
    // 文本清理和标准化
    return text
      .trim()
      .toLowerCase()
      .replace(/[^\w\s\u4e00-\u9fff]/g, ' ') // 保留中文字符
      .replace(/\s+/g, ' ')
      .trim();
  }
}

// src/customer-service/TicketManager.ts
export interface TicketWorkflow {
  id: string;
  name: string;
  description: string;
  steps: WorkflowStep[];
  conditions: WorkflowCondition[];
  autoActions: AutoAction[];
}

export interface WorkflowStep {
  id: string;
  name: string;
  type: StepType;
  assignee?: string;
  assigneeType: 'agent' | 'department' | 'auto';
  estimatedTime: number;
  requiredActions: string[];
  conditions: StepCondition[];
}

export class TicketManager {
  private ticketRepository: TicketRepository;
  private workflowEngine: WorkflowEngine;
  private assignmentEngine: AssignmentEngine;
  private notificationService: NotificationService;
  private slaService: SLAService;

  async createTicket(
    ticketData: Omit<Ticket, 'id' | 'createdAt' | 'updatedAt'>
  ): Promise<Ticket> {
    // 1. 验证工单数据
    const validatedTicket = await this.validateTicketData(ticketData);

    // 2. 创建工单
    const ticket: Ticket = {
      ...validatedTicket,
      id: generateId(),
      createdAt: new Date(),
      updatedAt: new Date(),
    };

    // 3. 保存工单
    const savedTicket = await this.ticketRepository.save(ticket);

    // 4. 启动工作流
    await this.workflowEngine.startWorkflow(savedTicket.id);

    // 5. 自动分配（如果配置了自动分配）
    if (!savedTicket.assignedAgent) {
      await this.attemptAutoAssignment(savedTicket);
    }

    // 6. 设置SLA
    await this.slaService.setSLA(savedTicket);

    // 7. 发送通知
    await this.sendTicketNotifications(savedTicket);

    return savedTicket;
  }

  async updateTicket(
    ticketId: string,
    updates: Partial<Ticket>,
    updatedBy: string
  ): Promise<Ticket> {
    // 1. 获取现有工单
    const existingTicket = await this.ticketRepository.findById(ticketId);
    if (!existingTicket) {
      throw new Error(`Ticket ${ticketId} not found`);
    }

    // 2. 验证更新权限
    await this.validateUpdatePermission(existingTicket, updatedBy);

    // 3. 应用更新
    const updatedTicket: Ticket = {
      ...existingTicket,
      ...updates,
      updatedAt: new Date(),
    };

    // 4. 保存更新
    const savedTicket = await this.ticketRepository.save(updatedTicket);

    // 5. 记录变更历史
    await this.recordTicketHistory(
      ticketId,
      existingTicket,
      updates,
      updatedBy
    );

    // 6. 触发工作流
    await this.workflowEngine.processUpdate(savedTicket, updates);

    // 7. 检查SLA状态
    await this.slaService.checkSLAStatus(savedTicket);

    // 8. 发送通知
    await this.sendUpdateNotifications(savedTicket, updates, updatedBy);

    return savedTicket;
  }

  async addMessage(
    ticketId: string,
    messageData: Omit<TicketMessage, 'id' | 'ticketId' | 'timestamp'>,
    senderId: string
  ): Promise<TicketMessage> {
    // 1. 获取工单信息
    const ticket = await this.ticketRepository.findById(ticketId);
    if (!ticket) {
      throw new Error(`Ticket ${ticketId} not found`);
    }

    // 2. 创建消息
    const message: TicketMessage = {
      id: generateId(),
      ticketId,
      sender: messageData.sender,
      content: messageData.content,
      timestamp: new Date(),
      isInternal: messageData.isInternal || false,
    };

    // 3. 保存消息
    const savedMessage = await this.ticketRepository.saveMessage(message);

    // 4. 更新工单
    await this.ticketRepository.save({
      ...ticket,
      updatedAt: new Date(),
    });

    // 5. 触发工作流
    await this.workflowEngine.processMessage(ticket, savedMessage);

    // 6. 发送通知
    await this.sendMessageNotifications(ticket, savedMessage);

    return savedMessage;
  }

  private async attemptAutoAssignment(ticket: Ticket): Promise<void> {
    // 1. 获取自动分配规则
    const assignmentRules = await this.getAssignmentRules(ticket);

    // 2. 查找合适的客服
    const candidateAgents = await this.assignmentEngine.findCandidates(
      ticket,
      assignmentRules
    );

    if (candidateAgents.length === 0) {
      // 没有合适的客服，保持未分配状态
      await this.notificationService.notifyDepartment({
        type: 'unassigned_ticket',
        ticketId: ticket.id,
        category: ticket.category,
        priority: ticket.priority,
      });
      return;
    }

    // 3. 选择最佳客服
    const selectedAgent = this.assignmentEngine.selectBestAgent(
      candidateAgents,
      ticket
    );

    // 4. 分配工单
    await this.assignTicket(ticket.id, selectedAgent.id, 'auto_assignment');
  }

  async assignTicket(
    ticketId: string,
    agentId: string,
    reason: string
  ): Promise<Ticket> {
    // 1. 获取工单和客服信息
    const [ticket, agent] = await Promise.all([
      this.ticketRepository.findById(ticketId),
      this.agentService.getAgent(agentId),
    ]);

    if (!ticket) {
      throw new Error(`Ticket ${ticketId} not found`);
    }

    if (!agent) {
      throw new Error(`Agent ${agentId} not found`);
    }

    // 2. 检查分配权限和条件
    await this.validateAssignment(ticket, agent, reason);

    // 3. 更新工单分配
    const updatedTicket = await this.ticketRepository.save({
      ...ticket,
      assignedAgent: agentId,
      updatedAt: new Date(),
    });

    // 4. 更新客服工作负载
    await this.agentService.updateWorkload(agentId, 1);

    // 5. 记录分配历史
    await this.recordAssignment(ticketId, agentId, reason);

    // 6. 发送通知
    await this.notificationService.notifyAgent(agentId, {
      type: 'ticket_assigned',
      ticketId: ticket.id,
      title: ticket.title,
      priority: ticket.priority,
      reason,
    });

    // 7. 启动客服工作流
    await this.workflowEngine.startAgentWorkflow(updatedTicket, agent);

    return updatedTicket;
  }

  async escalateTicket(
    ticketId: string,
    escalationData: EscalationData
  ): Promise<Ticket> {
    // 1. 获取工单信息
    const ticket = await this.ticketRepository.findById(ticketId);
    if (!ticket) {
      throw new Error(`Ticket ${ticketId} not found`);
    }

    // 2. 验证升级条件
    await this.validateEscalation(ticket, escalationData);

    // 3. 更新工单优先级和状态
    const updatedTicket = await this.ticketRepository.save({
      ...ticket,
      priority: escalationData.newPriority || ticket.priority,
      status: escalationData.newStatus || ticket.status,
      updatedAt: new Date(),
    });

    // 4. 处理重新分配
    if (escalationData.reassignTo) {
      await this.assignTicket(
        ticketId,
        escalationData.reassignTo,
        escalationData.reason
      );
    }

    // 5. 记录升级历史
    await this.recordEscalation(ticketId, escalationData);

    // 6. 发送升级通知
    await this.sendEscalationNotifications(updatedTicket, escalationData);

    // 7. 更新SLA
    await this.slaService.updateSLAOnEscalation(updatedTicket, escalationData);

    return updatedTicket;
  }

  async resolveTicket(
    ticketId: string,
    resolution: TicketResolution,
    resolvedBy: string
  ): Promise<Ticket> {
    // 1. 获取工单信息
    const ticket = await this.ticketRepository.findById(ticketId);
    if (!ticket) {
      throw new Error(`Ticket ${ticketId} not found`);
    }

    // 2. 验证解决权限
    await this.validateResolutionPermission(ticket, resolvedBy);

    // 3. 更新工单状态
    const resolvedTicket = await this.ticketRepository.save({
      ...ticket,
      status: 'resolved',
      resolvedAt: new Date(),
      updatedAt: new Date(),
      resolution,
    });

    // 4. 更新客服工作负载
    if (ticket.assignedAgent) {
      await this.agentService.updateWorkload(ticket.assignedAgent, -1);
    }

    // 5. 记录解决历史
    await this.recordResolution(ticketId, resolution, resolvedBy);

    // 6. 发送解决通知
    await this.sendResolutionNotifications(resolvedTicket, resolution);

    // 7. 更新统计信息
    await this.updateResolutionStatistics(resolvedTicket);

    // 8. 触发满意度调查
    await this.triggerSatisfactionSurvey(resolvedTicket);

    return resolvedTicket;
  }

  private async sendTicketNotifications(ticket: Ticket): Promise<void> {
    // 通知相关部门
    await this.notificationService.notifyDepartment({
      type: 'new_ticket',
      ticketId: ticket.id,
      category: ticket.category,
      priority: ticket.priority,
    });

    // 通知指定的客服（如果有）
    if (ticket.assignedAgent) {
      await this.notificationService.notifyAgent(ticket.assignedAgent, {
        type: 'new_assigned_ticket',
        ticketId: ticket.id,
        title: ticket.title,
      });
    }
  }
}

      setCustomerInfo(info);

        const newSession = await chatService.initiateChat(
          customerInfo || createAnonymousCustomer(),
          channel
        );
        setSession(newSession);
        onSessionStart?.(newSession);
      } catch (error) {
        console.error('Failed to initiate chat:', error);
      }
    }
    setIsOpen(true);
  };

  const handleWidgetClose = () => {
    setIsOpen(false);
  };

  const handleSendMessage = async () => {
    if (!inputValue.trim() || !session) return;

    const messageText = inputValue.trim();
    setInputValue('');
    setIsTyping(true);

    try {
      const message = await chatService.addMessage(session.id, {
        sender: { type: 'customer', id: customerInfo?.id || 'anonymous' },
        content: { type: 'text', text: messageText },
        type: 'user_message',
      });

      setMessages(prev => [...prev, message]);
    } catch (error) {
      console.error('Failed to send message:', error);
    } finally {
      setIsTyping(false);
    }
  };

  const handleWebSocketMessage = (event: MessageEvent) => {
    const data = JSON.parse(event.data);

    switch (data.type) {
      case 'new_message':
        setMessages(prev => [...prev, data.message]);
        break;
      case 'agent_typing':
        setIsTyping(data.isTyping);
        break;
      case 'session_updated':
        setSession(data.session);
        break;
      case 'session_closed':
        setSession(null);
        onSessionEnd?.(data.session);
        break;
    }
  };

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  if (!isOpen) {
    return (
      <ChatBubble
        position={position}
        theme={theme}
        onClick={handleWidgetOpen}
        unreadCount={
          messages.filter(m => m.sender.type !== 'customer' && !m.read).length
        }
      />
    );
  }

  return (
    <div
      className={`chat-widget chat-widget--${position} chat-widget--${theme}`}
    >
      <ChatHeader

        onClose={handleWidgetClose}
        onEndSession={handleEndSession}

          <ChatWelcome onSuggestionClick={handleSuggestionClick} />
        ) : (

              key={message.id}
              message={message}
              isOwn={message.sender.type === 'customer'}
            />
          ))
        )}

      </div>

        value={inputValue}
        onChange={setInputValue}

        onAttach={handleFileAttach}
        disabled={!session || isTyping}
        placeholder={session?.assignedAgent ? '输入消息...' : '正在连接客服...'}
      />

        <QueuePosition estimatedWaitTime={session.metadata.estimatedWaitTime} />
      )}
    </div>
  );
};

  };

        prev.map(t => (t.id === ticketId ? updatedTicket : t))

        ticketId,
        resolution,

        setSelectedTicket(null);
