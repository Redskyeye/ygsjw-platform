```typescript
// FreshGraduateOptimizer.ts
export class FreshGraduateOptimizer {
  async optimizeResume(
    resumeData: ParsedResume,
    config: FreshGraduateConfig
  ): Promise<OptimizedResume> {
    // 1. 突出教育背景
    const optimizedEducation = await this.optimizeEducation(resumeData.education);

    // 2. 强调项目经验
    const optimizedProjects = await this.optimizeProjects(resumeData.projects);

    // 3. 优化技能展示
    const optimizedSkills = await this.optimizeSkills(resumeData.skills);

    // 4. 生成个人总结
    const summary = await this.generateSummary(resumeData, config.targetRole);

    // 5. 优化实习和兼职经历
    const optimizedExperience = await this.optimizeExperience(
      resumeData.experience,
      config.focusInternships
    );

    return {
      ...resumeData,
      education: optimizedEducation,
      projects: optimizedProjects,
      skills: optimizedSkills,
      experience: optimizedExperience,
      summary,
      optimization: {
        mode: 'fresh_graduate',
        appliedStrategies: ['education_focus', 'project_emphasis', 'skill_highlighting'],
        score: await this.calculateOptimizationScore(resumeData, config)
      }
    };
  }

  private async optimizeEducation(education: Education[]): Promise<EnhancedEducation[]> {
    return education.map(edu => ({
      ...edu,
      relevantCourses: await this.extractRelevantCourses(edu),
      achievements: await this.identifyAcademicAchievements(edu),
      gpaHighlight: this.highlightGPA(edu.gpa),
      leadership: await this.extractLeadershipRoles(edu)
    }));
  }

  private async optimizeProjects(projects: Project[]): Promise<EnhancedProject[]> {
    const sortedProjects = this.sortProjectsByRelevance(projects);

    return await Promise.all(
      sortedProjects.map(async (project, index) => ({
        ...project,
        description: await this.enhanceProjectDescription(project),
        technologies: await this.categorizeTechnologies(project.technologies),
        impact: await this.quantifyProjectImpact(project),
        learningOutcomes: await this.extractLearningOutcomes(project)
      }))
    );
  }

  private async generateSummary(
    resumeData: ParsedResume,
    targetRole?: string
  ): Promise<string> {
    const prompt = `
      作为应届生，根据以下信息生成专业的个人总结：
      教育背景：${JSON.stringify(resumeData.education)}
      技能：${JSON.stringify(resumeData.skills)}
      项目经验：${JSON.stringify(resumeData.projects)}
      目标职位：${targetRole || '技术岗位'}

      要求：
      1. 突出学习能力和潜力
      2. 强调相关技能和项目经验
      3. 展现积极的工作态度
      4. 保持在150字以内
    `;

    return await this.aiService.generateText(prompt);
  }
}
```

```typescript
// EnglishResumeGenerator.ts
export class EnglishResumeGenerator {
  async generateEnglishResume(
    chineseResume: ParsedResume,
    config: EnglishConfig
  ): Promise<EnglishResume> {
    // 1. 翻译核心内容
    const translatedData = await this.translateResume(chineseResume, config.translationStyle);

    // 2. 适配英文表达习惯
    const adaptedData = await this.adaptToEnglishStyle(translatedData);

    // 3. 优化专业术语
    const optimizedData = await this.optimizeTechnicalTerms(adaptedData);

    // 4. 调整格式和结构
    const formattedResume = await this.formatForEnglishStandards(optimizedData);

    // 5. 语法和表达优化
    const polishedResume = await this.polishEnglishExpression(formattedResume);

    return polishedResume;
  }

  private async translateResume(
    resume: ParsedResume,
    style: 'formal' | 'professional' | 'casual'
  ): Promise<TranslatedResume> {
    const translationPrompts = {
      formal: '使用正式、专业的商务英语进行翻译',
      professional: '使用行业标准的专业术语进行翻译',
      casual: '使用自然、流畅的日常英语进行翻译'
    };

    const translated = {
      personalInfo: await this.translatePersonalInfo(resume.personalInfo, style),
      experience: await this.translateExperience(resume.experience, style),
      education: await this.translateEducation(resume.education, style),
      skills: await this.translateSkills(resume.skills, style),
      projects: await this.translateProjects(resume.projects, style)
    };

    return translated;
  }

  private async adaptToEnglishStyle(translated: TranslatedResume): Promise<AdaptedResume> {
    // 调整日期格式
    const adaptedExperience = translated.experience.map(exp => ({
      ...exp,
      duration: this.formatDateForEnglish(exp.duration),
      responsibilities: await this.convertToActionVerbs(exp.responsibilities),
      achievements: await this.quantifyAchievements(exp.achievements)
    }));

    // 优化技能描述
    const adaptedSkills = {
      technical: await this.categorizeSkillsByProficiency(translated.skills.technical),
      soft: await this.adaptSoftSkillsForWesternCulture(translated.skills.soft),
      languages: this.formatLanguageProficiency(translated.skills.languages)
    };

    return {
      ...translated,
      experience: adaptedExperience,
      skills: adaptedSkills
    };
  }

  private async convertToActionVerbs(responsibilities: string[]): Promise<string[]> {
    const actionVerbs = await this.loadActionVerbs();

    return responsibilities.map(resp => {
      // 将中文的责任描述转换为英文的动词开头格式
      return this.rephraseWithActionVerb(resp, actionVerbs);
    });
  }

  private async optimizeTechnicalTerms(data: AdaptedResume): Promise<TechnicalOptimizedResume> {
    const technicalTerms = await this.loadTechnicalTermsDictionary();

    return {
      ...data,
      skills: {
        ...data.skills,
        technical: data.skills.technical.map(skill =>
          this.standardizeTechnicalTerm(skill, technicalTerms)
        )
      },
      projects: data.projects.map(project => ({
        ...project,
        technologies: project.technologies.map(tech =>
          this.standardizeTechnicalTerm(tech, technicalTerms)
        )
      }))
    };
  }
}
```

```typescript
// ContentBeautifier.ts
export class ContentBeautifier {
  async beautifyContent(
    resumeData: ParsedResume,
    beautifyConfig: BeautifyConfig
  ): Promise<BeautifiedResume> {
    // 1. 语言优化
    const optimizedLanguage = await this.optimizeLanguage(resumeData, beautifyConfig);

    // 2. 结构优化
    const optimizedStructure = await this.optimizeStructure(optimizedLanguage);

    // 3. 视觉元素增强
    const enhancedVisual = await this.enhanceVisualElements(optimizedStructure);

    // 4. 关键词优化
    const keywordOptimized = await this.optimizeKeywords(enhancedVisual, beautifyConfig);

    return keywordOptimized;
  }

  private async optimizeLanguage(
    data: ParsedResume,
    config: BeautifyConfig
  ): Promise<LanguageOptimizedResume> {
    const optimized = { ...data };

    // 优化工作描述
    optimized.experience = await Promise.all(
      data.experience.map(async exp => ({
        ...exp,
        responsibilities: await this.enhanceResponsibilities(exp.responsibilities, config.tone),
        achievements: await this.enhanceAchievements(exp.achievements, config.impactLevel)
      }))
    );

    // 优化项目描述
    optimized.projects = await Promise.all(
      data.projects.map(async project => ({
        ...project,
        description: await this.enhanceProjectDescription(project.description, config.detailLevel),
        impact: await this.generateImpactStatement(project, config.metrics)
      }))
    );

    // 优化技能描述
    optimized.skills = await this.enhanceSkillsDescription(data.skills, config.emphasis);

    return optimized;
  }

  private async enhanceResponsibilities(
    responsibilities: string[],
    tone: 'professional' | 'dynamic' | 'conservative'
  ): Promise<string[]> {
    const tonePrompts = {
      professional: '使用专业、正式的语言描述工作职责',
      dynamic: '使用积极、有活力的语言描述工作职责',
      conservative: '使用稳重、保守的语言描述工作职责'
    };

    return await Promise.all(
      responsibilities.map(resp =>
        this.aiService.enhanceText(resp, tonePrompts[tone])
      )
    );
  }

  private async enhanceAchievements(
    achievements: string[],
    impactLevel: 'quantitative' | 'qualitative' | 'balanced'
  ): Promise<EnhancedAchievement[]> {
    return await Promise.all(
      achievements.map(async achievement => ({
        original: achievement,
        enhanced: await this.enhanceAchievementText(achievement, impactLevel),
        metrics: await this.extractOrGenerateMetrics(achievement),
        impact: await this.assessImpact(achievement)
      }))
    );
  }

  private async generateImpactStatement(
    project: Project,
    includeMetrics: boolean
  ): Promise<string> {
    const prompt = `
      为以下项目生成影响力陈述：
      项目名称：${project.name}
      项目描述：${project.description}
      使用技术：${project.technologies.join(', ')}
      项目时长：${project.duration}

      要求：
      1. 突出项目的实际影响和价值
      2. ${includeMetrics ? '包含具体的数据和指标' : '使用定性的影响力描述'}
      3. 保持在2-3句话内
      4. 使用强有力的动词开头
    `;

    return await this.aiService.generateText(prompt);
  }
}
```

```typescript
// KeywordOptimizer.ts
export class KeywordOptimizer {
  async optimizeKeywords(
    resumeData: ParsedResume,
    targetJob?: JobDescription,
    config: KeywordConfig = {}
  ): Promise<KeywordOptimizedResume> {
    // 1. 提取目标关键词
    const targetKeywords = targetJob
      ? await this.extractKeywordsFromJob(targetJob)
      : await this.getDefaultKeywords(resumeData.skills);

    // 2. 分析当前关键词密度
    const currentDensity = await this.analyzeKeywordDensity(resumeData);

    // 3. 优化关键词分布
    const optimizedData = await this.optimizeKeywordDistribution(
      resumeData,
      targetKeywords,
      currentDensity,
      config
    );

    // 4. 生成关键词摘要
    const keywordSummary = await this.generateKeywordSummary(optimizedData, targetKeywords);

    return {
      ...optimizedData,
      keywordSummary,
      optimization: {
        keywordsAdded: this.getAddedKeywords(resumeData, optimizedData),
        keywordsEnhanced: this.getEnhancedKeywords(resumeData, optimizedData),
        score: await this.calculateKeywordScore(optimizedData, targetKeywords)
      }
    };
  }

  private async extractKeywordsFromJob(job: JobDescription): Promise<KeywordSet> {
    const analysis = await this.aiService.analyzeText(job.description, {
      extractSkills: true,
      extractQualifications: true,
      extractResponsibilities: true,
      extractTechnologies: true
    });

    return {
      technical: analysis.skills.technical,
      soft: analysis.skills.soft,
      tools: analysis.technologies,
      qualifications: analysis.qualifications,
      industry: await this.extractIndustryTerms(job.description)
    };
  }

  private async optimizeKeywordDistribution(
    resume: ParsedResume,
    targetKeywords: KeywordSet,
    currentDensity: KeywordDensity,
    config: KeywordConfig
  ): Promise<KeywordOptimizedResume> {
    const optimized = { ...resume };

    // 优化技能部分
    optimized.skills = await this.optimizeSkillsKeywords(
      resume.skills,
      targetKeywords,
      config.skillEmphasis
    );

    // 优化工作经历
    optimized.experience = await Promise.all(
      resume.experience.map(exp =>
        this.optimizeExperienceKeywords(exp, targetKeywords, config.experienceDepth)
      )
    );

    // 优化项目描述
    optimized.projects = await Promise.all(
      resume.projects.map(project =>
        this.optimizeProjectKeywords(project, targetKeywords, config.projectDetail)
      )
    );

    // 生成关键词摘要
    optimized.keywordSummary = await this.generateKeywordSection(targetKeywords);

    return optimized;
  }

  private async optimizeSkillsKeywords(
    skills: SkillsData,
    targetKeywords: KeywordSet,
    emphasis: 'balanced' | 'technical' | 'comprehensive'
  ): Promise<EnhancedSkillsData> {
    // 技能优先级排序
    const prioritizedTechnical = this.prioritizeSkills(
      skills.technical,
      targetKeywords.technical
    );

    // 补充缺失的关键技能
    const enhancedTechnical = await this.enhanceSkillList(
      prioritizedTechnical,
      targetKeywords.technical
    );

    // 技能分类优化
    const categorizedSkills = await this.categorizeSkills(
      enhancedTechnical,
      targetKeywords.tools
    );

    return {
      technical: categorizedSkills,
      soft: await this.optimizeSoftSkills(skills.soft, targetKeywords.soft),
      languages: skills.languages,
      certifications: await this.optimizeCertifications(
        skills.certifications || [],
        targetKeywords.qualifications
      )
    };
  }
}
```

```typescript
// OptimizationEvaluator.ts
export class OptimizationEvaluator {
  async evaluateOptimization(
    originalResume: ParsedResume,
    optimizedResume: OptimizedResume,
    targetJob?: JobDescription
  ): Promise<EvaluationReport> {
    // 1. 内容质量评估
    const contentQuality = await this.assessContentQuality(optimizedResume);

    // 2. 关键词匹配度评估
    const keywordMatch = await this.assessKeywordMatching(optimizedResume, targetJob);

    // 3. 可读性评估
    const readability = await this.assessReadability(optimizedResume);

    // 4. 专业性评估
    const professionalism = await this.assessProfessionalism(optimizedResume);

    // 5. 对比分析
    const comparison = await this.compareWithOriginal(originalResume, optimizedResume);

    // 6. 综合评分
    const overallScore = this.calculateOverallScore({
      contentQuality,
      keywordMatch,
      readability,
      professionalism
    });

    return {
      overallScore,
      dimensions: {
        contentQuality,
        keywordMatch,
        readability,
        professionalism
      },
      comparison,
      recommendations: await this.generateRecommendations(optimizedResume, targetJob),
      improvementAreas: this.identifyImprovementAreas(comparison)
    };
  }

  private async assessContentQuality(resume: OptimizedResume): Promise<ContentQualityScore> {
    const metrics = {
      completeness: await this.assessCompleteness(resume),
      clarity: await this.assessClarity(resume),
      impact: await this.assessImpact(resume),
      relevance: await this.assessRelevance(resume),
      structure: await this.assessStructure(resume)
    };

    return {
      score: Object.values(metrics).reduce((sum, score) => sum + score, 0) / Object.keys(metrics).length,
      metrics,
      strengths: this.identifyStrengths(metrics),
      weaknesses: this.identifyWeaknesses(metrics)
    };
  }

  private async assessKeywordMatching(
    resume: OptimizedResume,
    targetJob?: JobDescription
  ): Promise<KeywordMatchScore> {
    if (!targetJob) {
      return { score: 0.5, details: '未提供目标职位' };
    }

    const jobKeywords = await this.extractJobKeywords(targetJob);
    const resumeKeywords = await this.extractResumeKeywords(resume);

    const matching = {
      exact: this.calculateExactMatch(jobKeywords, resumeKeywords),
      semantic: await this.calculateSemanticMatch(jobKeywords, resumeKeywords),
      contextual: await this.calculateContextualMatch(jobKeywords, resumeKeywords)
    };

    return {
      score: (matching.exact + matching.semantic + matching.contextual) / 3,
      details: matching,
      missingKeywords: this.findMissingKeywords(jobKeywords, resumeKeywords),
      overrepresentedKeywords: this.findOverrepresentedKeywords(jobKeywords, resumeKeywords)
    };
  }

  private async compareWithOriginal(
    original: ParsedResume,
    optimized: OptimizedResume
  ): Promise<ComparisonReport> {
    return {
      contentChanges: this.analyzeContentChanges(original, optimized),
      structuralChanges: this.analyzeStructuralChanges(original, optimized),
      linguisticImprovements: await this.analyzeLinguisticImprovements(original, optimized),
      keywordEnhancements: this.analyzeKeywordEnhancements(original, optimized),
      readabilityImprovement: await this.calculateReadabilityImprovement(original, optimized)
    };
  }
}
```

### 6. 优化历史管理
```typescript
// OptimizationHistory.ts
export class OptimizationHistory {
  async saveOptimization(
    originalId: string,
    optimizedResume: OptimizedResume,
    config: OptimizationConfig,
    evaluation: EvaluationReport
  ): Promise<OptimizationRecord> {
    const record: OptimizationRecord = {
      id: generateId(),
      originalId,
      timestamp: new Date(),
      config,
      result: optimizedResume,
      evaluation,
      version: await this.getNextVersion(originalId)
    };

    await this.persistRecord(record);
    return record;
  }

  async getOptimizationHistory(originalId: string): Promise<OptimizationRecord[]> {
    return await this.loadRecords(originalId);
  }

  async compareVersions(
    version1Id: string,
    version2Id: string
  ): Promise<VersionComparison> {
    const [version1, version2] = await Promise.all([
      this.loadRecord(version1Id),
      this.loadRecord(version2Id)
    ]);

    return {
      version1: {
        id: version1.id,
        timestamp: version1.timestamp,
        score: version1.evaluation.overallScore
      },
      version2: {
        id: version2.id,
        timestamp: version2.timestamp,
        score: version2.evaluation.overallScore
      },
      differences: await this.analyzeDifferences(version1.result, version2.result),
      improvements: this.identifyImprovements(version1, version2),
      regressions: this.identifyRegressions(version1, version2)
    };
  }

  async restoreVersion(originalId: string, versionId: string): Promise<OptimizedResume> {
    const record = await this.loadRecord(versionId);
    if (record.originalId !== originalId) {
      throw new Error('Version does not belong to the specified original resume');
    }
    return record.result;
  }
}
```
