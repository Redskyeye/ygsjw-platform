# 022 - 批量文件处理系统

## 概述
实现高效的批量文件处理系统，支持最多10个文件同时处理，单个文件最大50MB，提供拖拽上传、进度跟踪、结果预览等完整功能。

## 业务需求

### 核心功能
- **批量上传**：支持拖拽、点击选择多种上传方式
- **格式支持**：支持文档、图片、视频、音频等多种格式
- **进度跟踪**：实时显示处理进度和状态
- **结果预览**：提供处理结果预览和下载
- **错误处理**：完善的错误检测和重试机制

### 性能要求
- 最大同时处理文件数：10个
- 单个文件最大大小：50MB
- 文件上传速度：> 5MB/s
- 处理响应时间：< 3秒/文件
- 并发处理能力：5个文件同时处理

```typescript
export interface FileProcessor {
  processFile(file: File, options: ProcessingOptions): Promise<ProcessingResult>;
  processBatch(files: File[], options: BatchProcessingOptions): Promise<BatchResult>;
  getSupportedFormats(): string[];
  validateFile(file: File): ValidationResult;
}

export class UniversalFileProcessor implements FileProcessor {
  private processors: Map<string, SpecificFileProcessor>;

  async processFile(file: File, options: ProcessingOptions): Promise<ProcessingResult> {
    const processor = this.getProcessor(file.type);
    return processor.process(file, options);
  }

  async processBatch(files: File[], options: BatchProcessingOptions): Promise<BatchResult> {
    const chunks = this.chunkArray(files, options.concurrentLimit);
    const results: ProcessingResult[] = [];

    for (const chunk of chunks) {
      const chunkResults = await Promise.allSettled(
        chunk.map(file => this.processFile(file, options))
      );
      results.push(...this.extractResults(chunkResults));
    }

    return { results, summary: this.generateSummary(results) };
  }
}
```

```typescript
  // 分块上传
  uploadChunk(sessionId: string, fileId: string, chunk: Blob, chunkIndex: number): Promise<void>;

  completeUpload(sessionId: string, fileId: string): Promise<UploadResult>;

  // 获取处理状态

}

// 实现分块上传
export class ChunkedUploader {
  private chunkSize = 1024 * 1024; // 1MB chunks

  async uploadFile(file: File, onProgress?: (progress: number) => void): Promise<string> {
    const sessionId = await this.initiateUpload(file);
    const totalChunks = Math.ceil(file.size / this.chunkSize);

    for (let i = 0; i < totalChunks; i++) {
      const start = i * this.chunkSize;
      const end = Math.min(start + this.chunkSize, file.size);
      const chunk = file.slice(start, end);

      await this.uploadChunk(sessionId, chunk, i);

      if (onProgress) {
        onProgress((i + 1) / totalChunks * 100);
      }
    }

    return this.completeUpload(sessionId);
  }
}
```

### 处理队列系统
```typescript
// src/services/processingQueue.ts
export class ProcessingQueue {
  private queue: ProcessingTask[] = [];
  private processing = new Map<string, ProcessingTask>();
  private concurrency = 5;

  async addToQueue(task: ProcessingTask): Promise<void> {
    this.queue.push(task);
    this.processQueue();
  }

  private async processQueue(): Promise<void> {
    while (this.queue.length > 0 && this.processing.size < this.concurrency) {
      const task = this.queue.shift()!;
      this.processing.set(task.id, task);

      this.processTask(task)
        .finally(() => {
          this.processing.delete(task.id);
          this.processQueue();
        });
    }
  }

  private async processTask(task: ProcessingTask): Promise<void> {
    try {
      task.status = 'processing';
      task.startedAt = new Date();

      const result = await this.fileProcessor.processFile(task.file, task.options);

      task.status = 'completed';
      task.result = result;
      task.completedAt = new Date();

      this.notifyProgress(task);
    } catch (error) {
      task.status = 'error';
      task.error = error as Error;
      this.notifyError(task);
    }
  }
}
```

```typescript
export class ProcessingMetrics {
  trackProcessingTime(fileType: string, duration: number): void {
    this.metrics.histogram('file_processing_duration', duration, {
      file_type: fileType
    });
  }

  trackUploadSpeed(fileSize: number, duration: number): void {
    const speed = fileSize / duration * 1000; // bytes/second
    this.metrics.gauge('upload_speed', speed);
```
