# 028 - 搜索功能实现

## 概述
实现高性能的全文搜索功能，支持智能搜索建议、搜索结果高亮、搜索历史记录和搜索分析，为用户提供快速准确的信息检索体验。

## 业务需求

### 核心功能
- **全文搜索**：支持标题、内容、标签的全文检索
- **智能建议**：实时搜索建议和自动补全
- **结果高亮**：搜索关键词高亮显示
- **高级筛选**：多维度搜索筛选条件
- **搜索历史**：个人搜索历史记录
- **搜索分析**：搜索行为分析和优化

### 搜索类型
1. **快速搜索**：简单关键词搜索
2. **全文搜索**：完整内容搜索
3. **语义搜索**：AI智能语义理解
4. **分类搜索**：按分类筛选搜索
5. **标签搜索**：按标签搜索
6. **作者搜索**：按作者搜索
7. **日期搜索**：按时间范围搜索

### 搜索引擎架构
```typescript
// src/search/SearchEngine.ts
export interface SearchQuery {
  text?: string;
  filters: SearchFilters;
  pagination: Pagination;
  sortBy: SortOption;
  options: SearchOptions;
}

export interface SearchFilters {
  category?: string[];
  tags?: string[];
  author?: string[];
  status?: ArticleStatus[];
  dateRange?: DateRange;
  contentType?: ContentType[];
  wordCount?: RangeFilter;
  readingTime?: RangeFilter;
}

export interface SearchResult {
  items: SearchItem[];
  totalCount: number;
  currentPage: number;
  totalPages: number;
  facets: SearchFacet[];
  suggestions: SearchSuggestion[];
  corrections: SearchCorrection[];
  metadata: SearchMetadata;
}

export interface SearchItem {
  id: string;
  type: 'article' | 'document' | 'help' | 'faq';
  title: string;
  content: string;
  excerpt: string;
  highlights: Highlight[];
  score: number;
  metadata: ItemMetadata;
}

export class SearchEngine {
  private indexManager: IndexManager;
  private queryProcessor: QueryProcessor;
  private resultProcessor: ResultProcessor;
  private suggestionEngine: SuggestionEngine;
  private analyticsEngine: SearchAnalytics;

  async search(query: SearchQuery): Promise<SearchResult> {
    const startTime = Date.now();

    try {
      // 1. 预处理查询
      const processedQuery = await this.queryProcessor.process(query);

      // 2. 执行搜索
      const rawResults = await this.executeSearch(processedQuery);

      // 3. 处理结果
      const processedResults = await this.resultProcessor.process(
        rawResults,
        processedQuery
      );

      // 4. 生成建议
      const suggestions = await this.generateSuggestions(processedQuery);

      // 5. 生成分面
      const facets = await this.generateFacets(rawResults, processedQuery);

      // 6. 构建最终结果
      const result: SearchResult = {
        items: processedResults.items,
        totalCount: processedResults.totalCount,
        currentPage: query.pagination.page,
        totalPages: Math.ceil(processedResults.totalCount / query.pagination.limit),
        facets,
        suggestions,
        corrections: processedQuery.corrections || [],
        metadata: {
          queryTime: Date.now() - startTime,
          query: processedQuery.original,
          resultsCount: processedResults.items.length,
          indexUsed: this.indexManager.getCurrentIndex(),
          debug: query.options.debug ? this.generateDebugInfo(processedQuery) : undefined
        }
      };

      // 7. 记录搜索分析
      await this.analyticsEngine.recordSearch(query, result);

      return result;
    } catch (error) {
      console.error('Search failed:', error);
      throw new SearchError('Search execution failed', error);
    }
  }

  private async executeSearch(query: ProcessedQuery): Promise<RawSearchResult> {
    // 根据查询类型选择搜索策略
    switch (query.strategy) {
      case 'fulltext':
        return await this.executeFullTextSearch(query);
      case 'semantic':
        return await this.executeSemanticSearch(query);
      case 'hybrid':
        return await this.executeHybridSearch(query);
      default:
        return await this.executeBasicSearch(query);
    }
  }

  private async executeFullTextSearch(query: ProcessedQuery): Promise<RawSearchResult> {
    const searchRequest = {
      index: 'knowledge_base',
      body: {
        query: {
          bool: {
            must: this.buildMustClauses(query),
            filter: this.buildFilterClauses(query.filters),
            should: this.buildShouldClauses(query)
          }
        },
        highlight: {
          fields: {
            title: {
              pre_tags: ['<mark>'],
              post_tags: ['</mark>'],
              fragment_size: 150,
              number_of_fragments: 3
            },
            content: {
              pre_tags: ['<mark>'],
              post_tags: ['</mark>'],
              fragment_size: 150,
              number_of_fragments: 3
            }
          }
        },
        sort: this.buildSortCriteria(query.sortBy),
        from: (query.pagination.page - 1) * query.pagination.limit,
        size: query.pagination.limit,
        aggs: this.buildAggregations()
      }
    };

    const response = await this.indexManager.search(searchRequest);
    return this.parseSearchResponse(response);
  }

  private async executeSemanticSearch(query: ProcessedQuery): Promise<RawSearchResult> {
    // 1. 生成查询向量
    const queryVector = await this.generateQueryVector(query.text);

    // 2. 执行向量搜索
    const vectorSearchRequest = {
      index: 'knowledge_base_vectors',
      body: {
        query: {
          script_score: {
            query: this.buildBaseQuery(query),
            script: {
              source: 'cosineSimilarity(params.query_vector, \'embedding\') + 1.0',
              params: {
                query_vector: queryVector
              }
            }
          }
        },
        min_score: 0.3,
        size: query.pagination.limit * 2 // 获取更多结果进行重排
      }
    };

    const vectorResponse = await this.indexManager.search(vectorSearchRequest);

    // 3. 重排结果
    const rerankedResults = await this.rerankResults(
      vectorResponse.hits.hits,
      query
    );

    return {
      hits: {
        total: { value: rerankedResults.length },
        hits: rerankedResults.slice(0, query.pagination.limit)
      },
      aggregations: {}
    };
  }

  private async executeHybridSearch(query: ProcessedQuery): Promise<RawSearchResult> {
    // 1. 并行执行全文搜索和语义搜索
    const [fulltextResult, semanticResult] = await Promise.all([
      this.executeFullTextSearch(query),
      this.executeSemanticSearch(query)
    ]);

    // 2. 融合结果
    const fusedResults = this.fuseResults(
      fulltextResult.hits.hits,
      semanticResult.hits.hits,
      query.options.fusionWeight || 0.5
    );

    return {
      hits: {
        total: { value: fusedResults.length },
        hits: fusedResults.slice(0, query.pagination.limit)
      },
      aggregations: fulltextResult.aggregations
    };
  }

  private buildMustClauses(query: ProcessedQuery): any[] {
    const clauses: any[] = [];

    if (query.text) {
      clauses.push({
        multi_match: {
          query: query.text,
          fields: [
            'title^3',
            'content^2',
            'excerpt^1.5',
            'tags^2'
          ],
          type: 'best_fields',
          fuzziness: 'AUTO',
          prefix_length: 2
        }
      });
    }

    return clauses;
  }

  private buildFilterClauses(filters: SearchFilters): any[] {
    const clauseList: any[] = [];

    if (filters.category && filters.category.length > 0) {
      clauseList.push({
        terms: { 'category.id': filters.category }
      });
    }

    if (filters.tags && filters.tags.length > 0) {
      clauseList.push({
        terms: { 'tags.name': filters.tags }
      });
    }

    if (filters.author && filters.author.length > 0) {
      clauseList.push({
        terms: { 'author.id': filters.author }
      });
    }

    if (filters.status && filters.status.length > 0) {
      clauseList.push({
        terms: { status: filters.status }
      });
    }

    if (filters.dateRange) {
      clauseList.push({
        range: {
          updatedAt: {
            gte: filters.dateRange.from,
            lte: filters.dateRange.to
          }
        }
      });
    }

    return clauseList;
  }

  private fuseResults(
    fulltextResults: any[],
    semanticResults: any[],
    fusionWeight: number
  ): any[] {
    const resultScores = new Map<string, { ft: number; sem: number; hit: any }>();

    // 处理全文搜索结果
    fulltextResults.forEach(hit => {
      const id = hit._id;
      resultScores.set(id, {
        ft: hit._score,
        sem: 0,
        hit
      });
    });

    // 处理语义搜索结果
    semanticResults.forEach(hit => {
      const id = hit._id;
      const existing = resultScores.get(id);
      if (existing) {
        existing.sem = hit._score;
      } else {
        resultScores.set(id, {
          ft: 0,
          sem: hit._score,
          hit
        });
      }
    });

    // 融合分数并排序
    return Array.from(resultScores.values())
      .map(({ ft, sem, hit }) => ({
        ...hit,
        _score: ft * (1 - fusionWeight) + sem * fusionWeight
      }))
      .sort((a, b) => b._score - a._score);
  }
}
```

### 查询处理器
```typescript
// src/search/QueryProcessor.ts
export interface ProcessedQuery {
  original: string;
  cleaned: string;
  tokens: Token[];
  intent: SearchIntent;
  entities: Entity[];
  corrections: SearchCorrection[];
  strategy: SearchStrategy;
  filters: SearchFilters;
  options: SearchOptions;
}

export interface Token {
  text: string;
  type: TokenType;
  position: number;
  boost?: number;
}

export interface SearchIntent {
  type: 'informational' | 'navigational' | 'transactional' | 'commercial';
  confidence: number;
  entities: Entity[];
  concepts: Concept[];
}

export class QueryProcessor {
  private tokenizer: Tokenizer;
  private spellChecker: SpellChecker;
  private entityExtractor: EntityExtractor;
  private intentClassifier: IntentClassifier;

  async process(query: SearchQuery): Promise<ProcessedQuery> {
    // 1. 清理查询文本
    const cleanedText = this.cleanText(query.text || '');

    // 2. 分词
    const tokens = await this.tokenizer.tokenize(cleanedText);

    // 3. 拼写检查和纠正
    const corrections = await this.spellChecker.check(tokens);

    // 4. 实体提取
    const entities = await this.entityExtractor.extract(tokens);

    // 5. 意图识别
    const intent = await this.intentClassifier.classify(tokens, entities);

    // 6. 确定搜索策略
    const strategy = this.determineSearchStrategy(intent, query.options);

    // 7. 应用纠正
    const correctedTokens = this.applyCorrections(tokens, corrections);

    return {
      original: query.text || '',
      cleaned: cleanedText,
      tokens: correctedTokens,
      intent,
      entities,
      corrections,
      strategy,
      filters: query.filters,
      options: query.options
    };
  }

  private cleanText(text: string): string {
    return text
      .trim()
      .toLowerCase()
      .replace(/[^\w\s\u4e00-\u9fff]/g, ' ') // 保留中文字符
      .replace(/\s+/g, ' ')
      .trim();
  }

  private determineSearchStrategy(
    intent: SearchIntent,
    options: SearchOptions
  ): SearchStrategy {
    // 如果用户指定了策略，使用用户策略
    if (options.strategy) {
      return options.strategy;
    }

    // 根据意图确定策略
    if (intent.type === 'informational' && intent.entities.length > 0) {
      return 'semantic';
    }

    if (intent.type === 'navigational') {
      return 'fulltext';
    }

    // 默认使用混合搜索
    return 'hybrid';
  }

  private applyCorrections(tokens: Token[], corrections: SearchCorrection[]): Token[] {
    const correctionMap = new Map(
      corrections.map(c => [c.original, c.suggested])
    );

    return tokens.map(token => ({
      ...token,
      text: correctionMap.get(token.text) || token.text
    }));
  }
}
```

```typescript
// src/search/SuggestionEngine.ts
export interface SearchSuggestion {
  text: string;
  type: SuggestionType;
  score: number;
  source: SuggestionSource;
  metadata?: SuggestionMetadata;
}

export type SuggestionType = 'query' | 'completion' | 'correction' | 'entity';
export type SuggestionSource = 'history' | 'popular' | 'semantic' | 'spell';

export class SuggestionEngine {
  private historyStore: SearchHistoryStore;
  private popularQueries: PopularQueriesStore;
  private semanticIndex: SemanticIndex;
  private spellIndex: SpellIndex;

  async generateSuggestions(query: ProcessedQuery): Promise<SearchSuggestion[]> {
    const suggestions: SearchSuggestion[] = [];

    // 1. 查询补全建议
    const completions = await this.generateCompletions(query);
    suggestions.push(...completions);

    // 2. 拼写纠正建议
    if (query.corrections.length > 0) {
      const spellSuggestions = await this.generateSpellSuggestions(query);
      suggestions.push(...spellSuggestions);
    }

    // 3. 实体建议
    const entitySuggestions = await this.generateEntitySuggestions(query);
    suggestions.push(...entitySuggestions);

    // 4. 历史搜索建议
    const historySuggestions = await this.generateHistorySuggestions(query);
    suggestions.push(...historySuggestions);

    // 5. 热门查询建议
    const popularSuggestions = await this.generatePopularSuggestions(query);
    suggestions.push(...popularSuggestions);

    // 6. 排序和去重
    const processedSuggestions = this.processSuggestions(suggestions);

    return processedSuggestions.slice(0, 10); // 返回前10个建议
  }

  private async generateCompletions(query: ProcessedQuery): Promise<SearchSuggestion[]> {
    if (!query.text || query.text.length < 2) {
      return [];
    }

    const completionRequest = {
      index: 'search_completions',
      body: {
        suggest: {
          text: query.text,
          simple_phrase: {
            phrase: {
              field: 'suggest',
              size: 5,
              gram_size: 2,
              direct_generator: [{
                field: 'suggest',
                suggest_mode: 'missing'
              }]
            }
          }
        }
      }
    };

    const response = await this.semanticIndex.search(completionRequest);
    const suggestions = response.suggest.simple_phrase[0].options;

    return suggestions.map(option => ({
      text: option.text,
      type: 'completion' as const,
      score: option.score,
      source: 'semantic' as const,
      metadata: {
        payload: option.payload
      }
    }));
  }

  private async generateSpellSuggestions(query: ProcessedQuery): Promise<SearchSuggestion[]> {
    const suggestions: SearchSuggestion[] = [];

    for (const correction of query.corrections) {
      suggestions.push({
        text: correction.suggested,
        type: 'correction',
        score: correction.confidence,
        source: 'spell',
        metadata: {
          original: correction.original,
          rule: correction.rule
        }
      });
    }

    return suggestions;
  }

  private async generateEntitySuggestions(query: ProcessedQuery): Promise<SearchSuggestion[]> {
    const entitySuggestions: SearchSuggestion[] = [];

    for (const entity of query.entities) {
      // 基于实体类型生成建议
      switch (entity.type) {
        case 'category':
          const categorySuggestions = await this.searchCategories(entity.text);
          entitySuggestions.push(...categorySuggestions);
          break;
        case 'tag':
          const tagSuggestions = await this.searchTags(entity.text);
          entitySuggestions.push(...tagSuggestions);
          break;
        case 'author':
          const authorSuggestions = await this.searchAuthors(entity.text);
          entitySuggestions.push(...authorSuggestions);
          break;
      }
    }

    return entitySuggestions;
  }

  private async generateHistorySuggestions(query: ProcessedQuery): Promise<SearchSuggestion[]> {
    const userId = getCurrentUserId();
    const history = await this.historyStore.getUserHistory(userId, 50);

    const matchingHistory = history
      .filter(item => item.query.includes(query.text))
      .map(item => ({
        text: item.query,
        type: 'query' as const,
        score: item.frequency * 0.8, // 历史频率权重
        source: 'history' as const,
        metadata: {
          lastSearched: item.lastSearched,
          frequency: item.frequency
        }
      }));

    return matchingHistory;
  }

  private async generatePopularSuggestions(query: ProcessedQuery): Promise<SearchSuggestion[]> {
    const popularQueries = await this.popularQueries.getPopular(100);

    const matchingPopular = popularQueries
      .filter(item => item.query.includes(query.text))
      .map(item => ({
        text: item.query,
        type: 'query' as const,
        score: item.popularity * 0.9, // 流行度权重
        source: 'popular' as const,
        metadata: {
          popularity: item.popularity,
          trend: item.trend
        }
      }));

    return matchingPopular;
  }

  private processSuggestions(suggestions: SearchSuggestion[]): SearchSuggestion[] {
    // 去重
    const uniqueSuggestions = new Map<string, SearchSuggestion>();

    for (const suggestion of suggestions) {
      const existing = uniqueSuggestions.get(suggestion.text);
      if (!existing || suggestion.score > existing.score) {
        uniqueSuggestions.set(suggestion.text, suggestion);
      }
    }

    // 按分数排序
    return Array.from(uniqueSuggestions.values())
      .sort((a, b) => b.score - a.score);
  }
}
```

### 搜索分析引擎
```typescript
// src/search/SearchAnalytics.ts
export interface SearchAnalytics {
  totalSearches: number;
  uniqueQueries: number;
  averageResultCount: number;
  zeroResultQueries: number;
  popularQueries: QueryStatistics[];
  searchTrends: SearchTrend[];
  userBehavior: UserBehaviorMetrics;
  performanceMetrics: PerformanceMetrics;
}

export interface QueryStatistics {
  query: string;
  frequency: number;
  avgResultCount: number;
  clickThroughRate: number;
  avgPosition: number;
  trend: TrendDirection;
}

export interface SearchTrend {
  date: Date;
  searchCount: number;
  uniqueQueries: number;
  avgResultCount: number;
  avgResponseTime: number;
}

export class SearchAnalytics {
  private eventStore: SearchEventStore;
  private metricsCollector: MetricsCollector;

  async recordSearch(query: SearchQuery, result: SearchResult): Promise<void> {
    const event: SearchEvent = {
      id: generateId(),
      timestamp: new Date(),
      userId: getCurrentUserId(),
      sessionId: getSessionId(),
      query: query.text || '',
      filters: query.filters,
      resultCount: result.totalCount,
      responseTime: result.metadata.queryTime,
      clickedItems: [],
      position: 0
    };

    await this.eventStore.record(event);
    await this.updateRealTimeMetrics(event);
  }

  async recordClick(searchId: string, itemId: string, position: number): Promise<void> {
    const clickEvent: ClickEvent = {
      id: generateId(),
      timestamp: new Date(),
      searchId,
      itemId,
      position,
      userId: getCurrentUserId()
    };

    await this.eventStore.recordClick(clickEvent);
  }

  async generateAnalytics(dateRange: DateRange): Promise<SearchAnalytics> {
    const events = await this.eventStore.getEvents(dateRange);
    const clickEvents = await this.eventStore.getClickEvents(dateRange);

    return {
      totalSearches: events.length,
      uniqueQueries: new Set(events.map(e => e.query)).size,
      averageResultCount: this.calculateAverageResultCount(events),
      zeroResultQueries: events.filter(e => e.resultCount === 0).length,
      popularQueries: await this.calculatePopularQueries(events, clickEvents),
      searchTrends: await this.calculateSearchTrends(events, dateRange),
      userBehavior: await this.calculateUserBehavior(events, clickEvents),
      performanceMetrics: await this.calculatePerformanceMetrics(events)
    };
  }

  private async calculatePopularQueries(
    searchEvents: SearchEvent[],
    clickEvents: ClickEvent[]
  ): Promise<QueryStatistics[]> {
    const queryStats = new Map<string, QueryStatistics>();

    // 计算基础统计
    for (const event of searchEvents) {
      const existing = queryStats.get(event.query) || {
        query: event.query,
        frequency: 0,
        avgResultCount: 0,
        clickThroughRate: 0,
        avgPosition: 0,
        trend: 'stable' as TrendDirection
      };

      existing.frequency++;
      existing.avgResultCount = (existing.avgResultCount * (existing.frequency - 1) + event.resultCount) / existing.frequency;

      queryStats.set(event.query, existing);
    }

    // 计算点击率
    const clickStats = new Map<string, { clicks: number; totalPosition: number }>();
    for (const click of clickEvents) {
      const searchEvent = searchEvents.find(e => e.id === click.searchId);
      if (!searchEvent) continue;

      const existing = clickStats.get(searchEvent.query) || { clicks: 0, totalPosition: 0 };
      existing.clicks++;
      existing.totalPosition += click.position;
      clickStats.set(searchEvent.query, existing);
    }

    // 更新点击率
    for (const [query, stats] of queryStats) {
      const clickStat = clickStats.get(query);
      if (clickStat) {
        stats.clickThroughRate = clickStat.clicks / stats.frequency;
        stats.avgPosition = clickStat.totalPosition / clickStat.clicks;
      }
    }

    // 排序并返回
    return Array.from(queryStats.values())
      .sort((a, b) => b.frequency - a.frequency)
      .slice(0, 50);
  }

  private async calculateSearchTrends(
    events: SearchEvent[],
    dateRange: DateRange
  ): Promise<SearchTrend[]> {
    const dailyStats = new Map<string, SearchTrend>();

    // 按天分组
    for (const event of events) {
      const dayKey = event.timestamp.toISOString().split('T')[0];
      const existing = dailyStats.get(dayKey) || {
        date: new Date(dayKey),
        searchCount: 0,
        uniqueQueries: 0,
        avgResultCount: 0,
        avgResponseTime: 0
      };

      existing.searchCount++;
      existing.avgResultCount = (existing.avgResultCount * (existing.searchCount - 1) + event.resultCount) / existing.searchCount;
      existing.avgResponseTime = (existing.avgResponseTime * (existing.searchCount - 1) + event.responseTime) / existing.searchCount;

      dailyStats.set(dayKey, existing);
    }

    // 计算唯一查询数
    for (const [dayKey, trend] of dailyStats) {
      const dayEvents = events.filter(e => e.timestamp.toISOString().split('T')[0] === dayKey);
      trend.uniqueQueries = new Set(dayEvents.map(e => e.query)).size;
    }

    return Array.from(dailyStats.values())
      .sort((a, b) => a.date.getTime() - b.date.getTime());
  }
}
```

```tsx
  const searchEngine = useMemo(() => new SearchEngine(), []);
  const debouncedQuery = useDebounce(query, 300);

  useEffect(() => {
    if (debouncedQuery.length >= 2 && showSuggestions) {
      loadSuggestions(debouncedQuery);
    } else {
      setSuggestions([]);
    }
  }, [debouncedQuery, showSuggestions]);

  useEffect(() => {
    loadSearchHistory();
  }, []);

  const loadSuggestions = async (searchQuery: string) => {
    try {
      const suggestionsData = await searchEngine.getSuggestions(searchQuery);
      setSuggestions(suggestionsData);
    } catch (error) {
      console.error('Failed to load suggestions:', error);
    }
  };

  const loadSearchHistory = async () => {
    try {
      const history = await searchService.getSearchHistory();
      setSearchHistory(history);
    } catch (error) {
      console.error('Failed to load search history:', error);
    }
  };

  const handleSearch = async (searchQuery: string = query) => {
    if (!searchQuery.trim()) return;

    setIsSearching(true);
    setSuggestions([]);

    try {
      const searchQueryData: SearchQuery = {
        text: searchQuery,
        filters: selectedFilters,
        pagination: { page: 1, limit: 20 },
        sortBy: { field: 'relevance', order: 'desc' },
        options: {
          highlight: true,
          suggestions: true,
          debug: false
        }
      };

      const results = await searchEngine.search(searchQueryData);
      setSearchResults(results);

      // 记录搜索历史
      await searchService.saveSearch(searchQuery);
      setSearchHistory(prev => [searchQuery, ...prev.filter(h => h !== searchQuery)].slice(0, 10));

    } catch (error) {
      console.error('Search failed:', error);
    } finally {
      setIsSearching(false);
    }
  };

  const handleSuggestionSelect = (suggestion: SearchSuggestion) => {
    setQuery(suggestion.text);
    handleSearch(suggestion.text);
  };

  const handleFilterChange = (filters: SearchFilters) => {
    setSelectedFilters(filters);
    if (searchResults) {
      handleSearch();
    }
  };

  const handleResultClick = async (item: SearchItem) => {
    onResultSelect?.(item);
    // 记录点击
    await searchService.recordClick(item.id, searchResults!.items.indexOf(item));
  };

  return (
    <div className="search-interface">
      <div className="search-header">
        <div className="search-box-container">
          <SearchBox
            value={query}
            onChange={setQuery}
            onSearch={handleSearch}
            suggestions={suggestions}
            onSuggestionSelect={handleSuggestionSelect}
            history={searchHistory}
            isLoading={isSearching}
          />
        </div>

        {showFilters && (
          <div className="search-filters">
            <SearchFilters
              filters={selectedFilters}
              onChange={handleFilterChange}
              results={searchResults}
            />
          </div>
        )}
      </div>

      <div className="search-content">
        <div className="search-main">
          {searchResults && (
            <SearchResults
              results={searchResults}
              onResultClick={handleResultClick}
              onLoadMore={handleLoadMore}
            />
          )}

          {!searchResults && !isSearching && (
            <SearchEmptyState
              hasQuery={query.length > 0}
              onSuggestionClick={handleSuggestionSelect}
            />
          )}

          {isSearching && <SearchLoading />}
        </div>

        {searchResults && (
          <div className="search-sidebar">
            <SearchFacets
              facets={searchResults.facets}
              filters={selectedFilters}
              onChange={handleFilterChange}
            />

            <SearchSuggestions
              suggestions={searchResults.suggestions}
              onSuggestionClick={handleSuggestionSelect}
            />
          </div>
        )}
      </div>
    </div>
  );
};
```

```tsx
export const SearchBox: React.FC<SearchBoxProps> = ({
  value,
  onChange,
  onSearch,
  suggestions,
  onSuggestionSelect,
  history,
  isLoading
}) => {
  const [focused, setFocused] = useState(false);
  const [selectedSuggestionIndex, setSelectedSuggestionIndex] = useState(-1);
  const inputRef = useRef<HTMLInputElement>(null);

  const combinedSuggestions = useMemo(() => {
    const result: CombinedSuggestion[] = [];

    // 添加历史搜索
    if (history && history.length > 0) {
      result.push(
        { type: 'header', text: '搜索历史' },
        ...history.map(item => ({
          type: 'history',
          text: item,
          source: 'history' as const
        }))
      );
    }

    // 添加搜索建议
    if (suggestions && suggestions.length > 0) {
      if (result.length > 0) {
        result.push({ type: 'divider' });
      }
      result.push(
        { type: 'header', text: '搜索建议' },
        ...suggestions.map(suggestion => ({
          type: 'suggestion',
          text: suggestion.text,
          source: suggestion.source,
          metadata: suggestion.metadata
        }))
      );
    }

    return result;
  }, [history, suggestions]);

  const handleKeyDown = (e: React.KeyboardEvent) => {
    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault();
        setSelectedSuggestionIndex(prev =>
          Math.min(prev + 1, combinedSuggestions.length - 1)
        );
        break;

      case 'ArrowUp':
        e.preventDefault();
        setSelectedSuggestionIndex(prev => Math.max(prev - 1, -1));
        break;

      case 'Enter':
        e.preventDefault();
        if (selectedSuggestionIndex >= 0) {
          const selectedSuggestion = combinedSuggestions[selectedSuggestionIndex];
          if (selectedSuggestion.type === 'history' || selectedSuggestion.type === 'suggestion') {
            onSuggestionSelect(selectedSuggestion);
          }
        } else {
          onSearch();
        }
        break;

      case 'Escape':
        setSelectedSuggestionIndex(-1);
        setFocused(false);
        break;
    }
  };

  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    onChange(e.target.value);
    setSelectedSuggestionIndex(-1);
  };

  const handleSuggestionClick = (suggestion: CombinedSuggestion) => {
    if (suggestion.type === 'history' || suggestion.type === 'suggestion') {
      onSuggestionSelect(suggestion);
    }
  };

  return (
    <div className={`search-box ${focused ? 'focused' : ''}`}>
      <div className="search-input-container">
        <SearchIcon className="search-icon" />
        <input
          ref={inputRef}
          type="text"
          value={value}
          onChange={handleInputChange}
          onKeyDown={handleKeyDown}
          onFocus={() => setFocused(true)}
          onBlur={() => {
            setTimeout(() => setFocused(false), 200);
          }}
          placeholder="搜索文档、帮助、FAQ..."
          className="search-input"
          autoComplete="off"
        />

        {isLoading && (
          <LoadingSpinner className="search-loading" />
        )}

        {value && !isLoading && (
          <button
            className="search-clear"
            onClick={() => onChange('')}
            title="清除"
          >
            <ClearIcon />
          </button>
        )}
      </div>

      {focused && combinedSuggestions.length > 0 && (
        <div className="search-suggestions">
          {combinedSuggestions.map((suggestion, index) => (
            <div
              key={index}
              className={`suggestion-item ${
                suggestion.type === 'header' ? 'suggestion-header' : ''
              } ${
                suggestion.type === 'divider' ? 'suggestion-divider' : ''
              } ${
                index === selectedSuggestionIndex ? 'selected' : ''
              }`}
              onClick={() => handleSuggestionClick(suggestion)}
            >
              {suggestion.type === 'header' && (
                <span className="suggestion-header-text">{suggestion.text}</span>
              )}

              {suggestion.type === 'divider' && (
                <div className="suggestion-divider-line" />
              )}

              {(suggestion.type === 'history' || suggestion.type === 'suggestion') && (
                <div className="suggestion-content">
                  {suggestion.type === 'history' && (
                    <HistoryIcon className="suggestion-icon" />
                  )}
                  {suggestion.type === 'suggestion' && (
                    <SuggestionIcon className="suggestion-icon" />
                  )}
                  <span className="suggestion-text">{suggestion.text}</span>
                  {suggestion.type === 'suggestion' && suggestion.source === 'spell' && (
                    <span className="suggestion-spell">拼写纠正</span>
                  )}
                </div>
              )}
            </div>
          ))}
        </div>
      )}
    </div>
  );
};
```

```typescript
// src/search/__tests__/performance.test.ts
describe('Search Performance', () => {
  test('should return results within acceptable time', async () => {
    const engine = new SearchEngine();
    const query: SearchQuery = {
      text: 'performance test query',
      filters: {},
      pagination: { page: 1, limit: 20 },
      sortBy: { field: 'relevance', order: 'desc' },
      options: {}
    };

    const startTime = Date.now();
    const result = await engine.search(query);
    const endTime = Date.now();

    expect(endTime - startTime).toBeLessThan(1000); // 1秒内完成
    expect(result.metadata.queryTime).toBeLessThan(500);
  });

  test('should handle high concurrent searches', async () => {
    const engine = new SearchEngine();
    const concurrentSearches = 50;
    const promises = Array(concurrentSearches).fill(null).map(async (_, index) => {
      const query: SearchQuery = {
        text: `concurrent test ${index}`,
        filters: {},
        pagination: { page: 1, limit: 10 },
        sortBy: { field: 'relevance', order: 'desc' },
        options: {}
      };

      return engine.search(query);
    });

    const startTime = Date.now();
    const results = await Promise.all(promises);
    const endTime = Date.now();

    expect(results).toHaveLength(concurrentSearches);
    expect(endTime - startTime).toBeLessThan(5000); // 5秒内完成所有搜索
  });
});
```
