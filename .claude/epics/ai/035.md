```dockerfile
# 设置工作目录

# 复制源代码

# 生成Prisma客户端
RUN pnpm prisma generate

# 构建应用
RUN npm run build

# 生产阶段
FROM node:18-alpine AS runner

# 安装必要的系统包
RUN apk add --no-cache dumb-init

# 创建应用用户
RUN addgroup --system --gid 1001 nodejs && \
    adduser --system --uid 1001 nextjs

# 设置工作目录
WORKDIR /app

# 复制构建产物
COPY --from=builder --chown=nextjs:nodejs /app/.next/standalone ./
COPY --from=builder --chown=nextjs:nodejs /app/.next/static ./.next/static
COPY --from=builder --chown=nextjs:nodejs /app/public ./public
COPY --from=builder --chown=nextjs:nodejs /app/package.json ./package.json

# 安装生产依赖
RUN corepack enable pnpm && \
    pnpm install --frozen-lockfile --prod

# 创建非root用户可写的目录
RUN mkdir -p /app/uploads && \
    chown -R nextjs:nodejs /app/uploads

# 切换到非root用户
USER nextjs

# 暴露端口
EXPOSE 3000

# 环境变量
ENV NODE_ENV=production
ENV PORT=3000
ENV HOSTNAME="0.0.0.0"

# 健康检查
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
  CMD curl -f http://localhost:3000/api/health || exit 1

# 启动应用
ENTRYPOINT ["dumb-init", "--"]
CMD ["node", "server.js"]
```

#### 监控脚本
```javascript
// scripts/pm2-monitor.js
const pm2 = require('pm2');
const fs = require('fs');
const path = require('path');

class PM2Monitor {
    constructor() {
        this.logFile = path.join(__dirname, '../logs/pm2-monitor.log');
    }

    log(message) {
        const timestamp = new Date().toISOString();
        const logMessage = `[${timestamp}] ${message}\n`;
        console.log(logMessage.trim());
        fs.appendFileSync(this.logFile, logMessage);
    }

    async startMonitoring() {
        pm2.connect((err) => {
            if (err) {
                console.error('PM2连接失败:', err);
                process.exit(1);
            }

            this.log('📊 开始PM2监控');

            // 监控进程事件
            pm2.launchBus((err, bus) => {
                if (err) {
                    console.error('启动PM2事件总线失败:', err);
                    return;
                }

                // 进程重启事件
                bus.on('process:online', (proc) => {
                    this.log(`✅ 进程 ${proc.name} (PID: ${proc.pid}) 在线`);
                });

                bus.on('process:exit', (proc) => {
                    this.log(`❌ 进程 ${proc.name} (PID: ${proc.pid}) 退出`);
                });

                bus.on('process:restart', (proc) => {
                    this.log(`🔄 进程 ${proc.name} 重启`);
                });

                bus.on('log:out', (proc) => {
                    this.log(`📝 [${proc.name}] ${proc.data}`);
                });

                bus.on('log:err', (proc) => {
                    this.log(`🚨 [${proc.name}] ERROR: ${proc.data}`);
                });
            });

            // 定期检查进程状态
            setInterval(() => {
                this.checkProcessHealth();
            }, 30000); // 每30秒检查一次
        });
    }

    async checkProcessHealth() {
        pm2.list((err, processes) => {
            if (err) {
                this.log(`❌ 检查进程状态失败: ${err.message}`);
                return;
            }

            processes.forEach(proc => {
                if (proc.pm2_env.status !== 'online') {
                    this.log(`⚠️  进程 ${proc.name} 状态异常: ${proc.pm2_env.status}`);

                    // 尝试重启
                    if (proc.pm2_env.status === 'stopped') {
                        pm2.restart(proc.name, (restartErr) => {
                            if (restartErr) {
                                this.log(`❌ 重启进程 ${proc.name} 失败: ${restartErr.message}`);
                            } else {
                                this.log(`✅ 成功重启进程 ${proc.name}`);
                            }
                        });
                    }
                }

                // 检查内存使用
                const memoryMB = Math.round(proc.monit.memory / 1024 / 1024);
                if (memoryMB > 800) { // 超过800MB
                    this.log(`⚠️  进程 ${proc.name} 内存使用过高: ${memoryMB}MB`);
                }

                // 检查CPU使用
                const cpuPercent = Math.round(proc.monit.cpu);
                if (cpuPercent > 80) { // 超过80%
                    this.log(`⚠️  进程 ${proc.name} CPU使用过高: ${cpuPercent}%`);
                }
            });
        });
    }
}

// 启动监控

monitor.startMonitoring();
```

```bash
SECRETS_FILE=".env.production"
SECRETS_BACKUP_DIR="./backups/secrets"

# 创建备份目录
mkdir -p "$SECRETS_BACKUP_DIR"

# 生成随机密码
generate_password() {
    openssl rand -base64 32 | tr -d "=+/" | cut -c1-25
}

# 生成随机密钥
generate_secret() {
    openssl rand -base64 32
}

# 创建新的密钥文件
create_secrets() {
    echo "🔧 创建新的生产环境密钥文件..."

    # 备份现有密钥文件
    if [ -f "$SECRETS_FILE" ]; then
        cp "$SECRETS_FILE" "$SECRETS_BACKUP_DIR/env.production.$(date +%Y%m%d_%H%M%S).backup"
        echo "📁 现有密钥文件已备份"
    fi

    # 生成新密钥
    cat > "$SECRETS_FILE" << EOF
# 自动生成的生产环境密钥 - $(date)
# 请根据实际需求修改这些值

# 应用配置
NODE_ENV=production
PORT=3000
HOSTNAME=0.0.0.0

# 数据库配置
DATABASE_URL=postgresql://postgres:$(generate_password)@postgres:5432/epic_ai_prod
POSTGRES_DB=epic_ai_prod
POSTGRES_USER=postgres
POSTGRES_PASSWORD=$(generate_password)

# Redis配置
REDIS_URL=redis://redis:6379
REDIS_PASSWORD=$(generate_password)

# 认证配置
NEXTAUTH_URL=https://epic-ai.com
NEXTAUTH_SECRET=$(generate_secret)

# AI服务配置
OPENAI_API_KEY=sk-YOUR_OPENAI_API_KEY
OPENAI_ORG_ID=org-YOUR_OPENAI_ORG_ID

# 监控配置
GRAFANA_USER=admin
GRAFANA_PASSWORD=$(generate_password)

# 其他配置...
EOF

    chmod 600 "$SECRETS_FILE"
    echo "✅ 新密钥文件已创建: $SECRETS_FILE"
    echo "⚠️  请手动编辑文件并填入实际的API密钥"
}

# 验证密钥文件
validate_secrets() {
    echo "🔍 验证密钥文件..."

    if [ ! -f "$SECRETS_FILE" ]; then
        echo "❌ 密钥文件不存在: $SECRETS_FILE"
        exit 1
    fi

    # 检查必需的环境变量
    local required_vars=(
        "DATABASE_URL"
        "NEXTAUTH_SECRET"
        "OPENAI_API_KEY"
    )

    local missing_vars=()

    for var in "${required_vars[@]}"; do
        if ! grep -q "^$var=" "$SECRETS_FILE"; then
            missing_vars+=("$var")
        fi
    done

    if [ ${#missing_vars[@]} -gt 0 ]; then
        echo "❌ 缺少必需的环境变量:"
        for var in "${missing_vars[@]}"; do
            echo "   - $var"
        done
        exit 1
    fi

    echo "✅ 密钥文件验证通过"
}

# 主菜单
case "$1" in
    create)
        create_secrets
        ;;
    validate)
        validate_secrets
        ;;
    *)
        echo "用法: $0 {create|validate}"
        echo "  create   - 创建新的密钥文件"
        echo "  validate - 验证密钥文件"
        exit 1
        ;;
esac
```

```bash
#!/bin/bash
# scripts/deploy.sh
```

#### 恢复脚本
```bash
#!/bin/bash
# scripts/restore-db.sh

set -e

BACKUP_FILE=$1

if [ -z "$BACKUP_FILE" ]; then
    echo "❌ 请指定备份文件"
    echo "用法: $0 <backup_file>"
    exit 1
fi

if [ ! -f "$BACKUP_FILE" ]; then
    echo "❌ 备份文件不存在: $BACKUP_FILE"
    exit 1
fi

echo "🔄 开始数据库恢复..."

# 停止应用服务
echo "⏸️  停止应用服务..."
docker-compose stop app

# 恢复数据库
echo "📥 恢复数据库..."
if [[ $BACKUP_FILE == *.gz ]]; then
    gunzip -c "$BACKUP_FILE" | docker exec -i epic-ai-postgres psql -U postgres -d epic_ai_prod
else
    docker exec -i epic-ai-postgres psql -U postgres -d epic_ai_prod < "$BACKUP_FILE"
fi

# 重启应用服务
echo "▶️  重启应用服务..."
docker-compose start app

echo "✅ 数据库恢复完成！"
```
