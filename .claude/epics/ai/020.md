# 任务020: 进度追踪完善系统

## 📋 任务概述

完善面试辅导模块的进度追踪系统，提供从信息收集到PPT生成的全流程实时进度监控，包括详细的状态显示、时间预估、错误处理和用户通知等功能，确保用户清楚了解当前处理状态。

## 🎯 具体要求

### 20.1 多维度进度显示
- **整体进度**：显示整个面试辅导流程的总体进度
- **阶段进度**：按阶段显示进度（信息收集、解析处理、模板匹配、PPT生成）
- **步骤进度**：每个阶段的详细步骤进度
- **时间进度**：预计剩余时间和已用时间显示
- **质量进度**：数据处理质量评估进度

### 20.2 实时状态监控
- **处理状态**：待处理、处理中、已完成、失败、取消
- **子任务状态**：每个子任务的独立状态跟踪
- **资源状态**：系统资源使用情况（CPU、内存、存储）
- **队列状态**：任务队列长度和等待时间
- **错误状态**：错误类型、错误位置、错误恢复状态

### 20.3 智能时间预估
- **基础预估**：基于任务类型的标准时间预估
- **动态调整**：根据实际处理速度动态调整预估
- **历史参考**：基于历史数据的预估优化
- **负载考虑**：考虑系统负载的时间预估
- **异常处理**：异常情况下的时间重新预估

### 20.4 通知和提醒
- **进度通知**：关键节点的进度变化通知
- **完成通知**：任务完成时的通知推送
- **错误通知**：处理失败时的错误通知
- **超时提醒**：处理超时的提醒和警告
- **邮件通知**：重要状态的邮件通知选项

### 20.5 进度历史和分析
- **历史记录**：保存完整的处理历史记录
- **性能分析**：处理性能的分析和统计
- **趋势分析**：处理时间的变化趋势
- **异常分析**：异常情况的统计分析
- **用户行为分析**：用户操作行为分析

```typescript
// services/ProgressTracker.ts
interface ProgressStep {
  id: string;
  name: string;
  status: 'pending' | 'processing' | 'completed' | 'failed' | 'skipped';
  progress: number; // 0-100
  startTime?: Date;
  endTime?: Date;
  duration?: number; // 毫秒
  error?: string;
  metadata?: any;
}

interface ProgressPhase {
  id: string;
  name: string;
  steps: ProgressStep[];
  status: 'pending' | 'processing' | 'completed' | 'failed';
  progress: number;
  estimatedDuration?: number;
  actualDuration?: number;
}

interface OverallProgress {
  sessionId: string;
  phases: ProgressPhase[];
  currentPhase: string;
  currentStep: string;
  overallProgress: number;
  startTime: Date;
  estimatedEndTime?: Date;
  status: 'pending' | 'processing' | 'completed' | 'failed';
}

class ProgressTracker {
  private progressData: Map<string, OverallProgress> = new Map();
  private subscribers: Map<string, Set<ProgressSubscriber>> = new Map();

  createSession(sessionId: string, config: ProgressConfig): void {
    const session: OverallProgress = {
      sessionId,
      phases: this.initializePhases(config),
      currentPhase: config.phases[0],
      currentStep: '',
      overallProgress: 0,
      startTime: new Date(),
      status: 'pending'
    };

    this.progressData.set(sessionId, session);
    this.notifySubscribers(sessionId, session);
  }

  updateProgress(
    sessionId: string,
    phaseId: string,
    stepId: string,
    progress: number,
    status?: ProgressStep['status']
  ): void {
    const session = this.progressData.get(sessionId);
    if (!session) return;

    const phase = session.phases.find(p => p.id === phaseId);
    if (!phase) return;

    const step = phase.steps.find(s => s.id === stepId);
    if (!step) return;

    // 更新步骤进度
    step.progress = progress;
    if (status) step.status = status;

    // 更新阶段进度
    phase.progress = this.calculatePhaseProgress(phase);
    phase.status = this.determinePhaseStatus(phase);

    // 更新整体进度
    session.overallProgress = this.calculateOverallProgress(session);
    session.status = this.determineOverallStatus(session);
    session.currentPhase = phaseId;
    session.currentStep = stepId;

    // 更新时间预估
    this.updateTimeEstimate(session);

    // 通知订阅者
    this.notifySubscribers(sessionId, session);
  }

  subscribe(sessionId: string, callback: ProgressSubscriber): () => void {
    if (!this.subscribers.has(sessionId)) {
      this.subscribers.set(sessionId, new Set());
    }

    this.subscribers.get(sessionId)!.add(callback);

    // 返回取消订阅函数
    return () => {
      this.subscribers.get(sessionId)?.delete(callback);
    };
  }

  private calculatePhaseProgress(phase: ProgressPhase): number {
    const completedSteps = phase.steps.filter(s => s.status === 'completed').length;
    return (completedSteps / phase.steps.length) * 100;
  }

  private calculateOverallProgress(session: OverallProgress): number {
    const totalSteps = session.phases.reduce((sum, phase) => sum + phase.steps.length, 0);
    const completedSteps = session.phases.reduce(
      (sum, phase) => sum + phase.steps.filter(s => s.status === 'completed').length,
      0
    );
    return (completedSteps / totalSteps) * 100;
  }
}
```

### 20.7 智能时间预估
```typescript
// services/TimeEstimator.ts
interface TimeEstimate {
  estimatedTotalTime: number; // 毫秒
  estimatedRemainingTime: number;
  confidence: number; // 0-1
  factors: TimeFactor[];
}

interface TimeFactor {
  name: string;
  impact: number; // 时间影响系数
  value: any;
}

class TimeEstimator {
  private historicalData: HistoricalData[] = [];
  private systemMetrics: SystemMetrics;

  async estimateTime(
    sessionId: string,
    currentProgress: OverallProgress
  ): Promise<TimeEstimate> {
    // 基础时间预估
    const baseEstimate = this.getBaseEstimate(currentProgress);

    // 历史数据调整
    const historicalAdjustment = this.getHistoricalAdjustment(
      currentProgress,
      this.historicalData
    );

    // 系统负载调整
    const loadAdjustment = this.getLoadAdjustment(this.systemMetrics);

    // 异常情况调整
    const anomalyAdjustment = this.getAnomalyAdjustment(currentProgress);

    // 计算最终预估
    const totalTime = baseEstimate * historicalAdjustment * loadAdjustment * anomalyAdjustment;
    const remainingTime = totalTime * (1 - currentProgress.overallProgress / 100);

    return {
      estimatedTotalTime: totalTime,
      estimatedRemainingTime: remainingTime,
      confidence: this.calculateConfidence(baseEstimate, historicalAdjustment),
      factors: [
        { name: 'base', impact: 1, value: baseEstimate },
        { name: 'historical', impact: historicalAdjustment, value: historicalAdjustment },
        { name: 'load', impact: loadAdjustment, value: this.systemMetrics.cpuUsage },
        { name: 'anomaly', impact: anomalyAdjustment, value: anomalyAdjustment }
      ]
    };
  }

  private getBaseEstimate(progress: OverallProgress): number {
    let totalTime = 0;

    for (const phase of progress.phases) {
      for (const step of phase.steps) {
        const stepConfig = this.getStepConfig(step.id);
        totalTime += stepConfig.estimatedDuration;
      }
    }

    return totalTime;
  }

  private getHistoricalAdjustment(
    current: OverallProgress,
    historical: HistoricalData[]
  ): number {
    // 找到相似的历史记录
    const similarSessions = historical.filter(h => this.isSimilar(current, h));

    if (similarSessions.length === 0) return 1.0;

    // 计算平均调整系数
    const adjustments = similarSessions.map(h => h.actualDuration / h.estimatedDuration);
    return adjustments.reduce((sum, adj) => sum + adj, 0) / adjustments.length;
  }
}
```

```python
# services/progress_service.py
from flask_socketio import SocketIO, emit, join_room, leave_room
from services.progress_tracker import ProgressTracker
from services.time_estimator import TimeEstimator
from services.notification_service import NotificationService

socketio = SocketIO(cors_allowed_origins="*")

class ProgressService:
    def __init__(self):
        self.tracker = ProgressTracker()
        self.estimator = TimeEstimator()
        self.notifier = NotificationService()

    @socketio.on('subscribe_progress')
    def handle_subscribe(data):
        """订阅进度更新"""
        session_id = data.get('session_id')
        if session_id:
            join_room(f"progress_{session_id}")
            emit('subscribed', {'session_id': session_id})

    @socketio.on('unsubscribe_progress')
    def handle_unsubscribe(data):
        """取消订阅进度更新"""
        session_id = data.get('session_id')
        if session_id:
            leave_room(f"progress_{session_id}")

    async def update_progress(self, session_id: str, phase_id: str, step_id: str,
                            progress: int, status: str = None, error: str = None):
        """更新进度并通知客户端"""
        try:
            # 更新进度数据
            self.tracker.update_progress(session_id, phase_id, step_id, progress, status, error)

            # 获取最新进度
            current_progress = self.tracker.get_progress(session_id)

            # 计算时间预估
            time_estimate = await self.estimator.estimate_time(session_id, current_progress)

            # 发送WebSocket更新

                'session_id': session_id,
                'progress': current_progress.to_dict(),
                'time_estimate': time_estimate.to_dict(),

            }, room=f"progress_{session_id}")

            socketio.emit('progress_error', {
                'session_id': session_id,
                'error': str(e)

    async def check_notification_triggers(self, session_id: str, progress: OverallProgress):
        """检查通知触发条件"""
        # 阶段完成通知
        for phase in progress.phases:
            if phase.status == 'completed':
                await self.notifier.send_phase_completion_notification(
                    session_id, phase.id
                )

        # 整体完成通知
        if progress.status == 'completed':
            await self.notifier.send_completion_notification(session_id)

        # 错误通知
        if progress.status == 'failed':
            await self.notifier.send_error_notification(session_id, progress.error)

        # 超时警告
        if self.is_timeout_warning(progress):
            await self.notifier.send_timeout_warning(session_id)
```

```sql
-- 进度记录表
CREATE TABLE progress_records (
    id VARCHAR(50) PRIMARY KEY,
    session_id VARCHAR(50) NOT NULL,
    phase_id VARCHAR(50) NOT NULL,
    step_id VARCHAR(50) NOT NULL,
    progress INTEGER NOT NULL, -- 0-100
    status VARCHAR(20) NOT NULL, -- pending, processing, completed, failed, skipped
    start_time TIMESTAMP,
    end_time TIMESTAMP,
    duration INTEGER, -- 毫秒
    error_message TEXT,
    metadata TEXT, -- JSON格式的额外数据
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_session_phase (session_id, phase_id),
    INDEX idx_status (status),
    INDEX idx_created_at (created_at)
);

-- 时间预估记录表
CREATE TABLE time_estimates (
    id SERIAL PRIMARY KEY,
    session_id VARCHAR(50) NOT NULL,
    estimated_total_time INTEGER NOT NULL, -- 毫秒
    estimated_remaining_time INTEGER NOT NULL,
    confidence_level REAL NOT NULL, -- 0-1
    factors TEXT, -- JSON格式的影响因子
    actual_time INTEGER, -- 实际花费时间
    accuracy REAL, -- 预估准确度
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id) REFERENCES progress_records(session_id)
);

-- 进度通知记录表
CREATE TABLE progress_notifications (
    id SERIAL PRIMARY KEY,
    session_id VARCHAR(50) NOT NULL,
    notification_type VARCHAR(50) NOT NULL, -- phase_completion, completion, error, timeout
    message TEXT NOT NULL,
    recipient VARCHAR(200), -- 接收者
    channel VARCHAR(20) NOT NULL, -- websocket, email, sms
    sent_at TIMESTAMP,
    read_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_session_type (session_id, notification_type),
    INDEX idx_sent_at (sent_at)
);

-- 系统性能记录表
CREATE TABLE system_metrics (
    id SERIAL PRIMARY KEY,
    recorded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    cpu_usage REAL NOT NULL, -- CPU使用率
    memory_usage REAL NOT NULL, -- 内存使用率
    disk_usage REAL NOT NULL, -- 磁盘使用率
    active_tasks INTEGER NOT NULL, -- 活跃任务数
    queue_length INTEGER NOT NULL, -- 队列长度
    average_response_time INTEGER NOT NULL, -- 平均响应时间
    error_rate REAL NOT NULL -- 错误率
```
