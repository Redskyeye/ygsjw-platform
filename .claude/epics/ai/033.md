import { beforeAll, afterAll, beforeEach, afterEach } from 'vitest';
import { app } from '@/app';
import { prisma } from '@/lib/prisma';
import { redis } from '@/lib/redis';

// 测试应用实例
export const testApp = app;

// 清理数据库函数
export async function cleanupDatabase() {
  // 清理所有表，保留必要的测试数据
  const tablenames = await prisma.$queryRaw`SELECT tablename FROM pg_tables WHERE schemaname='public'`;

  for (const { tablename } of tablenames) {
    if (tablename !== '_prisma_migrations') {
      try {
        await prisma.$executeRawUnsafe(`TRUNCATE TABLE "public"."${tablename}" CASCADE;`);
      } catch (error) {
        console.log(`Note: ${tablename} doesn't exist, skipping`);
      }
    }
  }
}

// 清理Redis缓存
export async function cleanupRedis() {
  await redis.flushall();
}

// 创建测试用户
export async function createTestUser(overrides: Partial<any> = {}) {
  return prisma.user.create({
    data: {
      email: 'test@example.com',
      name: 'Test User',
      password: 'hashedpassword',
      ...overrides
    }
  });
}

// 获取测试认证token
export async function getTestAuthToken(user: any) {
  const response = await testApp.request('/api/auth/login', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      email: user.email,
      password: 'password123'
    })
  });

  return data.token;
}

// tests/integration/helpers/apiTestHelper.ts
import { testApp } from '../setup';

export class ApiTestHelper {
  private baseUrl = 'http://localhost:3000';
  private authToken: string | null = null;

  constructor(private defaultHeaders: Record<string, string> = {}) {}

  // 设置认证token
  setAuthToken(token: string) {
    this.authToken = token;
  }

  // 获取请求头
  private getHeaders(customHeaders: Record<string, string> = {}) {
    return {
      'Content-Type': 'application/json',
      ...(this.authToken && { Authorization: `Bearer ${this.authToken}` }),
      ...this.defaultHeaders,
      ...customHeaders
    };
  }

  // GET请求
  async get(path: string, headers: Record<string, string> = {}) {
    const response = await testApp.request(`${this.baseUrl}${path}`, {
      method: 'GET',
      headers: this.getHeaders(headers)
    });

    return {
      status: response.status,
      data: await response.json(),
      headers: response.headers
    };
  }

  // POST请求
  async post(path: string, data: any, headers: Record<string, string> = {}) {
    const response = await testApp.request(`${this.baseUrl}${path}`, {
      method: 'POST',
      headers: this.getHeaders(headers),
      body: JSON.stringify(data)
    });

    return {
      status: response.status,
      data: await response.json(),
      headers: response.headers
    };
  }

  // PUT请求
  async put(path: string, data: any, headers: Record<string, string> = {}) {
    const response = await testApp.request(`${this.baseUrl}${path}`, {
      method: 'PUT',
      headers: this.getHeaders(headers),
      body: JSON.stringify(data)
    });

    return {
      status: response.status,
      data: await response.json(),
      headers: response.headers
    };
  }

  // DELETE请求
  async delete(path: string, headers: Record<string, string> = {}) {
    const response = await testApp.request(`${this.baseUrl}${path}`, {
      method: 'DELETE',
      headers: this.getHeaders(headers)
    });

    return {
      status: response.status,
      data: await response.json(),
      headers: response.headers
    };
  }

  // 文件上传
  async upload(path: string, file: File, additionalData: Record<string, any> = {}) {
    const formData = new FormData();
    formData.append('file', file);

    Object.entries(additionalData).forEach(([key, value]) => {
      formData.append(key, String(value));
    });

    const response = await testApp.request(`${this.baseUrl}${path}`, {
      method: 'POST',
      headers: this.getHeaders({}),
      body: formData
    });

    return {
      status: response.status,
      data: await response.json(),
      headers: response.headers
    };
  }
}

// 创建默认的API测试助手实例
export const apiTestHelper = new ApiTestHelper();

import { describe, it, expect, beforeEach } from 'vitest';

        message: 'Hello',
        conversationId: null
      });

      const conversationId = firstResponse.data.conversationId;

      // 在同一对话中发送第二条消息
      const secondResponse = await apiTestHelper.post('/api/ai/chat', {
        message: 'How are you?',
        conversationId: conversationId
      });

      expect(secondResponse.status).toBe(200);
      expect(secondResponse.data.conversationId).toBe(conversationId);
    });

    it('should validate message content', async () => {
      const invalidData = {
        message: '', // empty message
        conversationId: null
      };

      const response = await apiTestHelper.post('/api/ai/chat', invalidData);

      expect(response.status).toBe(400);
      expect(response.data.success).toBe(false);
      expect(response.data.error).toContain('Message is required');
    });

    it('should handle rate limiting', async () => {
      // 快速发送多条消息测试限流
      const promises = Array.from({ length: 10 }, () =>
        apiTestHelper.post('/api/ai/chat', {
          message: 'Test message',
          conversationId: null
        })
      );

      const responses = await Promise.all(promises);

      // 至少有一些请求应该被限流
      const rateLimitedResponses = responses.filter(r => r.status === 429);
      expect(rateLimitedResponses.length).toBeGreaterThan(0);
    });
  });

  describe('GET /api/ai/conversations', () => {
    it('should return user conversations', async () => {
      // 创建一些对话
      await apiTestHelper.post('/api/ai/chat', { message: 'Hello 1', conversationId: null });
      await apiTestHelper.post('/api/ai/chat', { message: 'Hello 2', conversationId: null });

      const response = await apiTestHelper.get('/api/ai/conversations');

      expect(response.status).toBe(200);
      expect(response.data.success).toBe(true);
      expect(Array.isArray(response.data.conversations)).toBe(true);
      expect(response.data.conversations.length).toBeGreaterThanOrEqual(2);
    });

    it('should support pagination', async () => {
      // 创建多个对话
      for (let i = 0; i < 15; i++) {
        await apiTestHelper.post('/api/ai/chat', {
          message: `Message ${i}`,
          conversationId: null
        });
      }

      const response = await apiTestHelper.get('/api/ai/conversations?page=1&limit=10');

      expect(response.status).toBe(200);
      expect(response.data.conversations.length).toBeLessThanOrEqual(10);
      expect(response.data.pagination).toBeDefined();
      expect(response.data.pagination.total).toBeGreaterThanOrEqual(15);
    });
  });

  describe('GET /api/ai/conversations/:id/messages', () => {
    it('should return conversation messages', async () => {
      // 创建对话和消息

        message: 'Hello',
        conversationId: null
      });

      expect(response.status).toBe(200);
      expect(response.data.success).toBe(true);
      expect(Array.isArray(response.data.messages)).toBe(true);

      expect(response.status).toBe(404);
      expect(response.data.success).toBe(false);
    });
  });

      expect(response.status).toBe(200);
      // AI响应可能需要更长时间，但应该在合理范围内
      expect(responseTime).toBeLessThan(5000);
    });
  });

  describe('Concurrent Request Handling', () => {
    it('should handle multiple concurrent requests', async () => {
      const concurrentRequests = 10;

      const promises = Array.from({ length: concurrentRequests }, () =>
        apiTestHelper.get('/api/user/profile')
      );

      const startTime = Date.now();
      const responses = await Promise.all(promises);
      const endTime = Date.now();

      // 所有请求都应该成功
      responses.forEach(response => {
        expect(response.status).toBe(200);
      });

      // 总时间应该在合理范围内
      const totalTime = endTime - startTime;
      expect(totalTime).toBeLessThan(1000);
    });
  });

  describe('Memory and Resource Usage', () => {
    it('should not leak memory during repeated operations', async () => {
      const iterations = 50;

      for (let i = 0; i < iterations; i++) {
        await apiTestHelper.post('/api/ai/chat', {
          message: `Test message ${i}`,
          conversationId: null
        });
      }

      // 验证系统仍然响应正常
      const response = await apiTestHelper.get('/api/ai/conversations');
      expect(response.status).toBe(200);
      expect(response.data.conversations.length).toBe(iterations);
    });
  });
