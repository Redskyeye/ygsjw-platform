# 任务019: 模板预览系统

## 📋 任务概述

开发一个功能完整的实时模板预览系统，支持用户在生成PPT前实时预览模板效果，提供交互式编辑、多模板对比、动画效果预览等功能，确保用户对最终生成的PPT效果有直观的了解。

## 🎯 具体要求

### 19.1 实时预览功能
- **即时渲染**：用户数据变化时实时更新预览
- **多页预览**：支持多页面PPT的完整预览
- **缩放控制**：支持页面缩放（25%-200%）
- **全屏预览**：提供全屏模式查看演示效果
- **响应式布局**：适配不同屏幕尺寸的预览

### 19.2 交互式编辑
- **在线编辑**：直接在预览界面编辑文本内容
- **拖拽调整**：支持元素位置和大小调整
- **颜色定制**：实时修改模板配色方案
- **字体选择**：提供字体选择和大小调整
- **图片替换**：支持上传和替换图片元素

### 19.3 多模板对比
- **并排对比**：同时显示多个模板效果
- **快速切换**：一键切换不同模板风格
- **模板筛选**：按行业、职位、风格筛选模板
- **收藏功能**：收藏常用模板和配置
- **历史记录**：保存用户预览历史

### 19.4 动画效果预览
- **转场动画**：预览页面切换动画效果
- **元素动画**：预览元素出现和强调动画
- **时间轴控制**：控制动画播放时间和顺序
- **动画库**：提供多种预设动画效果
- **自定义动画**：支持自定义动画参数

### 19.5 导出和分享
- **草稿保存**：保存当前预览状态为草稿
- **分享链接**：生成预览分享链接
- **评论反馈**：支持预览页面的评论和标注
- **版本管理**：支持多个版本的预览对比
- **协作编辑**：支持多人同时编辑预览

```typescript
// services/PreviewEngine.ts
interface PreviewConfig {
  templateId: string;
  userData: UserData;
  customStyles: CustomStyles;
  animationSettings: AnimationSettings;
  displayMode: 'edit' | 'preview' | 'presentation';
}

interface SlidePreview {
  id: string;
  elements: PreviewElement[];
  layout: LayoutConfig;
  animations: AnimationConfig[];
  transitions: TransitionConfig;
}

class PreviewEngine {
  private canvas: HTMLCanvasElement;
  private renderer: SlideRenderer;
  private animationController: AnimationController;

  async renderPreview(config: PreviewConfig): Promise<SlidePreview[]> {
    // 渲染预览页面
    const template = await this.loadTemplate(config.templateId);
    const processedData = this.processUserData(config.userData);

    return template.slides.map(slide =>
      this.renderer.renderSlide(slide, processedData, config.customStyles)
    );
  }

  async updatePreview(
    slideId: string,
    updates: Partial<PreviewElement>
  ): Promise<void> {
    // 更新预览内容
    await this.renderer.updateElement(slideId, updates);
    this.refreshCanvas();
  }

  playAnimation(slideId: string, animationId: string): Promise<void> {
    // 播放动画效果
    return this.animationController.play(slideId, animationId);
  }
}
```

```typescript
    const updatedData = updateTextData(previewData, elementId, newText);
    setPreviewData(updatedData);
    onPreviewChange(updatedData);
  };

  const handleStyleChange = (elementId: string, newStyles: CSSProperties) => {
    const updatedStyles = updateElementStyles(elementId, newStyles);
    onPreviewChange({ ...previewData, customStyles: updatedStyles });
  };

    <div className="interactive-editor">
      <Toolbar
        mode={editMode}
        onModeChange={setEditMode}
        onSave={() => onSave(previewData)}
      />
      <SlideCanvas
        slide={slides[currentSlide]}
        selectedElement={selectedElement}
        onElementClick={handleElementClick}
        onTextChange={handleTextChange}
        onStyleChange={handleStyleChange}

        onSlideChange={setCurrentSlide}
```

```python
# services/preview_service.py
from fastapi import FastAPI, WebSocket
from services.template_loader import TemplateLoader
from services.slide_renderer import SlideRenderer
from services.animation_service import AnimationService

app = FastAPI()

class PreviewService:
    def __init__(self):
        self.template_loader = TemplateLoader()
        self.slide_renderer = SlideRenderer()
        self.animation_service = AnimationService()

    async def generate_preview(
        self,
        template_id: str,
        user_data: dict,
        custom_styles: dict = None
    ) -> dict:
        """生成预览数据"""
        try:
            # 加载模板
            template = await self.template_loader.load_template(template_id)

            # 处理用户数据
            processed_data = self.process_user_data(user_data, template)

            # 渲染页面
            slides = []
            for slide_config in template['slides']:
                slide = await self.slide_renderer.render_slide(
                    slide_config,
                    processed_data,
                    custom_styles or {}
                )
                slides.append(slide)

            return {
                'template_id': template_id,
                'slides': slides,
                'total_slides': len(slides),
                'generated_at': datetime.utcnow().isoformat()
            }

        except Exception as e:
            logger.error(f"Preview generation failed: {str(e)}")
            raise

    async def update_preview_element(
        self,
        session_id: str,
        slide_index: int,
        element_id: str,
        updates: dict
    ) -> dict:
        """更新预览元素"""
        # 获取当前预览状态
        preview_state = await self.get_preview_state(session_id)

        # 更新元素
        slide = preview_state['slides'][slide_index]
        element = self.find_element(slide, element_id)

        if element:
            element.update(updates)

            # 重新渲染页面
            updated_slide = await self.slide_renderer.rerender_slide(slide)
            preview_state['slides'][slide_index] = updated_slide

            # 保存状态
            await self.save_preview_state(session_id, preview_state)

            return updated_slide

        raise ValueError(f"Element {element_id} not found")

@app.websocket("/ws/preview/{session_id}")
async def preview_websocket(websocket: WebSocket, session_id: str):
    await websocket.accept()

    try:
        while True:
            # 接收客户端消息
            data = await websocket.receive_json()

            if data['type'] == 'update_element':
                result = await preview_service.update_preview_element(
                    session_id,
                    data['slide_index'],
                    data['element_id'],
                    data['updates']
                )
                await websocket.send_json({
                    'type': 'element_updated',
                    'slide_index': data['slide_index'],
                    'result': result
                })

            elif data['type'] == 'play_animation':
                await animation_service.play_animation(
                    session_id,
                    data['slide_index'],
                    data['animation_id']
                )
                await websocket.send_json({
                    'type': 'animation_started',
                    'slide_index': data['slide_index'],
                    'animation_id': data['animation_id']
                })

    except WebSocketDisconnect:
        logger.info(f"Preview WebSocket disconnected: {session_id}")
```

```sql
-- 预览会话表
CREATE TABLE preview_sessions (
    id VARCHAR(50) PRIMARY KEY,
    user_id INTEGER REFERENCES users(id),
    template_id VARCHAR(50) REFERENCES ppt_templates(id),
    session_data TEXT, -- JSON格式的会话数据
    preview_data TEXT, -- JSON格式的预览数据
    custom_styles TEXT, -- JSON格式的自定义样式
    status VARCHAR(20) DEFAULT 'active', -- active, saved, expired
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP
);

-- 预览操作记录表
CREATE TABLE preview_operations (
    id SERIAL PRIMARY KEY,
    session_id VARCHAR(50) REFERENCES preview_sessions(id),
    operation_type VARCHAR(50) NOT NULL, -- 'edit', 'style_change', 'animation'
    element_id VARCHAR(100),
    operation_data TEXT, -- JSON格式的操作数据
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 模板收藏表
CREATE TABLE user_template_favorites (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id),
    template_id VARCHAR(50) REFERENCES ppt_templates(id),
    custom_config TEXT, -- JSON格式的自定义配置
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, template_id)
);

-- 分享链接表
CREATE TABLE preview_shares (
    id VARCHAR(50) PRIMARY KEY,
    session_id VARCHAR(50) REFERENCES preview_sessions(id),
    share_token VARCHAR(100) UNIQUE NOT NULL,
    expires_at TIMESTAMP,
    view_count INTEGER DEFAULT 0,
    max_views INTEGER DEFAULT 100,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```
