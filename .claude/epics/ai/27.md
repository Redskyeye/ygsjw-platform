# 027 - 知识库管理系统

## 概述
实现完整的知识库管理系统，支持Markdown内容的创建、编辑、版本控制和分类管理，为用户提供结构化的知识存储和检索平台。

## 业务需求

### 核心功能
- **内容管理**：Markdown文档的创建、编辑、删除
- **版本控制**：文档版本历史和回滚功能
- **分类体系**：多层级分类和标签系统
- **协作编辑**：多人协作和审阅功能
- **内容审核**：内容审核和发布流程

### 知识库结构
1. **产品文档**：功能说明、使用指南
2. **技术文档**：API文档、开发指南
3. **操作手册**：操作流程、最佳实践
4. **FAQ**：常见问题解答
5. **故障排除**：问题诊断和解决方案
6. **更新日志**：版本更新说明
7. **培训材料**：培训文档和视频

```typescript
export interface KnowledgeArticle {

export class KnowledgeEngine {
  private contentProcessor: MarkdownProcessor;
  private versionControl: VersionControlSystem;
  private searchEngine: SearchEngine;
  private categoryManager: CategoryManager;
  private tagManager: TagManager;

  async createArticle(articleData: CreateArticleRequest): Promise<KnowledgeArticle> {
    // 1. 验证输入数据
    const validatedData = await this.validateArticleData(articleData);

    // 2. 处理Markdown内容
    const processedContent = await this.contentProcessor.process(validatedData.content);

    // 3. 提取元数据
    const extractedMetadata = await this.extractMetadata(processedContent);

    // 4. 创建文章对象
    const article: KnowledgeArticle = {
      id: generateId(),
      title: validatedData.title,
      content: processedContent.html,
      excerpt: this.generateExcerpt(processedContent.text),
      category: validatedData.category,
      tags: await this.processTags(validatedData.tags),
      author: validatedData.author,
      status: 'draft',
      visibility: validatedData.visibility || 'private',
      metadata: {
        ...extractedMetadata,
        wordCount: this.countWords(processedContent.text),
        readingTime: this.estimateReadingTime(processedContent.text),
        lastModifiedBy: validatedData.author.id
      },
      versions: [],
      createdAt: new Date(),
      updatedAt: new Date()
    };

    // 5. 创建初始版本
    const initialVersion = await this.versionControl.createVersion(
      article.id,
      validatedData.content,
      validatedData.author,
      'Initial version'
    );
    article.versions.push(initialVersion);

    // 6. 保存文章
    const savedArticle = await this.saveArticle(article);

    // 7. 更新分类和标签统计
    await this.updateCategoryStats(savedArticle.category);
    await this.updateTagStats(savedArticle.tags);

    return savedArticle;
  }

  async updateArticle(
    articleId: string,
    updateData: UpdateArticleRequest
  ): Promise<KnowledgeArticle> {
    // 1. 获取现有文章
    const existingArticle = await this.getArticle(articleId);
    if (!existingArticle) {
      throw new Error(`Article ${articleId} not found`);
    }

    // 2. 检查权限
    await this.checkEditPermission(existingArticle, updateData.author);

    // 3. 处理内容更新
    const processedContent = updateData.content
      ? await this.contentProcessor.process(updateData.content)
      : { html: existingArticle.content, text: existingArticle.content };

    // 4. 创建新版本
    const newVersion = await this.versionControl.createVersion(
      articleId,
      updateData.content || existingArticle.content,
      updateData.author,
      updateData.changeMessage || 'Updated article'
    );

    // 5. 更新文章
    const updatedArticle: KnowledgeArticle = {
      ...existingArticle,
      title: updateData.title || existingArticle.title,
      content: processedContent.html,
      excerpt: updateData.content ? this.generateExcerpt(processedContent.text) : existingArticle.excerpt,
      category: updateData.category || existingArticle.category,
      tags: updateData.tags ? await this.processTags(updateData.tags) : existingArticle.tags,
      status: updateData.status || existingArticle.status,
      metadata: {
        ...existingArticle.metadata,
        wordCount: updateData.content ? this.countWords(processedContent.text) : existingArticle.metadata.wordCount,
        readingTime: updateData.content ? this.estimateReadingTime(processedContent.text) : existingArticle.metadata.readingTime,
        lastModifiedBy: updateData.author.id
      },
      versions: [...existingArticle.versions, newVersion],
      updatedAt: new Date()
    };

    // 6. 保存更新
    const savedArticle = await this.saveArticle(updatedArticle);

    // 7. 更新统计信息
    if (updateData.category && updateData.category.id !== existingArticle.category.id) {
      await this.updateCategoryStats(existingArticle.category, -1);
      await this.updateCategoryStats(updateData.category, 1);
    }

    return savedArticle;
  }

  async searchArticles(query: SearchQuery): Promise<SearchResult> {
    // 1. 构建搜索条件
    const searchCriteria = this.buildSearchCriteria(query);

    // 2. 执行搜索
    const searchResults = await this.searchEngine.search(searchCriteria);

    // 3. 处理搜索结果
    const processedResults = await Promise.all(
      searchResults.map(result => this.processSearchResult(result))
    );

    // 4. 排序和分页
    const sortedResults = this.sortSearchResults(processedResults, query.sortBy);
    const paginatedResults = this.paginateResults(sortedResults, query.pagination);

    return {
      articles: paginatedResults,
      totalCount: searchResults.length,
      currentPage: query.pagination?.page || 1,
      totalPages: Math.ceil(searchResults.length / (query.pagination?.limit || 10)),
      facets: await this.generateSearchFacets(searchResults),
      suggestions: await this.generateSearchSuggestions(query)
    };
  }

  private async validateArticleData(data: CreateArticleRequest): Promise<CreateArticleRequest> {
    // 标题验证
    if (!data.title || data.title.trim().length === 0) {
      throw new ValidationError('Title is required');
    }

    if (data.title.length > 200) {
      throw new ValidationError('Title is too long (max 200 characters)');
    }

    // 内容验证
    if (!data.content || data.content.trim().length === 0) {
      throw new ValidationError('Content is required');
    }

    // 分类验证
    if (data.category) {
      const categoryExists = await this.categoryManager.exists(data.category.id);
      if (!categoryExists) {
        throw new ValidationError('Invalid category');
      }
    }

    return data;
  }

  private generateExcerpt(content: string, maxLength: number = 200): string {
    const plainText = content.replace(/[#*`]/g, '').trim();
    if (plainText.length <= maxLength) {
      return plainText;
    }
    return plainText.substring(0, maxLength - 3) + '...';
  }

  private countWords(content: string): number {
    return content.trim().split(/\s+/).length;
  }

  private estimateReadingTime(content: string): number {
    const wordsPerMinute = 200;
    const words = this.countWords(content);
    return Math.ceil(words / wordsPerMinute);
  }
}
```

```typescript
// src/knowledge/MarkdownProcessor.ts
export interface ProcessedContent {
  html: string;
  text: string;
  toc: TableOfContents;
  headings: Heading[];
  links: Link[];
  images: Image[];
  codeBlocks: CodeBlock[];
  metadata: ContentMetadata;
}

export interface TableOfContents {
  items: TOCItem[];
  maxDepth: number;
}

export interface TOCItem {
  id: string;
  title: string;
  level: number;
  children: TOCItem[];
}

export interface Heading {
  id: string;
  text: string;
  level: number;
  anchor: string;
}

export class MarkdownProcessor {
  private markdown: MarkdownIt;
  private pluginManager: PluginManager;

  constructor() {
    this.markdown = new MarkdownIt({
      html: true,
      linkify: true,
      typographer: true,
      breaks: true
    });

    this.setupPlugins();
  }

  private setupPlugins(): void {
    // 添加插件
    this.markdown.use(markdownItAnchor, {
      level: [1, 2, 3, 4, 5, 6],
      slugify: this.slugify,
      permalink: true,
      permalinkClass: 'header-anchor'
    });

    this.markdown.use(markdownItToc, {
      includeLevel: [1, 2, 3, 4, 5, 6],
      containerClass: 'table-of-contents',
      slugify: this.slugify
    });

    this.markdown.use(markdownItHighlight, {
      auto: true,
      langPrefix: 'language-'
    });

    this.markdown.use(markdownItTaskList, {
      enabled: true
    });

    this.markdown.use(markdownItKatex, {
      throwOnError: false,
      errorColor: '#cc0000'
    });
  }

  async process(content: string): Promise<ProcessedContent> {
    // 1. 预处理内容
    const preprocessed = await this.preprocess(content);

    // 2. 解析Markdown
    const html = this.markdown.render(preprocessed);

    // 3. 提取纯文本
    const text = this.extractText(html);

    // 4. 生成目录
    const toc = this.extractTOC(html);

    // 5. 提取结构化信息
    const headings = this.extractHeadings(html);
    const links = this.extractLinks(html);
    const images = this.extractImages(html);
    const codeBlocks = this.extractCodeBlocks(preprocessed);

    // 6. 提取元数据
    const metadata = await this.extractMetadata(preprocessed);

    return {
      html,
      text,
      toc,
      headings,
      links,
      images,
      codeBlocks,
      metadata
    };
  }

  private async preprocess(content: string): Promise<string> {
    // 处理Front Matter
    let processedContent = content;
    const frontMatter = this.extractFrontMatter(content);
    if (frontMatter) {
      processedContent = processedContent.replace(/^---[\s\S]*?---\n/, '');
    }

    // 处理自定义语法
    processedContent = await this.processCustomSyntax(processedContent);

    // 处理引用
    processedContent = this.processReferences(processedContent);

    return processedContent;
  }

  private extractFrontMatter(content: string): Record<string, any> | null {
    const match = content.match(/^---\n([\s\S]*?)\n---/);
    if (!match) return null;

    try {
      return yaml.parse(match[1]);
    } catch (error) {
      console.error('Failed to parse front matter:', error);
      return null;
    }
  }

  private extractTOC(html: string): TableOfContents {
    const tocItems: TOCItem[] = [];
    const headingRegex = /<h([1-6])[^>]*id="([^"]*)"[^>]*>(.*?)<\/h[1-6]>/g;
    let match;
    let maxDepth = 0;

    while ((match = headingRegex.exec(html)) !== null) {
      const level = parseInt(match[1]);
      const id = match[2];
      const title = this.stripHTML(match[3]);

      tocItems.push({
        id,
        title,
        level,
        children: []
      });

      maxDepth = Math.max(maxDepth, level);
    }

    // 构建层次结构
    const structuredTOC = this.buildTOCStructure(tocItems);

    return {
      items: structuredTOC,
      maxDepth
    };
  }

  private extractHeadings(html: string): Heading[] {
    const headings: Heading[] = [];
    const headingRegex = /<h([1-6])[^>]*id="([^"]*)"[^>]*>(.*?)<\/h[1-6]>/g;
    let match;

    while ((match = headingRegex.exec(html)) !== null) {
      headings.push({
        id: match[2],
        text: this.stripHTML(match[3]),
        level: parseInt(match[1]),
        anchor: match[2]
      });
    }

    return headings;
  }

  private extractLinks(html: string): Link[] {
    const links: Link[] = [];
    const linkRegex = /<a[^>]*href="([^"]*)"[^>]*>(.*?)<\/a>/g;
    let match;

    while ((match = linkRegex.exec(html)) !== null) {
      links.push({
        url: match[1],
        text: this.stripHTML(match[2]),
        isExternal: this.isExternalLink(match[1]),
        isValid: this.isValidURL(match[1])
      });
    }

    return links;
  }

  private extractImages(html: string): Image[] {
    const images: Image[] = [];
    const imageRegex = /<img[^>]*src="([^"]*)"[^>]*alt="([^"]*)"[^>]*>/g;
    let match;

    while ((match = imageRegex.exec(html)) !== null) {
      images.push({
        src: match[1],
        alt: match[2],
        isValid: this.isValidImageURL(match[1])
      });
    }

    return images;
  }

  private extractCodeBlocks(content: string): CodeBlock[] {
    const codeBlocks: CodeBlock[] = [];
    const codeBlockRegex = /```(\w+)?\n([\s\S]*?)```/g;
    let match;

    while ((match = codeBlockRegex.exec(content)) !== null) {
      codeBlocks.push({
        language: match[1] || 'text',
        code: match[2],
        lines: match[2].split('\n').length
      });
    }

    return codeBlocks;
  }

  private slugify(text: string): string {
    return text
      .toLowerCase()
      .replace(/[^\w\s-]/g, '')
      .replace(/[\s_-]+/g, '-')
      .replace(/^-+|-+$/g, '');
  }

  private stripHTML(html: string): string {
    return html.replace(/<[^>]*>/g, '');
  }

  private isExternalLink(url: string): boolean {
    return url.startsWith('http://') || url.startsWith('https://');
  }

  private isValidURL(url: string): boolean {
    try {
      new URL(url);
      return true;
    } catch {
      return false;
    }
  }

  private isValidImageURL(url: string): boolean {
    if (this.isExternalLink(url)) {
      return this.isValidURL(url);
    }
    return url.startsWith('/') || url.startsWith('./') || url.startsWith('../');
  }

  private buildTOCStructure(items: TOCItem[]): TOCItem[] {
    const result: TOCItem[] = [];
    const stack: TOCItem[] = [];

    for (const item of items) {
      while (stack.length > 0 && stack[stack.length - 1].level >= item.level) {
        stack.pop();
      }

      if (stack.length === 0) {
        result.push(item);
      } else {
        stack[stack.length - 1].children.push(item);
      }

      stack.push(item);
    }

    return result;
  }
}
```

```typescript
// src/knowledge/VersionControl.ts
export interface VersionDiff {
  changes: DiffChange[];
  summary: DiffSummary;
  conflictCount: number;
  mergeable: boolean;
}

export interface DiffChange {
  type: 'addition' | 'deletion' | 'modification';
  line: number;
  content: string;
  context?: string;
}

export interface DiffSummary {
  additions: number;
  deletions: number;
  modifications: number;
  linesAdded: number;
  linesDeleted: number;
}

export class VersionControlSystem {
  private diffEngine: DiffEngine;
  private mergeEngine: MergeEngine;

  async createVersion(
    articleId: string,
    content: string,
    author: Author,
    message: string
  ): Promise<ArticleVersion> {
    // 1. 获取最新版本
    const latestVersion = await this.getLatestVersion(articleId);
    const versionNumber = this.generateVersionNumber(latestVersion);

    // 2. 计算差异
    const diff = latestVersion
      ? await this.diffEngine.diff(latestVersion.content, content)
      : this.createInitialDiff(content);

    // 3. 创建版本对象
    const version: ArticleVersion = {
      id: generateId(),
      version: versionNumber,
      title: '', // 将在更新文章时设置
      content,
      changes: {
        diff,
        summary: this.summarizeDiff(diff),
        conflictCount: 0,
        mergeable: true
      },
      author,
      createdAt: new Date(),
      message,
      isPublished: false
    };

    // 4. 保存版本
    return await this.saveVersion(version);
  }

  async compareVersions(
    articleId: string,
    fromVersion: string,
    toVersion: string
  ): Promise<VersionDiff> {
    const [fromArticle, toArticle] = await Promise.all([
      this.getVersion(articleId, fromVersion),
      this.getVersion(articleId, toVersion)
    ]);

    if (!fromArticle || !toArticle) {
      throw new Error('Version not found');
    }

    return await this.diffEngine.diff(fromArticle.content, toArticle.content);
  }

  async revertToVersion(
    articleId: string,
    version: string,
    author: Author,
    message: string
  ): Promise<ArticleVersion> {
    const targetVersion = await this.getVersion(articleId, version);
    if (!targetVersion) {
      throw new Error(`Version ${version} not found`);
    }

    return await this.createVersion(
      articleId,
      targetVersion.content,
      author,
      message || `Revert to version ${version}`
    );
  }

  async mergeVersions(
    articleId: string,
    baseVersion: string,
    sourceVersion: string,
    targetVersion: string,
    author: Author
  ): Promise<ArticleVersion> {
    const [base, source, target] = await Promise.all([
      this.getVersion(articleId, baseVersion),
      this.getVersion(articleId, sourceVersion),
      this.getVersion(articleId, targetVersion)
    ]);

    if (!base || !source || !target) {
      throw new Error('One or more versions not found');
    }

    const mergeResult = await this.mergeEngine.merge(
      base.content,
      source.content,
      target.content
    );

    if (!mergeResult.success) {
      throw new Error(`Merge failed: ${mergeResult.conflicts.length} conflicts`);
    }

    return await this.createVersion(
      articleId,
      mergeResult.content,
      author,
      `Merge ${sourceVersion} and ${targetVersion} into ${baseVersion}`
    );
  }

  private generateVersionNumber(latestVersion?: ArticleVersion): string {
    if (!latestVersion) {
      return '1.0.0';
    }

    const parts = latestVersion.version.split('.').map(Number);
    parts[2]++; // 增加补丁版本号

    return parts.join('.');
  }

  private summarizeDiff(diff: DiffChange[]): DiffSummary {
    const summary: DiffSummary = {
      additions: 0,
      deletions: 0,
      modifications: 0,
      linesAdded: 0,
      linesDeleted: 0
    };

    for (const change of diff) {
      switch (change.type) {
        case 'addition':
          summary.additions++;
          summary.linesAdded++;
          break;
        case 'deletion':
          summary.deletions++;
          summary.linesDeleted++;
          break;
        case 'modification':
          summary.modifications++;
          summary.linesAdded++;
          summary.linesDeleted++;
          break;
      }
    }

    return summary;
  }

  private createInitialDiff(content: string): DiffChange[] {
    const lines = content.split('\n');
    return lines.map((line, index) => ({
      type: 'addition' as const,
      line: index + 1,
      content: line
    }));
  }
}
```

```tsx
  const handleSave = async () => {
    if (!validateForm()) return;

    setIsSaving(true);
    try {
      const articleData = {
        title,
        content,
        category,
        tags,
        status,
        visibility,
        author: getCurrentUser(),
        changeMessage: getChangeMessage()
      };

      let savedArticle: KnowledgeArticle;
      if (mode === 'create') {
        savedArticle = await knowledgeService.createArticle(articleData);
      } else {
        savedArticle = await knowledgeService.updateArticle(article!.id, articleData);
      }

      onSave?.(savedArticle);
    } catch (error) {
      console.error('Save failed:', error);
    } finally {
      setIsSaving(false);
    }
  };

  const handlePreview = () => {
    setPreviewMode(!previewMode);
  };

  const handleVersionRestore = async (version: ArticleVersion) => {
    if (!confirm('确定要恢复到此版本吗？当前修改将会丢失。')) {
      return;
    }

    try {
      const restoredArticle = await knowledgeService.revertToVersion(
        article!.id,
        version.version,
        getCurrentUser()
      );

      setTitle(restoredArticle.title);
      setContent(restoredArticle.content);
    } catch (error) {
      console.error('Version restore failed:', error);
    }
  };

  const validateForm = (): boolean => {
    if (!title.trim()) {
      alert('请输入标题');
      return false;
    }
    if (!content.trim()) {
      alert('请输入内容');
      return false;
    }
    return true;
  };

  return (
    <div className="article-editor">
      <div className="editor-header">
        <div className="editor-title">
          <input
            type="text"
            value={title}
            onChange={(e) => setTitle(e.target.value)}
            placeholder="输入文章标题..."
            className="title-input"
          />
        </div>

        <div className="editor-toolbar">
          <div className="toolbar-group">
            <button
              className={`toolbar-btn ${previewMode ? 'active' : ''}`}
              onClick={handlePreview}
              title="预览"
            >
              <EyeIcon />
            </button>
            <button className="toolbar-btn" title="插入图片">
              <ImageIcon />
            </button>
            <button className="toolbar-btn" title="插入链接">
              <LinkIcon />
            </button>
            <button className="toolbar-btn" title="插入代码">
              <CodeIcon />
            </button>
          </div>

          <div className="toolbar-actions">
            <button className="btn btn-secondary" onClick={onCancel}>
              取消
            </button>
            <button
              className="btn btn-primary"
              onClick={handleSave}
              disabled={isSaving}
            >
              {isSaving ? '保存中...' : '保存'}
            </button>
          </div>
        </div>
      </div>

      <div className="editor-content">
        <div className="editor-sidebar">
          <div className="editor-section">
            <h3>分类</h3>
            <CategorySelector
              value={category}
              onChange={setCategory}
              placeholder="选择分类"
            />
          </div>

          <div className="editor-section">
            <h3>标签</h3>
            <TagEditor
              value={tags}
              onChange={setTags}
              placeholder="添加标签"
            />
          </div>

          <div className="editor-section">
            <h3>状态</h3>
            <StatusSelector
              value={status}
              onChange={setStatus}
              options={[
                { value: 'draft', label: '草稿' },
                { value: 'review', label: '待审核' },
                { value: 'published', label: '已发布' }
              ]}
            />
          </div>

          <div className="editor-section">
            <h3>可见性</h3>
            <VisibilitySelector
              value={visibility}
              onChange={setVisibility}
              options={[
                { value: 'private', label: '私有' },
                { value: 'internal', label: '内部' },
                { value: 'public', label: '公开' }
              ]}
            />
          </div>

          {versions.length > 0 && (
            <div className="editor-section">
              <h3>版本历史</h3>
              <VersionHistory
                versions={versions}
                onRestore={handleVersionRestore}
                onCompare={handleVersionCompare}
              />
            </div>
          )}
        </div>

        <div className="editor-main">
          {previewMode ? (
            <PreviewPanel content={content} />
          ) : (
            <MDEditor
              ref={editorRef}
              value={content}
              onChange={(value) => setContent(value || '')}
              height={600}
              preview="edit"
              hideToolbar={false}
              visibleDragBar={false}
              textareaProps={{
                placeholder: '开始编写Markdown内容...'
              }}
              extraCommands={[
                // 自定义命令
                imageCommand,
                linkCommand,
                codeCommand,
                tableCommand
              ]}
            />
          )}
        </div>
      </div>
    </div>
  );
};
```

```tsx
// src/components/CategoryTree/CategoryTree.tsx
export const CategoryTree: React.FC<CategoryTreeProps> = ({
  categories,
  onCategorySelect,
  selectedCategory,
  level = 0
}) => {
  const [expandedCategories, setExpandedCategories] = useState<Set<string>>(new Set());

  const handleToggle = (categoryId: string) => {
    const newExpanded = new Set(expandedCategories);
    if (newExpanded.has(categoryId)) {
      newExpanded.delete(categoryId);
    } else {
      newExpanded.add(categoryId);
    }
    setExpandedCategories(newExpanded);
  };

  const handleSelect = (category: Category) => {
    onCategorySelect(category);
  };

  const renderCategory = (category: Category) => {
    const hasChildren = category.children && category.children.length > 0;
    const isExpanded = expandedCategories.has(category.id);
    const isSelected = selectedCategory?.id === category.id;

    return (
      <div key={category.id} className="category-item">
        <div
          className={`category-node ${isSelected ? 'selected' : ''}`}
          style={{ paddingLeft: `${level * 20}px` }}
        >
          <div className="category-content">
            {hasChildren && (
              <button
                className="expand-btn"
                onClick={() => handleToggle(category.id)}
              >
                <ChevronIcon direction={isExpanded ? 'down' : 'right'} />
              </button>
            )}

            <button
              className="category-btn"
              onClick={() => handleSelect(category)}
            >
              {category.icon && (
                <span className="category-icon">{category.icon}</span>
              )}
              <span className="category-name">{category.name}</span>
              <span className="article-count">({category.articleCount})</span>
            </button>
          </div>
        </div>

        {hasChildren && isExpanded && (
          <div className="category-children">
            {category.children!.map(child => (
              <CategoryTree
                key={child.id}
                categories={[child]}
                onCategorySelect={onCategorySelect}
                selectedCategory={selectedCategory}
                level={level + 1}
              />
            ))}
          </div>
        )}
      </div>
    );
  };

  return (
    <div className="category-tree">
      {categories.map(category => renderCategory(category))}
    </div>
  );
};
```
