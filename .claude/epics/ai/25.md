# 025 - 问题确认系统

## 概述
实现智能问题确认系统，通过AI诊断分析用户提出的问题，提供准确的问题确认和解决方案建议，确保用户理解问题本质并采取正确行动。

## 业务需求

### 核心功能
- **问题诊断**：AI智能分析用户问题描述
- **问题分类**：自动分类问题类型和优先级
- **确认机制**：多轮对话确认问题理解
- **解决方案建议**：提供针对性解决方案
- **历史记录**：保存问题和解决方案历史

### 问题类型分类
1. **技术问题**：系统故障、功能异常、性能问题
2. **业务问题**：流程疑问、操作指导、最佳实践
3. **数据问题**：数据异常、分析结果、质量问题
4. **用户问题**：账户问题、权限问题、使用疑问
5. **集成问题**：API集成、第三方服务、数据同步
6. **性能问题**：响应慢、资源占用、优化需求
7. **其他问题**：功能建议、反馈、其他咨询

```typescript
// src/diagnosis/DiagnosisEngine.ts
export interface UserQuestion {
  id: string;
  content: string;
  context?: QuestionContext;
  attachments?: Attachment[];
  timestamp: Date;
  userId: string;
  sessionId: string;
}

export interface QuestionContext {
  currentModule?: string;
  recentActions?: string[];
  systemState?: SystemState;
  userHistory?: UserHistory;
  environmentInfo?: EnvironmentInfo;
}

export interface DiagnosisResult {
  questionId: string;
  problemType: ProblemType;
  category: ProblemCategory;
  severity: Severity;
  confidence: number;
  understanding: ProblemUnderstanding;
  suggestedActions: SuggestedAction[];
  clarifyingQuestions?: ClarifyingQuestion[];
  estimatedResolution: ResolutionEstimate;
  metadata: DiagnosisMetadata;
}

export class DiagnosisEngine {
  private nlpProcessor: NLPProcessor;
  private patternMatcher: PatternMatcher;
  private knowledgeBase: KnowledgeBase;
  private contextAnalyzer: ContextAnalyzer;

  async diagnoseQuestion(question: UserQuestion): Promise<DiagnosisResult> {
    // 1. 预处理问题内容
    const processedContent = await this.preprocessQuestion(question);

    // 2. 提取关键信息
    const keyInformation = await this.extractKeyInformation(processedContent);

    // 3. 分析上下文
    const contextAnalysis = await this.analyzeContext(question.context);

    // 4. 模式匹配
    const patternMatches = await this.matchPatterns(keyInformation, contextAnalysis);

    // 5. 知识库查询
    const knowledgeResults = await this.queryKnowledgeBase(keyInformation);

    // 6. 生成诊断结果
    const diagnosis = await this.generateDiagnosis(
      keyInformation,
      contextAnalysis,
      patternMatches,
      knowledgeResults
    );

    // 7. 确认理解
    const confirmedDiagnosis = await this.confirmUnderstanding(diagnosis, question);

    return confirmedDiagnosis;
  }

  private async preprocessQuestion(question: UserQuestion): Promise<ProcessedContent> {
    // 文本清洗和标准化
    const cleanedText = this.nlpProcessor.cleanText(question.content);

    // 语言检测
    const language = this.nlpProcessor.detectLanguage(cleanedText);

    // 情感分析
    const sentiment = this.nlpProcessor.analyzeSentiment(cleanedText);

    // 关键词提取
    const keywords = await this.nlpProcessor.extractKeywords(cleanedText);

    // 实体识别
    const entities = await this.nlpProcessor.extractEntities(cleanedText);

    // 意图识别
    const intent = await this.nlpProcessor.identifyIntent(cleanedText);

    return {
      originalText: question.content,
      cleanedText,
      language,
      sentiment,
      keywords,
      entities,
      intent,
      attachments: question.attachments
    };
  }

  private async extractKeyInformation(content: ProcessedContent): Promise<KeyInformation> {
    return {
      mainProblem: this.identifyMainProblem(content),
      symptoms: this.identifySymptoms(content),
      affectedComponents: this.identifyAffectedComponents(content),
      errorMessages: this.extractErrorMessages(content),
      expectedBehavior: this.extractExpectedBehavior(content),
      actualBehavior: this.extractActualBehavior(content),
      reproductionSteps: this.extractReproductionSteps(content),
      urgency: this.assessUrgency(content)
    };
  }

  private async matchPatterns(
    keyInfo: KeyInformation,
    context: ContextAnalysis
  ): Promise<PatternMatch[]> {
    const patterns = await this.patternMatcher.findMatchingPatterns(
      keyInfo,
      context
    );

    return patterns
      .filter(match => match.confidence > 0.6)
      .sort((a, b) => b.confidence - a.confidence)
      .slice(0, 5); // 返回前5个匹配
  }

  private async generateDiagnosis(
    keyInfo: KeyInformation,
    context: ContextAnalysis,
    patterns: PatternMatch[],
    knowledge: KnowledgeQueryResult
  ): Promise<DiagnosisResult> {
    // 问题类型分类
    const problemType = this.classifyProblemType(keyInfo, patterns, knowledge);

    // 严重程度评估
    const severity = this.assessSeverity(keyInfo, context, patterns);

    // 生成理解
    const understanding = this.generateUnderstanding(keyInfo, patterns, knowledge);

    // 建议行动
    const suggestedActions = await this.generateSuggestedActions(
      problemType,
      severity,
      understanding
    );

    // 生成澄清问题
    const clarifyingQuestions = this.generateClarifyingQuestions(understanding);

    return {
      questionId: keyInfo.questionId,
      problemType,
      category: this.mapToCategory(problemType),
      severity,
      confidence: this.calculateConfidence(patterns, knowledge),
      understanding,
      suggestedActions,
      clarifyingQuestions,
      estimatedResolution: this.estimateResolution(problemType, severity),
      metadata: {
        processedAt: new Date(),
        patterns: patterns.map(p => p.id),
        knowledgeSources: knowledge.sources,
        processingTime: 0
      }
    };
  }
}
```

```typescript
// src/confirmation/ConfirmationManager.ts
export interface ConfirmationDialogue {
  id: string;
  questionId: string;
  initialUnderstanding: ProblemUnderstanding;
  currentStep: ConfirmationStep;
  steps: ConfirmationStep[];
  responses: StepResponse[];
  isCompleted: boolean;
  finalUnderstanding?: ProblemUnderstanding;
}

export interface ConfirmationStep {
  id: string;
  type: 'understanding' | 'clarification' | 'verification' | 'confirmation';
  content: StepContent;
  expectedResponse?: ExpectedResponse;
  options?: ResponseOption[];
  isRequired: boolean;
}

export interface StepContent {
  title: string;
  description: string;
  details?: string[];
  examples?: Example[];
  visualAids?: VisualAid[];
}

export class ConfirmationManager {
  async initiateConfirmation(
    diagnosis: DiagnosisResult,
    question: UserQuestion
  ): Promise<ConfirmationDialogue> {
    const dialogue: ConfirmationDialogue = {
      id: generateId(),
      questionId: question.id,
      initialUnderstanding: diagnosis.understanding,
      currentStep: this.createInitialStep(diagnosis),
      steps: [],
      responses: [],
      isCompleted: false
    };

    dialogue.steps = this.buildConfirmationSteps(diagnosis, dialogue);

    return dialogue;
  }

  async processResponse(
    dialogue: ConfirmationDialogue,
    response: StepResponse
  ): Promise<ConfirmationDialogue> {
    // 验证响应
    const validatedResponse = await this.validateResponse(
      dialogue.currentStep,
      response
    );

    // 添加响应记录
    dialogue.responses.push(validatedResponse);

    // 更新理解
    const updatedUnderstanding = await this.updateUnderstanding(
      dialogue,
      validatedResponse
    );

    // 确定下一步
    const nextStep = this.determineNextStep(dialogue, validatedResponse);

    if (nextStep) {
      dialogue.currentStep = nextStep;
    } else {
      // 对话完成
      dialogue.isCompleted = true;
      dialogue.finalUnderstanding = updatedUnderstanding;
    }

    return dialogue;
  }

  private createInitialStep(diagnosis: DiagnosisResult): ConfirmationStep {
    return {
      id: 'initial-understanding',
      type: 'understanding',
      content: {
        title: '我理解您的问题是...',
        description: this.formatUnderstandingDescription(diagnosis.understanding),
        details: this.extractKeyDetails(diagnosis),
        examples: this.getRelevantExamples(diagnosis)
      },
      expectedResponse: {
        type: 'confirmation',
        prompt: '这个理解正确吗？'
      },
      options: [
        { id: 'correct', label: '正确，继续', value: true },
        { id: 'partially-correct', label: '部分正确，需要补充', value: 'partial' },
        { id: 'incorrect', label: '不正确，需要重新理解', value: false }
      ],
      isRequired: true
    };
  }

  private buildConfirmationSteps(
    diagnosis: DiagnosisResult,
    dialogue: ConfirmationDialogue
  ): ConfirmationStep[] {
    const steps: ConfirmationStep[] = [];

    // 如果有澄清问题，添加澄清步骤
    if (diagnosis.clarifyingQuestions && diagnosis.clarifyingQuestions.length > 0) {
      diagnosis.clarifyingQuestions.forEach((question, index) => {
        steps.push({
          id: `clarification-${index}`,
          type: 'clarification',
          content: {
            title: question.title,
            description: question.description,
            details: question.details
          },
          expectedResponse: {
            type: question.responseType,
            prompt: question.prompt
          },
          options: question.options,
          isRequired: question.required
        });
      });
    }

    // 添加验证步骤
    steps.push({
      id: 'verification',
      type: 'verification',
      content: {
        title: '问题验证',
        description: '让我确认一下我对您问题的理解',
        details: this.generateVerificationDetails(diagnosis)
      },
      expectedResponse: {
        type: 'confirmation',
        prompt: '您是否确认这个理解？'
      },
      isRequired: true
    });

    // 添加最终确认步骤
    steps.push({
      id: 'final-confirmation',
      type: 'confirmation',
      content: {
        title: '确认解决方案',
        description: '基于对您问题的理解，我建议以下解决方案',
        details: this.formatSolutionDetails(diagnosis.suggestedActions)
      },
      expectedResponse: {
        type: 'confirmation',
        prompt: '您是否接受这个解决方案？'
      },
      options: [
        { id: 'accept', label: '接受方案', value: 'accept' },
        { id: 'modify', label: '需要修改', value: 'modify' },
        { id: 'reject', label: '不接受，需要其他方案', value: 'reject' }
      ],
      isRequired: true
    });

    return steps;
  }
}
```

```typescript
// src/solutions/SolutionEngine.ts
export interface Solution {
  id: string;
  title: string;
  description: string;
  type: SolutionType;
  complexity: Complexity;
  estimatedTime: number;
  requiredResources: Resource[];
  steps: SolutionStep[];
  successRate: number;
  riskLevel: RiskLevel;
  prerequisites: string[];
  alternatives: string[];
  verificationCriteria: VerificationCriterion[];
}

export interface SolutionStep {
  id: string;
  order: number;
  title: string;
  description: string;
  instructions: Instruction[];
  expectedOutcome: string;
  verificationMethod: VerificationMethod;
  estimatedTime: number;
  requiredSkills: string[];
}

export class SolutionEngine {
  private solutionDatabase: SolutionDatabase;
  private solutionRanker: SolutionRanker;
  private personalizationEngine: PersonalizationEngine;

  async generateSolutions(
    diagnosis: DiagnosisResult,
    userContext: UserContext
  ): Promise<Solution[]> {
    // 1. 查询基础解决方案
    const baseSolutions = await this.solutionDatabase.querySolutions(
      diagnosis.problemType,
      diagnosis.severity
    );

    // 2. 个性化筛选
    const personalizedSolutions = await this.personalizeSolutions(
      baseSolutions,
      diagnosis,
      userContext
    );

    // 3. 排序解决方案
    const rankedSolutions = await this.solutionRanker.rankSolutions(
      personalizedSolutions,
      diagnosis,
      userContext
    );

    // 4. 生成详细步骤
    const detailedSolutions = await Promise.all(
      rankedSolutions.map(solution => this.generateDetailedSteps(solution, diagnosis))
    );

    return detailedSolutions.slice(0, 5); // 返回前5个方案
  }

  private async personalizeSolutions(
    solutions: Solution[],
    diagnosis: DiagnosisResult,
    userContext: UserContext
  ): Promise<Solution[]> {
    return solutions.filter(solution => {
      // 检查用户技能匹配
      if (!this.checkSkillCompatibility(solution, userContext.skills)) {
        return false;
      }

      // 检查资源可用性
      if (!this.checkResourceAvailability(solution, userContext.resources)) {
        return false;
      }

      // 检查先决条件
      if (!this.checkPrerequisites(solution, userContext.history)) {
        return false;
      }

      return true;
    });
  }

  private async generateDetailedSteps(
    solution: Solution,
    diagnosis: DiagnosisResult
  ): Promise<Solution> {
    const detailedSteps: SolutionStep[] = [];

    for (const baseStep of solution.steps) {
      const detailedStep = await this.enrichStep(baseStep, diagnosis);
      detailedSteps.push(detailedStep);
    }

    return {
      ...solution,
      steps: detailedSteps,
      verificationCriteria: this.generateVerificationCriteria(solution, diagnosis)
    };
  }

  private enrichStep(
    step: SolutionStep,
    diagnosis: DiagnosisResult
  ): SolutionStep {
    return {
      ...step,
      instructions: this.generateDetailedInstructions(step, diagnosis),
      verificationMethod: this.enhanceVerificationMethod(step, diagnosis),
      contextualTips: this.generateContextualTips(step, diagnosis),
      commonPitfalls: this.identifyCommonPitfalls(step, diagnosis),
      alternativeApproaches: this.suggestAlternativeApproaches(step, diagnosis)
    };
  }
}
```

```tsx
export const ProblemConfirmation: React.FC<ProblemConfirmationProps> = ({
  initialQuestion,
  onDiagnosisComplete,
  onSolutionSelected
}) => {
  const [currentDialogue, setCurrentDialogue] = useState<ConfirmationDialogue | null>(null);
  const [currentStep, setCurrentStep] = useState<ConfirmationStep | null>(null);
  const [isProcessing, setIsProcessing] = useState(false);
  const [diagnosis, setDiagnosis] = useState<DiagnosisResult | null>(null);

  const diagnosisEngine = useMemo(() => new DiagnosisEngine(), []);
  const confirmationManager = useMemo(() => new ConfirmationManager(), []);

  useEffect(() => {
    initializeDiagnosis();
  }, [initialQuestion]);

  const initializeDiagnosis = async () => {
    setIsProcessing(true);
    try {
      const diagnosisResult = await diagnosisEngine.diagnoseQuestion(initialQuestion);
      setDiagnosis(diagnosisResult);

      const dialogue = await confirmationManager.initiateConfirmation(
        diagnosisResult,
        initialQuestion
      );
      setCurrentDialogue(dialogue);
      setCurrentStep(dialogue.currentStep);
    } catch (error) {
      console.error('Diagnosis failed:', error);
    } finally {
      setIsProcessing(false);
    }
  };

  const handleStepResponse = async (response: StepResponse) => {
    if (!currentDialogue) return;

    setIsProcessing(true);
    try {
      const updatedDialogue = await confirmationManager.processResponse(
        currentDialogue,
        response
      );
      setCurrentDialogue(updatedDialogue);
      setCurrentStep(updatedDialogue.currentStep);

      if (updatedDialogue.isCompleted) {
        onDiagnosisComplete?.(updatedDialogue.finalUnderstanding);
      }
    } catch (error) {
      console.error('Response processing failed:', error);
    } finally {
      setIsProcessing(false);
    }
  };

  if (isProcessing && !currentStep) {
    return <LoadingView message="正在分析您的问题..." />;
  }

  return (
    <div className="problem-confirmation">
      <div className="confirmation-header">
        <h2>问题确认与诊断</h2>
        <ProgressBar
          current={currentDialogue?.responses.length || 0}
          total={currentDialogue?.steps.length || 1}
        />
      </div>

      <div className="confirmation-content">
        {currentStep && (
          <ConfirmationStep
            step={currentStep}
            onResponse={handleStepResponse}
            disabled={isProcessing}
          />
        )}

        {diagnosis && (
          <DiagnosisSummary
            diagnosis={diagnosis}
            isVisible={!currentDialogue?.currentStep}
          />
        )}
      </div>

      <div className="confirmation-sidebar">
        <ProblemHistory userId={initialQuestion.userId} />
        <RelatedQuestions problemType={diagnosis?.problemType} />
      </div>
    </div>
  );
};
```

```tsx
// src/components/ProblemConfirmation/ConfirmationStep.tsx
interface ConfirmationStepProps {
  step: ConfirmationStep;
  onResponse: (response: StepResponse) => void;
  disabled?: boolean;
}

export const ConfirmationStep: React.FC<ConfirmationStepProps> = ({
  step,
  onResponse,
  disabled = false
}) => {
  const [selectedResponse, setSelectedResponse] = useState<any>(null);
  const [textResponse, setTextResponse] = useState('');
  const [isValid, setIsValid] = useState(false);

  useEffect(() => {
    setIsValid(validateResponse(step, selectedResponse, textResponse));
  }, [selectedResponse, textResponse, step]);

  const handleSubmit = () => {
    if (!isValid) return;

    const response: StepResponse = {
      stepId: step.id,
      type: step.expectedResponse?.type || 'text',
      value: selectedResponse || textResponse,
      timestamp: new Date(),
      metadata: {}
    };

    onResponse(response);
  };

  return (
    <div className="confirmation-step">
      <div className="step-header">
        <div className="step-indicator">
          <span className="step-type">{getStepTypeLabel(step.type)}</span>
          <h3>{step.content.title}</h3>
        </div>
        {step.isRequired && (
          <span className="required-indicator">必填</span>
        )}
      </div>

      <div className="step-content">
        <p className="step-description">{step.content.description}</p>

        {step.content.details && step.content.details.length > 0 && (
          <div className="step-details">
            <h4>详细信息：</h4>
            <ul>
              {step.content.details.map((detail, index) => (
                <li key={index}>{detail}</li>
              ))}
            </ul>
          </div>
        )}

        {step.content.examples && step.content.examples.length > 0 && (
          <div className="step-examples">
            <h4>示例：</h4>
            {step.content.examples.map((example, index) => (
              <ExampleCard key={index} example={example} />
            ))}
          </div>
        )}

        {step.content.visualAids && step.content.visualAids.length > 0 && (
          <div className="visual-aids">
            {step.content.visualAids.map((aid, index) => (
              <VisualAid key={index} aid={aid} />
            ))}
          </div>
        )}
      </div>

      <div className="step-response">
        {renderResponseInput(step, selectedResponse, setSelectedResponse, textResponse, setTextResponse)}

        <div className="response-actions">
          <button
            className="btn btn-primary"
            onClick={handleSubmit}
            disabled={!isValid || disabled}
          >
            {step.type === 'confirmation' ? '确认' : '下一步'}
          </button>

          {step.type !== 'understanding' && (
            <button
              className="btn btn-secondary"
              onClick={() => onResponse({ stepId: step.id, type: 'skip', value: null, timestamp: new Date() })}
              disabled={disabled}
            >
              跳过
            </button>
          )}
        </div>
      </div>
    </div>
  );
};

function renderResponseInput(
  step: ConfirmationStep,
  selectedResponse: any,
  setSelectedResponse: (value: any) => void,
  textResponse: string,
  setTextResponse: (value: string) => void
) {
  switch (step.expectedResponse?.type) {
    case 'confirmation':
      return (
        <div className="confirmation-options">
          {step.options?.map((option) => (
            <label key={option.id} className="option-card">
              <input
                type="radio"
                name="confirmation"
                value={option.value}
                checked={selectedResponse === option.value}
                onChange={(e) => setSelectedResponse(e.target.value)}
              />
              <span className="option-label">{option.label}</span>
            </label>
          ))}
        </div>
      );

    case 'multiple-choice':
      return (
        <div className="multiple-choice-options">
          {step.options?.map((option) => (
            <label key={option.id} className="checkbox-card">
              <input
                type="checkbox"
                value={option.value}
                checked={Array.isArray(selectedResponse) && selectedResponse.includes(option.value)}
                onChange={(e) => {
                  const current = Array.isArray(selectedResponse) ? selectedResponse : [];
                  if (e.target.checked) {
                    setSelectedResponse([...current, option.value]);
                  } else {
                    setSelectedResponse(current.filter((v: any) => v !== option.value));
                  }
                }}
              />
              <span className="option-label">{option.label}</span>
            </label>
          ))}
        </div>
      );

    case 'rating':
      return (
        <div className="rating-input">
          <div className="rating-scale">
            {[1, 2, 3, 4, 5].map((value) => (
              <button
                key={value}
                className={`rating-point ${selectedResponse === value ? 'selected' : ''}`}
                onClick={() => setSelectedResponse(value)}
              >
                {value}
              </button>
            ))}
          </div>
          <div className="rating-labels">
            <span>不满意</span>
            <span>非常满意</span>
          </div>
        </div>
      );

    case 'text':
    default:
      return (
        <div className="text-input">
          <textarea
            value={textResponse}
            onChange={(e) => setTextResponse(e.target.value)}
            placeholder={step.expectedResponse?.prompt || '请输入您的回答...'}
            rows={4}
          />
        </div>
      );
  }
}
```

### Pinia Store
```typescript
// src/stores/confirmationStore.ts
export const useConfirmationStore = defineStore('confirmation', {
  state: () => ({
    currentDialogue: null as ConfirmationDialogue | null,
    diagnosisHistory: [] as DiagnosisResult[],
    solutionHistory: [] as Solution[],
    userPreferences: {
      language: 'zh-CN',
      confirmationStyle: 'detailed' as 'concise' | 'detailed' | 'interactive',
      autoProceed: false,
      showExamples: true
    },
    statistics: {
      totalQuestions: 0,
      correctDiagnoses: 0,
      successfulSolutions: 0,
      averageConfirmationTime: 0
    }
  }),

  getters: {
    currentStep: (state) => state.currentDialogue?.currentStep,
    progress: (state) => {
      if (!state.currentDialogue) return 0;
      return (state.currentDialogue.responses.length / state.currentDialogue.steps.length) * 100;
    },
    canProceed: (state) => {
      if (!state.currentDialogue) return false;
      const currentResponse = state.currentDialogue.responses[state.currentDialogue.responses.length - 1];
      return currentResponse && currentResponse.value !== null;
    }
  },

  actions: {
    async startDiagnosis(question: UserQuestion): Promise<DiagnosisResult> {
      const engine = new DiagnosisEngine();
      const diagnosis = await engine.diagnoseQuestion(question);

      this.diagnosisHistory.unshift(diagnosis);
      this.statistics.totalQuestions++;

      return diagnosis;
    },

    async startConfirmation(diagnosis: DiagnosisResult, question: UserQuestion): Promise<ConfirmationDialogue> {
      const manager = new ConfirmationManager();
      const dialogue = await manager.initiateConfirmation(diagnosis, question);

      this.currentDialogue = dialogue;
      return dialogue;
    },

    async processResponse(response: StepResponse): Promise<ConfirmationDialogue> {
      if (!this.currentDialogue) {
        throw new Error('No active dialogue');
      }

      const manager = new ConfirmationManager();
      const updatedDialogue = await manager.processResponse(this.currentDialogue, response);

      this.currentDialogue = updatedDialogue;

      if (updatedDialogue.isCompleted) {
        this.statistics.correctDiagnoses++;
        this.currentDialogue = null;
      }

      return updatedDialogue;
    },

    recordSolution(solution: Solution, feedback: SolutionFeedback): void {
      this.solutionHistory.unshift({
        ...solution,
        appliedAt: new Date(),
        feedback
      });

      if (feedback.successful) {
        this.statistics.successfulSolutions++;
      }
    },

    updatePreferences(preferences: Partial<typeof this.userPreferences>): void {
      this.userPreferences = { ...this.userPreferences, ...preferences };
    },

    clearHistory(): void {
      this.diagnosisHistory = [];
      this.solutionHistory = [];
      this.currentDialogue = null;
    }
  }
});
```
