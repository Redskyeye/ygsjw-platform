# 023 - 4D分析可视化系统

## 概述

实现创新的4D分析可视化系统，通过雷达图展示多维度分析结果，为用户提供直观、交互式的数据洞察体验。

## 业务需求

### 核心功能

- **4D雷达图**：展示四维分析数据的雷达图可视化
- **交互式探索**：支持数据点交互、缩放、旋转
- **动态更新**：实时数据更新和动画效果
- **多视图切换**：支持2D、3D、4D视图切换
- **数据导出**：支持图表导出为多种格式

### 分析维度定义

1. **技术维度**：技术可行性、复杂度评估
2. **业务维度**：商业价值、市场需求
3. **风险维度**：风险等级、不确定性
4. **时间维度**：时间成本、实施周期

```typescript
export class WebGL4DRenderer {
  private canvas: HTMLCanvasElement;
  private gl: WebGLRenderingContext;
  private shaderProgram: WebGLProgram;
  private vertexBuffer: WebGLBuffer;
  private indexBuffer: WebGLBuffer;

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    this.gl = canvas.getContext('webgl2')!;
    this.initializeWebGL();
  }

  render4DRadar(data: DataPoint4D[], config: RenderConfig): void {
    this.prepareVertexData(data);
    this.setupShaders();
    this.renderFrame();
  }

  private prepareVertexData(data: DataPoint4D[]): void {
    // 将4D数据转换为3D顶点坐标
    const vertices = data.map(point => this.project4DTo3D(point.dimensions));
    this.vertexBuffer = this.createBuffer(vertices);
  }

  private project4DTo3D(dimensions4D: DataPoint4D['dimensions']): Float32Array {
    // 使用四维到三维的投影算法
    const { technical, business, risk, time } = dimensions4D;

    // 立体投影公式
    const w = 1 + time / 100; // 时间维度作为权重
    const x = (technical - 50) * w;
    const y = (business - 50) * w;
    const z = (risk - 50) * w;

    return new Float32Array([x, y, z]);
  }
}
```

```typescript
// src/analysis/AnalysisEngine.ts
export class AnalysisEngine4D {
  private dimensions = ['technical', 'business', 'risk', 'time'];

  analyzeData(inputData: any[]): DataPoint4D[] {
    return inputData.map(item => this.analyzeItem(item));
  }

  private analyzeItem(item: any): DataPoint4D {
    return {
      id: this.generateId(),
      name: item.name,
      dimensions: {
        technical: this.calculateTechnicalScore(item),
        business: this.calculateBusinessScore(item),
        risk: this.calculateRiskScore(item),
        time: this.calculateTimeScore(item),
      },
      metadata: {
        category: this.determineCategory(item),
        priority: this.determinePriority(item),
        description: item.description || '',
        tags: item.tags || [],
      },
      color: this.generateColor(item),
      opacity: 0.8,
    };
  }

  private calculateTechnicalScore(item: any): number {
    // 技术复杂度评分算法
    const factors = {
      complexity: item.complexity || 0,
      expertise: item.requiredExpertise || 0,
      resources: item.requiredResources || 0,
      dependencies: item.dependencies?.length || 0,
    };

    // 使用加权算法计算技术分数
    const weights = {
      complexity: 0.4,
      expertise: 0.3,
      resources: 0.2,
      dependencies: 0.1,
    };
    return this.calculateWeightedScore(factors, weights);
  }

  calculateTrends(data: DataPoint4D[]): TrendAnalysis {
    // 分析数据趋势
    return {
      direction: this.determineTrendDirection(data),
      velocity: this.calculateTrendVelocity(data),
      acceleration: this.calculateTrendAcceleration(data),
      predictions: this.generatePredictions(data),
    };
  }
}
```

```typescript
// src/visualization/D3RadarChart.ts
import * as d3 from 'd3';

export class D3RadarChart {
  private svg: d3.Selection<SVGSVGElement, unknown, null, undefined>;
  private width: number;
  private height: number;
  private radius: number;

  constructor(
    container: HTMLElement,
    width: number = 600,
    height: number = 600
  ) {
    this.width = width;
    this.height = height;
    this.radius = Math.min(width, height) / 2 - 40;

    this.svg = d3
      .select(container)
      .append('svg')
      .attr('width', width)
      .attr('height', height)
      .append('g')
      .attr('transform', `translate(${width / 2},${height / 2})`);
  }

  render(data: DataPoint4D[]): void {
    this.clear();
    this.drawGrid();
    this.drawAxes();
    this.drawData(data);
    this.addInteractivity();
  }

  private drawGrid(): void {
    const levels = 5;
    const angleSlice = (Math.PI * 2) / 4; // 4 dimensions

    // 绘制同心圆网格
    for (let level = 1; level <= levels; level++) {
      const levelFactor = this.radius * (level / levels);

      this.svg
        .append('circle')
        .attr('r', levelFactor)
        .attr('class', 'grid-circle')
        .style('fill', 'none')
        .style('stroke', '#e0e0e0')
        .style('stroke-opacity', 0.5);
    }

    // 绘制径向线
    for (let angle = 0; angle < 4; angle++) {
      const x = levelFactor * Math.cos(angleSlice * angle - Math.PI / 2);
      const y = levelFactor * Math.sin(angleSlice * angle - Math.PI / 2);

      this.svg
        .append('line')
        .attr('x1', 0)
        .attr('y1', 0)
        .attr('x2', x)
        .attr('y2', y)
        .attr('class', 'axis-line')
        .style('stroke', '#e0e0e0')
        .style('stroke-opacity', 0.5);
    }
  }

  private drawData(data: DataPoint4D[]): void {
    const angleSlice = (Math.PI * 2) / 4;

    data.forEach((point, index) => {
      const lineData = this.createLineData(point.dimensions, angleSlice);

      // 绘制数据区域
      this.svg
        .append('polygon')
        .datum(lineData)
        .attr('class', 'data-area')
        .attr('points', d => d.map(p => `${p.x},${p.y}`).join(' '))
        .style('fill', point.color || this.defaultColor(index))
        .style('fill-opacity', point.opacity || 0.3)
        .style('stroke', point.color || this.defaultColor(index))
        .style('stroke-width', 2);

      // 绘制数据点
      this.svg
        .selectAll('.data-point')
        .data(lineData)
        .enter()
        .append('circle')
        .attr('class', 'data-point')
        .attr('cx', d => d.x)
        .attr('cy', d => d.y)
        .attr('r', 4)
        .style('fill', point.color || this.defaultColor(index))
        .style('stroke', 'white')
        .style('stroke-width', 2);
    });
  }

  private createLineData(
    dimensions: DataPoint4D['dimensions'],
    angleSlice: number
  ) {
    return Object.entries(dimensions).map(([key, value], index) => {
      const angle = angleSlice * index - Math.PI / 2;
      const x = this.radius * (value / 100) * Math.cos(angle);
      const y = this.radius * (value / 100) * Math.sin(angle);
      return { x, y, dimension: key, value };
    });
  }
}
```

```typescript
// src/visualization/ThreeDRadar.ts
import * as THREE from 'three';

export class ThreeDRadarEnhancement {
  private scene: THREE.Scene;
  private camera: THREE.PerspectiveCamera;
  private renderer: THREE.WebGLRenderer;
  private animationId: number;

  constructor(container: HTMLElement) {
    this.initThreeJS(container);
  }

  private initThreeJS(container: HTMLElement): void {
    // 场景设置
    this.scene = new THREE.Scene();
    this.scene.background = new THREE.Color(0xf0f0f0);

    // 相机设置
    this.camera = new THREE.PerspectiveCamera(
      75,
      container.clientWidth / container.clientHeight,
      0.1,
      1000
    );
    this.camera.position.set(0, 5, 10);
    this.camera.lookAt(0, 0, 0);

    // 渲染器设置
    this.renderer = new THREE.WebGLRenderer({ antialias: true });
    this.renderer.setSize(container.clientWidth, container.clientHeight);
    this.renderer.shadowMap.enabled = true;
    container.appendChild(this.renderer.domElement);

    // 添加灯光
    this.addLighting();

    // 添加控制器
    this.addControls();
  }

  render4DData(data: DataPoint4D[]): void {
    // 清除现有对象
    this.clearScene();

    // 创建4D数据可视化
    data.forEach((point, index) => {
      this.create4DShape(point, index);
    });

    // 开始动画
    this.animate();
  }

  private create4DShape(point: DataPoint4D, index: number): void {
    // 创建4D形状（使用颜色和大小表示第4维）
    const geometry = new THREE.SphereGeometry(
      0.5 + point.dimensions.time / 100, // 时间维度影响大小
      32,
      32
    );

    const material = new THREE.MeshPhongMaterial({
      color: new THREE.Color(point.color || this.getColorByIndex(index)),
      opacity: point.opacity || 0.8,
      transparent: true,
    });

    const mesh = new THREE.Mesh(geometry, material);

    // 根据前3个维度设置位置
    mesh.position.set(
      (point.dimensions.technical - 50) / 10,
      (point.dimensions.business - 50) / 10,
      (point.dimensions.risk - 50) / 10
    );

    mesh.castShadow = true;
    mesh.receiveShadow = true;

    // 添加交互数据
    mesh.userData = { point, index };

    this.scene.add(mesh);
  }

  private animate(): void {
    this.animationId = requestAnimationFrame(() => this.animate());

    // 旋转场景
    this.scene.rotation.y += 0.005;

    // 渲染
    this.renderer.render(this.scene, this.camera);
  }
}
```

```typescript
// src/animation/Transitions.ts
export class ChartAnimations {
  private duration = 600;
  private easing = 'cubic-bezier(0.4, 0.0, 0.2, 1)';

  animateDataTransition(
    fromData: DataPoint4D[],
    toData: DataPoint4D[],
    onUpdate: (interpolatedData: DataPoint4D[]) => void
  ): void {
    const startTime = Date.now();

    const animate = () => {
      const elapsed = Date.now() - startTime;
      const progress = Math.min(elapsed / this.duration, 1);
      const easedProgress = this.ease(progress);

      const interpolatedData = fromData.map((fromPoint, index) => {
        const toPoint = toData[index];
        return this.interpolateDataPoint(fromPoint, toPoint, easedProgress);
      });

      onUpdate(interpolatedData);

      if (progress < 1) {
        requestAnimationFrame(animate);
      }
    };

    animate();
  }

  private interpolateDataPoint(
    from: DataPoint4D,
    to: DataPoint4D,
    progress: number
  ): DataPoint4D {
    return {
      ...from,
      dimensions: {
        technical: this.lerp(
          from.dimensions.technical,
          to.dimensions.technical,
          progress
        ),
        business: this.lerp(
          from.dimensions.business,
          to.dimensions.business,
          progress
        ),
        risk: this.lerp(from.dimensions.risk, to.dimensions.risk, progress),
        time: this.lerp(from.dimensions.time, to.dimensions.time, progress),
      },
      opacity: this.lerp(from.opacity || 0.8, to.opacity || 0.8, progress),
    };
  }

  private ease(t: number): number {
    // 缓动函数
    return t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;
  }

  private lerp(start: number, end: number, progress: number): number {
    return start + (end - start) * progress;
  }
}
```

### 渲染优化

```typescript
// src/optimization/Renderer.ts
export class OptimizedRenderer {
  private frameCache = new Map<string, any>();
  private lastRenderTime = 0;
  private targetFPS = 60;
  private frameInterval = 1000 / this.targetFPS;

  render(data: DataPoint4D[], container: HTMLElement): void {
    const now = Date.now();

    // 帧率限制
    if (now - this.lastRenderTime < this.frameInterval) {
      return;
    }

    // 检查缓存
    const cacheKey = this.generateCacheKey(data);
    if (this.frameCache.has(cacheKey)) {
      this.renderFromCache(cacheKey, container);
      return;
    }

    // 执行渲染
    this.performRender(data, container);
    this.lastRenderTime = now;
  }

  private generateCacheKey(data: DataPoint4D[]): string {
    return data
      .map(point => `${point.id}-${JSON.stringify(point.dimensions)}`)
      .join('|');
  }

  private performRender(data: DataPoint4D[], container: HTMLElement): void {
    // 使用requestAnimationFrame优化渲染
    requestAnimationFrame(() => {
      this.clearContainer(container);
      this.renderData(data, container);
      this.cacheRender(data, container);
    });
  }
}
```

```typescript
// src/optimization/Data.ts
export class DataOptimizer {
  // 数据采样
  sampleData(data: DataPoint4D[], maxPoints: number = 100): DataPoint4D[] {
    if (data.length <= maxPoints) {
      return data;
    }

    const step = Math.ceil(data.length / maxPoints);
    return data.filter((_, index) => index % step === 0);
  }

  // 数据聚合
  aggregateData(data: DataPoint4D[], aggregationLevel: number): DataPoint4D[] {
    const groups = this.groupByCategory(data);

    return Object.entries(groups).map(([category, points]) => ({
      id: `aggregated-${category}`,
      name: `${category} (聚合)`,
      dimensions: this.averageDimensions(points),
      metadata: {
        category,
        priority: this.determineAggregatedPriority(points),
        description: `${points.length} 个数据点的聚合结果`,
        tags: this.aggregateTags(points),
      },
      color: this.getColorByCategory(category),
      opacity: 0.6,
    }));
  }

  private averageDimensions(points: DataPoint4D[]): DataPoint4D['dimensions'] {
    const sum = points.reduce(
      (acc, point) => ({
        technical: acc.technical + point.dimensions.technical,
        business: acc.business + point.dimensions.business,
        risk: acc.risk + point.dimensions.risk,
        time: acc.time + point.dimensions.time,
      }),
      { technical: 0, business: 0, risk: 0, time: 0 }
    );

    const count = points.length;
    return {
      technical: Math.round(sum.technical / count),
      business: Math.round(sum.business / count),
      risk: Math.round(sum.risk / count),
      time: Math.round(sum.time / count),
    };
  }
}
```
