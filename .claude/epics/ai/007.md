  code: string

// src/lib/api/response.ts
import type { ApiResponse } from './types'

export class ApiResponseBuilder {
  static success<T>(data: T, meta?: any): ApiResponse<T> {
    return {
      success: true,
      data,
      meta: {
        timestamp: new Date().toISOString(),
        requestId: crypto.randomUUID(),
        ...meta,
      },
    }
  }

  static error(
    code: string,
    message: string,
    statusCode = 500,
    details?: any
  ): ApiResponse {
    return {
      success: false,
      error: {
        code,
        message,
        details,
      },
      meta: {
        timestamp: new Date().toISOString(),
        requestId: crypto.randomUUID(),
      },
    }
  }

  static paginated<T>(
    data: T[],
    page: number,
    limit: number,
    total: number
  ): ApiResponse<T[]> {
    return this.success(data, {
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    })
  }
}

import { auth } from '@/lib/auth'

export async function withAuth(request: NextRequest) {
  const session = await auth()

  if (!session) {
    return NextResponse.json(
      ApiResponseBuilder.error('UNAUTHORIZED', 'Authentication required', 401),
      { status: 401 }
    )
  }

  return session
}

export async function withRole(request: NextRequest, requiredRoles: string[]) {
  const session = await withAuth(request)

  if (session instanceof NextResponse) {
    return session
  }

  if (!requiredRoles.includes(session.user.role)) {
    return NextResponse.json(
      ApiResponseBuilder.error('FORBIDDEN', 'Insufficient permissions', 403),
      { status: 403 }
    )
  }

  return session
}

// src/lib/api/middleware/validation.ts
import { NextRequest, NextResponse } from 'next/server'
import { z } from 'zod'
import { ApiResponseBuilder } from '../response'

export function withValidation<T>(schema: z.ZodSchema<T>) {
  return async (request: NextRequest) => {
    try {
      const body = await request.json()
      const validatedData = schema.parse(body)
      return validatedData
    } catch (error) {
      if (error instanceof z.ZodError) {
        return NextResponse.json(
          ApiResponseBuilder.error(
            'VALIDATION_ERROR',
            'Invalid request data',
            400,
            error.errors
          ),
          { status: 400 }
        )
      }
      throw error
    }
  }
}

// src/lib/api/middleware/cors.ts
import { NextRequest, NextResponse } from 'next/server'

export function withCors(request: NextRequest) {
  const origin = request.headers.get('origin')
  const allowedOrigins = [
    'http://localhost:3000',
    'https://epic-ai.com',
    'https://www.epic-ai.com',
  ]

  const corsHeaders = {
    'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type, Authorization',
    'Access-Control-Max-Age': '86400',
  }

  if (allowedOrigins.includes(origin || '')) {
    corsHeaders['Access-Control-Allow-Origin'] = origin!
  }

  if (request.method === 'OPTIONS') {
    return new NextResponse(null, { status: 200, headers: corsHeaders })
  }

  return corsHeaders
}

// src/lib/api/middleware/rate-limit.ts
import { NextRequest, NextResponse } from 'next/server'

const rateLimitStore = new Map<string, { count: number; resetTime: number }>()

export function withRateLimit(limit: number, windowMs: number) {
  return (request: NextRequest) => {
    const clientId = request.ip || 'anonymous'
    const now = Date.now()
    const windowStart = now - windowMs

    // 清理过期记录
    for (const [key, value] of rateLimitStore.entries()) {
      if (value.resetTime < now) {
        rateLimitStore.delete(key)
      }
    }

    const clientData = rateLimitStore.get(clientId)

    if (!clientData || clientData.resetTime < now) {
      // 新窗口或过期窗口
      rateLimitStore.set(clientId, {
        count: 1,
        resetTime: now + windowMs,
      })
      return null
    }

    if (clientData.count >= limit) {
      return NextResponse.json(
        ApiResponseBuilder.error(
          'RATE_LIMIT_EXCEEDED',
          'Too many requests',
          429,
          {
            limit,
            windowMs,
            retryAfter: Math.ceil((clientData.resetTime - now) / 1000),
          }
        ),
        {
          status: 429,
          headers: {
            'Retry-After': Math.ceil((clientData.resetTime - now) / 1000).toString(),
          },
        }
      )
    }

    clientData.count++
    return null
  }
}

// src/app/api/v1/projects/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { ApiResponseBuilder } from '@/lib/api/response'
import { withAuth, withRole } from '@/lib/api/middleware/auth'
import { withValidation } from '@/lib/api/middleware/validation'
import { withRateLimit } from '@/lib/api/middleware/rate-limit'
import { withCors } from '@/lib/api/middleware/cors'
import { createProjectSchema } from '@/lib/api/schemas/project'
import { ProjectService } from '@/lib/services/project.service'
import { z } from 'zod'

const querySchema = z.object({
  page: z.coerce.number().min(1).default(1),
  limit: z.coerce.number().min(1).max(100).default(20),
  status: z.string().optional(),
  priority: z.string().optional(),
  search: z.string().optional(),
})

export async function GET(request: NextRequest) {
  try {
    // CORS处理
    const corsHeaders = withCors(request)
    if (request.method === 'OPTIONS') {
      return new NextResponse(null, { status: 200, headers: corsHeaders })
    }

    // 速率限制
    const rateLimitResponse = withRateLimit(100, 15 * 60 * 1000)(request)
    if (rateLimitResponse) return rateLimitResponse

    // 认证
    const session = await withAuth(request)
    if (session instanceof NextResponse) return session

    // 查询参数验证
    const { page, limit, status, priority, search } = querySchema.parse(
      Object.fromEntries(request.nextUrl.searchParams)
    )

    // 构建查询条件
    const where = {
      OR: [
        { ownerId: session.user.id },
        { members: { some: { userId: session.user.id } } },
      ],
      ...(status && { status }),
      ...(priority && { priority }),
      ...(search && {
        OR: [
          { name: { contains: search } },
          { description: { contains: search } },
        ],
      }),
    }

    // 获取项目列表
    const [projects, total] = await Promise.all([
      ProjectService.findMany({
        where,
        skip: (page - 1) * limit,
        take: limit,
        orderBy: { updatedAt: 'desc' },
      }),
      ProjectService.count({ where }),
    ])

    return NextResponse.json(
      ApiResponseBuilder.paginated(projects, page, limit, total),
      { headers: corsHeaders }
    )
  } catch (error) {
    console.error('Projects GET error:', error)
    return NextResponse.json(
      ApiResponseBuilder.error('INTERNAL_ERROR', 'Internal server error'),
      { status: 500, headers: withCors(request) }
    )
  }
}

export async function POST(request: NextRequest) {
  try {
    const corsHeaders = withCors(request)
    if (request.method === 'OPTIONS') {
      return new NextResponse(null, { status: 200, headers: corsHeaders })
    }

    const rateLimitResponse = withRateLimit(20, 60 * 1000)(request)
    if (rateLimitResponse) return rateLimitResponse

    const session = await withAuth(request)
    if (session instanceof NextResponse) return session

    const validatedData = await withValidation(createProjectSchema)(request)
    if (validatedData instanceof NextResponse) return validatedData

    const project = await ProjectService.create({
      ...validatedData,
      owner: { connect: { id: session.user.id } },
      members: {
        create: {
          userId: session.user.id,
          role: 'OWNER',
        },
      },
    })

    return NextResponse.json(
      ApiResponseBuilder.success(project),
      {
        status: 201,
        headers: corsHeaders,
      }
    )
  } catch (error) {
    console.error('Projects POST error:', error)
    return NextResponse.json(
      ApiResponseBuilder.error('INTERNAL_ERROR', 'Internal server error'),
      { status: 500, headers: withCors(request) }
    )
  }
}

// src/app/api/v1/projects/[id]/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { ApiResponseBuilder } from '@/lib/api/response'
import { withAuth } from '@/lib/api/middleware/auth'
import { withValidation } from '@/lib/api/middleware/validation'
import { updateProjectSchema } from '@/lib/api/schemas/project'
import { ProjectService } from '@/lib/services/project.service'

interface RouteParams {
  params: Promise<{ id: string }>
}

export async function GET(request: NextRequest, { params }: RouteParams) {
  try {
    const corsHeaders = withCors(request)
    if (request.method === 'OPTIONS') {
      return new NextResponse(null, { status: 200, headers: corsHeaders })
    }

    const session = await withAuth(request)
    if (session instanceof NextResponse) return session

    const { id } = await params

    const project = await ProjectService.findById(id, true)
    if (!project) {
      return NextResponse.json(
        ApiResponseBuilder.error('NOT_FOUND', 'Project not found', 404),
        { status: 404, headers: corsHeaders }
      )
    }

    // 检查权限
    const hasAccess = project.ownerId === session.user.id ||
      project.members.some(member => member.userId === session.user.id)

    if (!hasAccess) {
      return NextResponse.json(
        ApiResponseBuilder.error('FORBIDDEN', 'Access denied', 403),
        { status: 403, headers: corsHeaders }
      )
    }

    return NextResponse.json(
      ApiResponseBuilder.success(project),
      { headers: corsHeaders }
    )
  } catch (error) {
    console.error('Project GET error:', error)
    return NextResponse.json(
      ApiResponseBuilder.error('INTERNAL_ERROR', 'Internal server error'),
      { status: 500, headers: withCors(request) }
    )
  }
}

export async function PUT(request: NextRequest, { params }: RouteParams) {
  try {
    const corsHeaders = withCors(request)
    if (request.method === 'OPTIONS') {
      return new NextResponse(null, { status: 200, headers: corsHeaders })
    }

    const session = await withAuth(request)
    if (session instanceof NextResponse) return session

    const { id } = await params
    const validatedData = await withValidation(updateProjectSchema)(request)
    if (validatedData instanceof NextResponse) return validatedData

    // 检查项目是否存在且用户有权限
    const project = await ProjectService.findById(id)
    if (!project) {
      return NextResponse.json(
        ApiResponseBuilder.error('NOT_FOUND', 'Project not found', 404),
        { status: 404, headers: corsHeaders }
      )
    }

    // 只有项目所有者和管理员可以编辑
    const hasEditPermission = project.ownerId === session.user.id ||
      project.members.some(member =>
        member.userId === session.user.id &&
        ['OWNER', 'ADMIN'].includes(member.role)
      )

    if (!hasEditPermission) {
      return NextResponse.json(
        ApiResponseBuilder.error('FORBIDDEN', 'Insufficient permissions', 403),
        { status: 403, headers: corsHeaders }
      )
    }

    const updatedProject = await ProjectService.update(id, validatedData)

    return NextResponse.json(
      ApiResponseBuilder.success(updatedProject),
      { headers: corsHeaders }
    )
  } catch (error) {
    console.error('Project PUT error:', error)
    return NextResponse.json(
      ApiResponseBuilder.error('INTERNAL_ERROR', 'Internal server error'),
      { status: 500, headers: withCors(request) }
    )
  }
}

export async function DELETE(request: NextRequest, { params }: RouteParams) {
  try {
    const corsHeaders = withCors(request)
    if (request.method === 'OPTIONS') {
      return new NextResponse(null, { status: 200, headers: corsHeaders })
    }

    const session = await withAuth(request)
    if (session instanceof NextResponse) return session

    const { id } = await params

    const project = await ProjectService.findById(id)
    if (!project) {
      return NextResponse.json(
        ApiResponseBuilder.error('NOT_FOUND', 'Project not found', 404),
        { status: 404, headers: corsHeaders }
      )
    }

    // 只有项目所有者可以删除
    if (project.ownerId !== session.user.id) {
      return NextResponse.json(
        ApiResponseBuilder.error('FORBIDDEN', 'Only project owner can delete', 403),
        { status: 403, headers: corsHeaders }
      )
    }

    await ProjectService.delete(id)

    return NextResponse.json(
      ApiResponseBuilder.success(null),
      { headers: corsHeaders }
    )
  } catch (error) {
    console.error('Project DELETE error:', error)
    return NextResponse.json(
      ApiResponseBuilder.error('INTERNAL_ERROR', 'Internal server error'),
      { status: 500, headers: withCors(request) }
    )
  }
}

// src/app/api/v1/health/route.ts
import { NextResponse } from 'next/server'
import { ApiResponseBuilder } from '@/lib/api/response'
import { healthCheck, getDatabaseStats } from '@/lib/db-utils'

export async function GET() {
  try {
    const [dbHealth, dbStats] = await Promise.all([
      healthCheck(),
      getDatabaseStats(),
    ])

    const health = {
      status: 'healthy',
      timestamp: new Date().toISOString(),
      version: process.env.npm_package_version || '1.0.0',
      environment: process.env.NODE_ENV || 'development',
      services: {
        database: dbHealth,
      },
      stats: dbStats,
      uptime: process.uptime(),
    }

    // 如果任何服务不健康，返回503状态
    const isHealthy = Object.values(health.services).every(
      (service: any) => service.status === 'healthy'
    )

    return NextResponse.json(
      ApiResponseBuilder.success(health),
      { status: isHealthy ? 200 : 503 }
    )
  } catch (error) {
    return NextResponse.json(
      ApiResponseBuilder.error('HEALTH_CHECK_FAILED', 'Health check failed'),
      { status: 503 }
    )
  }
}
