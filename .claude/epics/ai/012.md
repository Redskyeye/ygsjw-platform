            }}
          />
        )}
      </div>
    </div>
  );

// CoreProjectSelector.tsx
interface CoreProjectSelectorProps {
  projects: Project[];
  selectedProjects: string[];
  maxSelections: number;
  onSelectionChange: (selectedIds: string[]) => void;
}

const CoreProjectSelector: React.FC<CoreProjectSelectorProps> = ({
  projects,
  selectedProjects,
  maxSelections,
  onSelectionChange
}) => {
  const [searchTerm, setSearchTerm] = useState('');
  const [filterTags, setFilterTags] = useState<string[]>([]);

  const filteredProjects = useMemo(() => {
    return projects.filter(project => {
      const matchesSearch = project.name.toLowerCase().includes(searchTerm.toLowerCase()) ||
                          project.description.toLowerCase().includes(searchTerm.toLowerCase());
      const matchesTags = filterTags.length === 0 ||
                         filterTags.some(tag => project.technologies.includes(tag));
      return matchesSearch && matchesTags;
    });
  }, [projects, searchTerm, filterTags]);

  return (
    <div className="core-project-selector">
      <div className="selector-header">
        <h3>选择核心项目 (最多{maxSelections}个)</h3>
        <div className="search-controls">
          <SearchInput
            value={searchTerm}
            onChange={setSearchTerm}
            placeholder="搜索项目名称或描述..."
          />
          <TagFilter
            availableTags={getAllTechnologies(projects)}
            selectedTags={filterTags}
            onSelectionChange={setFilterTags}
          />
        </div>
      </div>

      <div className="projects-grid">
        {filteredProjects.map(project => (
          <ProjectCard
            key={project.id}
            project={project}
            isSelected={selectedProjects.includes(project.id)}
            canSelect={selectedProjects.length < maxSelections}
            onSelect={() => toggleProjectSelection(project.id)}
          />
        ))}
      </div>

      <div className="selected-projects">
        <h4>已选择的项目</h4>
        <SelectedProjectsList
          projects={projects.filter(p => selectedProjects.includes(p.id))}
          onRemove={(projectId) => removeProjectSelection(projectId)}
          onReorder={handleProjectReorder}
        />
      </div>
    </div>
  );
};

interface ConfigPreviewProps {
  originalData: ParsedResume;
  config: ResumeConfig;

}

const ConfigPreview: React.FC<ConfigPreviewProps> = ({
  originalData,
  config,
  onConfigChange
}) => {
  const [previewData, setPreviewData] = useState<PreviewData | null>(null);
  const [isGenerating, setIsGenerating] = useState(false);

  useEffect(() => {
    generatePreview();
  }, [config]);

  const generatePreview = async () => {
    setIsGenerating(true);
    try {
      const preview = await generateConfigPreview(originalData, config);
      setPreviewData(preview);

      console.error('Preview generation failed:', error);
    } finally {
      setIsGenerating(false);
    }

  return (
    <div className="config-preview">
      <div className="preview-header">
        <h3>配置预览</h3>

      </div>

      <div className="preview-content">
        {isGenerating ? (
          <PreviewSkeleton />

          <ResumePreview data={previewData} />

          </div>
        )}
      </div>

// configSlice.ts
const configSlice = createSlice({
  name: 'resumeConfig',
  initialState: {
    currentConfig: getDefaultConfig(),
    presets: [],
    validationErrors: [],
    isDirty: false
  },
  reducers: {
    updateConfig: (state, action) => {
      state.currentConfig = { ...state.currentConfig, ...action.payload };
      state.isDirty = true;
      validateConfig(state.currentConfig);
    },

    resetConfig: (state) => {
      state.currentConfig = getDefaultConfig();
      state.isDirty = false;
      state.validationErrors = [];
    },

    loadPreset: (state, action) => {
      state.currentConfig = action.payload.config;
      state.isDirty = false;
      validateConfig(state.currentConfig);
    },

    savePreset: (state, action) => {
      const { name, description } = action.payload;
      const newPreset: ConfigPreset = {
        id: generateId(),
        name,
        description,
        config: state.currentConfig,
        tags: extractTags(state.currentConfig),
        isDefault: false,
        createdAt: new Date(),
        usageCount: 0
      };
      state.presets.push(newPreset);
    }
  }
});

// 验证配置的中间件
const configValidationMiddleware: Middleware = (store) => (next) => (action) => {
  const result = next(action);

  if (action.type.startsWith('config/')) {
    const state = store.getState() as RootState;
    const errors = validateResumeConfig(state.config.currentConfig);

    if (errors.length > 0) {

// configValidator.ts
export function validateResumeConfig(config: ResumeConfig): ValidationError[] {
  const errors: ValidationError[] = [];

  // 验证工作经历配置
  if (config.experience.enabled.length === 0) {
    errors.push({
      field: 'experience.enabled',
      message: '至少需要选择一个工作经历',
      type: 'warning'
    });
  }

  // 验证项目选择
  if (config.projects.core.length > 5) {
    errors.push({
      field: 'projects.core',
      message: '核心项目数量不能超过5个',
      type: 'error'
    });
  }

  // 验证技能配置
  const enabledSkills = [
    ...config.skills.technical.filter(s => s.enabled),
    ...config.skills.soft.filter(s => s.enabled)
  ];

  if (enabledSkills.length < 5) {
    errors.push({
      field: 'skills',
      message: '建议至少选择5个技能展示',
      type: 'warning'
    });
  }

  // 验证目标岗位匹配度
  if (config.targetJob && config.focusAreas.length === 0) {
    errors.push({
      field: 'focusAreas',
      message: '设置目标岗位时，建议配置重点关注的技能领域',
      type: 'warning'
    });
  }

  return errors;
}

export function generateConfigSuggestions(config: ResumeConfig): ConfigSuggestion[] {
  const suggestions: ConfigSuggestion[] = [];

  // 根据目标岗位生成建议
  if (config.targetJob) {
    const requiredSkills = extractRequiredSkills(config.targetJob);
    const missingSkills = findMissingSkills(config.skills, requiredSkills);

    if (missingSkills.length > 0) {
      suggestions.push({
        type: 'skills',
        priority: 'high',
        title: '补充目标岗位要求的技能',
        description: `建议添加以下技能：${missingSkills.join(', ')}`,
        action: () => addMissingSkills(missingSkills)
      });
    }
  }

  // 根据经验水平生成建议
  if (config.mode === 'fresh_graduate') {
    suggestions.push({
      type: 'experience',
      priority: 'medium',
      title: '突出项目和实习经历',
      description: '作为应届生，建议重点展示学术项目和实习经验',
      action: () => emphasizeProjects()
    });
  }

  return suggestions;
}
